    43→  const [stage, setStage] = useState<Stage>("home");

    53→  const toastIdRef = useRef(0);
    54→

    57→    setToast({ id, text, sub });

    60→

    62→    setLog((cur) => [{ ...item, id: ++logIdRef.current }, ...cur].slice(0, 12));
    63→  }, []);
    64→

    68→      const r = await fetch("/api/learning");
    69→      if (!r.ok) return;
    70→      const d = await r.json();
    71→      setLearnedApps(new Set<string>((d.learnedRules ?? []).map((x: { appName: string }) => x.appName)));
    72→    } catch { /* 静默 */ }
    73→  }, []);
    74→  useEffect(() => {
    75→    let alive = true;
    76→    fetch("/api/learning")
    77→      .then((r) => (r.ok ? r.json() : null))
    78→      .then((d) => {
    79→        if (!alive || !d) return;
    80→        setLearnedApps(new Set<string>((d.learnedRules ?? []).map((x: { appName: string }) => x.appName)));
    81→      })
    82→      .catch(() => {});
    83→    return () => { alive = false; };
    84→  }, []);
    85→

    92→          method: "POST",
    93→          headers: { "Content-Type": "application/json" },

    95→            appKey: app.key,
    96→            kind: "block",
    97→            action: p.action,
    98→            latencyMs: p.totalMs,
    99→            confidence: p.confidence / 100,
   100→            scenario: p.scenario,
   101→            adType: app.adType,
   102→            note,
   103→          }),
   104→        });

   110→          }
   111→        }
   112→      } catch { /* 静默 */ }

   114→    [app, pushToast, refreshLearning],
   115→  );
   116→

   120→    setPlan(null);
   121→    setFakePhase(null);
   122→    setStage("launch");
   123→  }, []);

   132→            name: app.name,
   133→            category: app.category,
   134→            adType: app.adType,
   135→            guard: app.guard,
   136→            ruleName: app.ruleName,
   137→            learnedRuleName: learnedApps.has(app.name) ? `${app.name} 学习指纹` : null,
   138→            traits: app.traits ?? null,
   139→          },
   140→          { preempt: isPreempt },
   141→        );
   142→        setPlan(p);

   144→          // P0 预拦截：广告未及呈现，直接进入拦截成功层

   146→          setBlocks((n) => n + 1);
   147→          setMsSum((s) => s + p.totalMs);

   149→          void postEvent(p, "P0 预拦截（学习指纹）");

   152→          setStage("ad");
   153→        }

   155→      return () => clearTimeout(t);
   156→    }
   157→    if (stage === "flash") {
   158→      const t = setTimeout(() => setStage("main"), 1250);
   159→      return () => clearTimeout(t);
   160→    }
   161→  }, [stage, app, learnedApps, postEvent, pushLog]);
   162→

   176→          setMsSum((s) => s + plan.totalMs);

   202→          void postEvent(plan, plan.ruleName);

   215→    );
   216→  }, [pushToast]);

   220→    setStage("home");

   222→    setFakePhase(null);
   223→  }, []);

   231→        <div className="relative w-full max-w-[360px] rounded-[2.6rem] border-[10px] border-zinc-800 bg-zinc-800 shadow-2xl shadow-zinc-900/25 ring-1 ring-zinc-700">

   258→          同一应用第二次启动将触发 <span className="font-bold text-violet-600">P0 预拦截</span>：窗口附加阶段直接阻断，广告未及呈现
   259→        </p>

   261→
   262→      {/* ═══ 右：引擎面板 ═══ */}
