   322→    public static ScanFind walk(NodeView root, Ctx ctx) {
   323→        ScanFind f = new ScanFind();
   324→        if (root == null) return f;
   325→        /* ═══ v5.0.0 小程序/团购容器保护（L0，纵深防御第三层）═══
   326→         * 抖音团购券/微信/支付宝/QQ 小程序容器内：树扫描直接返回空——
   327→         * 任何跳过候选/角落候选/热区信号都不收集（服务侧事件入口已拦，
   328→         * 这里兜住定时扫描/补扫针等迟到路径）。「广告未关」的容器加载页
   329→         * 文案（「跳过广告」字样）绝不触发点击（issue 抖音团购券根治）。 */
   330→        if (MiniProgramGuard.isMiniProgramWindow(ctx.pkg, ctx.activityId)) return f;
   331→        String rootCls = nz(root.cls()).toLowerCase();
   332→        f.dialogForm = rootCls.contains("dialog") || rootCls.contains("popup");
   333→        int[] budget = {ctx.budgetNodes};
   334→        visit(root, 0, budget, f, ctx);
   335→        // v3.8.0 候选排序：真键优先——① 自身可点（无祖先中转）② 文案更短（精确「跳过」
   336→        // 优先于「5s后可跳过」类长文案）③ 面积更小。防「第 1 候选是广告图内假文本」
   337→        try {
   338→            java.util.Collections.sort(f.skipCands, (a, b) -> {
   339→                if (a.selfClick != b.selfClick) return a.selfClick ? -1 : 1;
   340→                int la = a.label == null ? 0 : a.label.length();
   341→                int lb = b.label == null ? 0 : b.label.length();
   342→                if (la != lb) return la - lb;
   343→                return Double.compare(areaRatio(a.tapBounds, ctx), areaRatio(b.tapBounds, ctx));
   344→            });
   345→        } catch (Throwable ignored) {
   346→        }
   347→        return f;
   348→    }

   350→    private static void visit(NodeView node, int depth, int[] budget, ScanFind f, Ctx ctx) {
   351→        if (node == null || depth > 32 || budget[0] <= 0) return;
   352→        budget[0]--;
   353→        f.nodeCount++;
   354→        try {
   355→            String text = nz(node.text());
   356→            String desc = nz(node.desc());
   357→            String label = text.length() > 0 ? text : desc;
   358→            String cls = nz(node.cls()).toLowerCase();
   359→            boolean vis = node.visible();
   360→            if (label.length() > LONG_TEXT_LEN) f.longText = true; // 长文本=正常内容页（v3.5.0）
   361→            if (node.editable() || cls.contains("edittext")) f.editPresent = true;
   362→            if (node.scrollable() || cls.contains("scrollview") || cls.contains("recyclerview")
   363→                    || cls.contains("viewpager") || cls.contains("listview")) f.scrollPresent = true;
   364→
   365→            int[] b = node.bounds();
   366→            boolean clickable = node.clickable();
   367→            double areaRatio = areaRatio(b, ctx);
   368→
   369→            // 全屏热区：可点且 ≥55% 屏幕
   370→            if (!f.hotZone && clickable && areaRatio >= HOT_ZONE_AREA) f.hotZone = true;
   371→
   372→            if (label.length() > 0) {
   373→                if (label.length() <= 14 && matchesSkip(label)) {
   374→                    /* ═══ v4.5.0 乱点根治：兜底路径同款形状门（与 RuleEngine.build
   375→                     * 一致）——菜单/抽屉列表行绝不入候选：issue#42-44 实锤抽屉
   376→                     * 菜单「跳过」项被点（rule 路径已挡，此处堵 walk→decide 兜底
   377→                     * 路径）。被拒节点也不设 fakeSkip（菜单行非陷阱信号）。 ═══ */
   378→                    boolean shapeOk;
   379→                    if (ctx.sdkAdWin || ctx.learnedCls) {
   380→                        shapeOk = true; // 高置信广告窗豁免
   381→                    } else if (b == null || ctx.screenW <= 0) {
   382→                        shapeOk = true; // 形状未知：保守放行（绝不错杀真开屏）
   383→                    } else {
   384→                        double wRatio = (double) (b[2] - b[0]) / ctx.screenW;
   385→                        shapeOk = wRatio <= (clickable ? 0.70 : 0.55) && !isMenuListRow(node);
   386→                    }
   387→                    if (!shapeOk) {
   388→                        // 菜单/全宽行：不入 skipCands、不设 fakeSkip、不进倒计时
   389→                    } else {
   390→                        int cd = parseCountdown(label);
   391→                        if (cd > f.countdownN) f.countdownN = cd; // v3.8.0 倒计时解析（「跳过 3」→3，服务侧门控用）
   392→                        NodeView click = resolveClickable(node, 4);
   393→                        if (click != null && isSmallClick(click, ctx) && vis) {
   394→                            if (!f.foundSkip) {
   395→                                f.foundSkip = true;
   396→                                f.skipNode = click;
   397→                                // v3.6.0：手势点击用【原标签控件】坐标（点得准），
   398→                                // 不再用可点祖先中心（祖先常是全屏容器，点中心=点开广告，
   399→                                // 真机「已点击跳过但广告纹丝不动」根因之一）；performAction 仍走祖先
   400→                                f.skipBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   401→                                f.skipLabel = label;
   402→                            }
   403→                            // v3.7.0 多候选收集：广告图内常嵌假「跳过」文本（或同窗多个跳过键），
   404→                            // 复验第 N 次点第 N 个候选——不再同一坐标点四遍（issue#10 根因）
   405→                            if (f.skipCands.size() < SKIP_CANDS_MAX) {
   406→                                SkipCand c = new SkipCand();
   407→                                c.clickNode = click;
   408→                                c.tapBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   409→                                c.label = label;
   410→                                c.selfClick = (click == node);
   411→                                f.skipCands.add(c);
   412→                            }
   413→                        } else {
   414→                            f.fakeSkip = true; // 大面积/不可点 = 图内假跳过陷阱
   415→                            // v3.8.0：假跳过≠不可点——大容器可点祖先吞掉 performAction 时，
   416→                            // 标签自身坐标仍是有效触点（GKD clickCenter 方法论）：手势点标签
   417→                            // 位置，由应用按坐标分发到真跳过键。仅收角落形态小标签（防中央诱饵）
   418→                            if (vis && b != null && areaRatio(b, ctx) < FAKE_TAP_AREA_MAX
   419→                                    && learnableSkipShape(b, ctx.screenW, ctx.screenH)
   420→                                    && f.fakeCands.size() < SKIP_CANDS_MAX) {
   421→                                SkipCand c = new SkipCand();
   422→                                c.clickNode = click;
   423→                                c.tapBounds = b;
   424→                                c.label = label;
   425→                                c.gestureOnly = true;
   426→                                f.fakeCands.add(c);
   427→                            }
   428→                        }
   429→                    }
   430→                } else if (label.length() <= 22 && matchesAny(label, GYRO_HINTS)) {
   431→                    f.gyroHint = true;
   432→                } else if (label.length() <= 6 && isCloseText(label)) {
   433→                    NodeView click = resolveClickable(node, 4);
   434→                    if (click != null && isSmallClick(click, ctx) && vis && !f.foundClose) {
   435→                        f.foundClose = true;
   436→                        f.closeNode = click;
   437→                        // v3.6.0：同 L1——手势点原标签坐标，performAction 走可点祖先
   438→                        f.closeBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   439→                        f.closeLabel = label;
   440→                    }
   441→                } else if (isAdBadge(label)) {
   442→                    f.adBadge = true;
   443→                    f.badgeLabel = label;
   444→                } else if (label.length() <= 14 && matchesAny(label, CTA_TEXTS)) {
   445→                    f.ctaCount++;
   446→                }
   447→            } else if (b != null && vis && node.childCount() == 0 && isVidSkip(node)) {
   448→                // v3.8.0：viewId 跳过键（GKD 全局规则同款）：vid 含 skip（穿山甲 tt_splash_skip_btn /
   449→                // 微博 iv_skip 类零文本图片键）或倒计时控件（count+down 非 download）——
   450→                // 零文本图片跳过键的头号信号，此前只能靠角落形态猜，猜不中就只能返回键
   451→                NodeView click = resolveClickable(node, 4);
   452→                if (click != null && isSmallClick(click, ctx)) {
   453→                    if (!f.foundSkip) {
   454→                        f.foundSkip = true;
   455→                        f.skipNode = click;
   456→                        f.skipBounds = b;
   457→                        f.skipLabel = "跳过";
   458→                    }
   459→                    if (f.skipCands.size() < SKIP_CANDS_MAX) {
   460→                        SkipCand c = new SkipCand();
   461→                        c.clickNode = click;
   462→                        c.tapBounds = b;
   463→                        c.label = "跳过";
   464→                        c.selfClick = (click == node);
   465→                        f.skipCands.add(c);
   466→                    }
   467→                } else if (areaRatio(b, ctx) < FAKE_TAP_AREA_MAX
   468→                        && learnableSkipShape(b, ctx.screenW, ctx.screenH)
   469→                        && f.fakeCands.size() < SKIP_CANDS_MAX) {
   470→                    SkipCand c = new SkipCand();
   471→                    c.clickNode = click;
   472→                    c.tapBounds = b;
   473→                    c.label = "跳过";
   474→                    c.gestureOnly = true;
   475→                    f.fakeCands.add(c);
   476→                }
   477→                int cd = parseCountdown(node.viewId());
   478→                if (cd > f.countdownN) f.countdownN = cd;
   479→            } else if (b != null && vis && f.corners.size() < MAX_CORNERS && isCornerControl(b, ctx)
   480→                    && !hasCornerChild(node, ctx)) {
   481→                // 【v3.5.0】角落无字小可点控件（多候选）：疑似图片跳过键
   482→                // （微博/喜马拉雅第一方开屏）；自身不可点时向上找可点祖先；
   483→                // 若还有更小的角落子控件则让位（子控件坐标点得更准）
   484→                NodeView click = clickable ? node : resolveClickable(node, 4);
   485→                if (click != null) {
   486→                    CornerCand c = new CornerCand();
   487→                    c.clickNode = click;
   488→                    c.tapBounds = b;
   489→                    f.corners.add(c);
   490→                }
   491→            }
   492→
   493→            int n = node.childCount();
   494→            for (int i = 0; i < n; i++) {
   495→                visit(node.child(i), depth + 1, budget, f, ctx);
   496→            }
   497→        } catch (Throwable ignored) {
   498→        }
   499→    }

   501→    /** 顶部区域判定（开屏×关闭键防误点：真广告×在顶部，隐私/业务弹窗按钮在底部） */
   502→    static boolean inTopArea(int[] b, Ctx ctx) {
   503→        if (b == null || ctx.screenH <= 0) return false;
   504→        int cy = (b[1] + b[3]) / 2;
   505→        return cy <= ctx.screenH * CLOSE_TOP_MAX;
   506→    }

   508→    /** 是否存在更小的角落子控件（有则父容器让位：子控件坐标点得更准） */
   509→    static boolean hasCornerChild(NodeView node, Ctx ctx) {
   510→        int n = node.childCount();
   511→        for (int i = 0; i < n; i++) {
   512→            NodeView c = node.child(i);
   513→            if (c == null) continue;
   514→            if (nz(c.text()).length() > 0 || nz(c.desc()).length() > 0) continue;
   515→            int[] cb = c.bounds();
   516→            if (cb != null && isCornerControl(cb, ctx)) return true;
   517→        }
   518→        return false;
   519→    }
   520→
   521→    /** 角落控件判定：顶部 14% 高内 / v4.3.0 底部 18% 高内（部分应用跳过键
   522→     *  在右下角且位置随机——issue#38「按钮每次出现在不同地方」），
   523→     *  贴右缘(≥72%)或左缘(≤28%)、边长 24px~35% 宽、无任何文本 */
   524→    static boolean isCornerControl(int[] b, Ctx ctx) {
   525→        if (b == null || ctx.screenW <= 0 || ctx.screenH <= 0) return false;
   526→        int w = b[2] - b[0];
   527→        int h = b[3] - b[1];
   528→        if (w < CORNER_MIN_PX || h < CORNER_MIN_PX) return false;
   529→        if (w > ctx.screenW * 0.35 || h > ctx.screenH * 0.20) return false;
   530→        int cy = (b[1] + b[3]) / 2;
   531→        boolean topZone = b[1] <= ctx.screenH * CORNER_TOP_MAX;
   532→        boolean bottomZone = cy >= ctx.screenH * 0.82; // v4.3.0：底部角落（右下/左下）
   533→        if (!topZone && !bottomZone) return false;
   534→        boolean rightEdge = b[2] >= ctx.screenW * CORNER_RIGHT_MIN;
   535→        boolean leftEdge = b[0] <= ctx.screenW * CORNER_LEFT_MAX;
   536→        return rightEdge || leftEdge;
   537→    }

   549→    static boolean isSmallClick(NodeView n, Ctx ctx) {
   550→        if (n == null || !n.clickable()) return false;
   551→        int[] b = n.bounds();
   552→        if (ctx.screenW <= 0 || b == null) return true; // 无屏幕信息时放宽
   553→        double ratio = areaRatio(b, ctx);

   582→    /**
   583→     * v3.6.0 学习形状判定（比扫描用的 isCornerControl 更宽）：
   584→     * 手动点击教学时，跳过键可能是——
   585→     *  · 顶部 25% 区域内的小控件（最常见：右上「跳过」图片键）；
   586→     *  · 底部 20% 区域内的小控件（部分应用跳过键在右下角，真机反馈）；
   587→     *  · 标准角落形态（顶部 14% + 贴左/右缘）。
   588→     * 面积上限由调用方按 POS_LEARN_AREA_MAX 校验（大热区=CTA，不学）。

   590→    public static boolean learnableSkipShape(int[] b, int screenW, int screenH) {
   591→        if (b == null || screenW <= 0 || screenH <= 0) return false;
   592→        if (cornerLike(b, screenW, screenH)) return true;
   593→        int cy = (b[1] + b[3]) / 2;
   594→        boolean topArea = cy <= screenH * 0.25;
   595→        boolean bottomArea = cy >= screenH * 0.80;
   596→        boolean small = (b[2] - b[0]) <= screenW * 0.5 && (b[3] - b[1]) <= screenH * 0.15;
   597→        return (topArea || bottomArea) && small;
   598→    }

   608→    /* ═════════ v4.5.0 乱点根治：竖列菜单/抽屉列表行检测（李跳跳式形状门）═════════
   609→     * 真机日志实锤（issue#42-44）：应用右侧抽屉（BaseCustomDialogFragment，
   610→     * 播放历史/录音/扫一扫/切换模式 菜单）被引擎当广告窗扫描，菜单内
   611→     * 「跳过」字样列表项被当跳过键点击；用户点开抽屉 4s 让路期满后引擎
   612→     * 连点 2 次 = 「进入 App 乱点」元凶。
   613→     * 判定核心：沿候选祖先链找【列表容器】——候选所在的行与其 ≥3 个
   614→     * 兄弟行【同列同宽竖排】（x 重叠、y 互不重叠）= 垂直菜单列表。
   615→     * ⚠ 比较对象是「候选的祖先行 vs 兄弟行」（行由统一模板渲染必然同宽），
   616→     * 而非「候选自身 vs 行」（文字总比行窄，比不出差异——v4.5.0 首版教训）。
   617→     * 纯 Java 零 Android 依赖，JVM 可单测。 */
   618→    public static boolean isMenuListRow(NodeView node) {
   619→        try {
   620→            if (node == null) return false;
   621→            NodeView self = node;
   622→            NodeView p = node.parent();
   623→            for (int lvl = 0; lvl < 3 && p != null; lvl++) {
   624→                int n = p.childCount();
   625→                if (n >= 4) {
   626→                    int[] sb = self.bounds();
   627→                    if (sb != null && sb[2] > sb[0] && sb[3] > sb[1]) {
   628→                        int sw = sb[2] - sb[0], sh = sb[3] - sb[1];
   629→                        int sameCol = 1; // 含候选所在行自身
   630→                        for (int i = 0; i < n && i < 14; i++) {
   631→                            NodeView c = p.child(i);
   632→                            if (c == null || c == self) continue;
   633→                            int[] cb = c.bounds();
   634→                            if (cb == null || cb[2] <= cb[0] || cb[3] <= cb[1]) continue;
   635→                            int cw = cb[2] - cb[0], ch = cb[3] - cb[1];
   636→                            // ① 同宽（±25%）：菜单行由统一模板渲染，宽度一致
   637→                            if (Math.abs(cw - sw) > Math.max(sw * 0.25, 1)) continue;
   638→                            // ② 同列：x 区间与候选行重叠 ≥ 两者较窄者的 50%
   639→                            int ox = Math.min(sb[2], cb[2]) - Math.max(sb[0], cb[0]);
   640→                            if (ox < Math.min(cw, sw) * 0.5) continue;
   641→                            // ③ 竖排：y 中心间隔 ≥ 两者较高者的 60%（互不重叠的行）
   642→                            int cyS = (sb[1] + sb[3]) / 2, cyC = (cb[1] + cb[3]) / 2;
   643→                            if (Math.abs(cyC - cyS) < Math.max(sh, ch) * 0.6) continue;
   644→                            sameCol++;
   645→                        }
   646→                        if (sameCol >= 4) return true; // 候选行+3 兄弟行同列同宽 = 列表
   647→                    }
   648→                }
   649→                self = p;
   650→                p = p.parent();
   651→            }
   652→        } catch (Throwable ignored) {
   653→        }
   654→        return false;
   655→    }

   657→    /** v4.5.0：文本规则命中形状门（RuleEngine.build 后置校验共用）。
   658→     *  真跳过键形态（全分辨率自适应的相对判定，参考李跳跳形状白名单）：
   659→     *   · 标签控件自身宽度 ≤55% 屏宽（全宽行 = 列表项/横幅，绝不点）；
   660→     *   · 自身可点的按钮放宽到 ≤70%（部分广告底部「点击跳过」长条按钮）；
   661→     *   · 竖列菜单行一票否决（isMenuListRow）。
   662→     *  豁免：SDK 广告窗 / 学习窗指纹窗（上下文已是高置信广告场景）。 */
   663→    public static boolean validSkipTextShape(NodeView node, int[] b, Ctx ctx) {
   664→        if (b == null || b.length < 4 || ctx.screenW <= 0) return false;
   665→        int w = b[2] - b[0];
   666→        if (w <= 0) return false;
   667→        boolean selfClickable = false;
   668→        try {
   669→            selfClickable = node != null && node.clickable();
   670→        } catch (Throwable ignored) {
   671→        }
   672→        double wRatio = (double) w / ctx.screenW;
   673→        if (selfClickable ? wRatio > 0.70 : wRatio > 0.55) return false;
   674→        if (isMenuListRow(node)) return false;
   675→        return true;
   676→    }

   680→    public static Decision decide(ScanFind f, Ctx ctx) {
   681→        /* ═══ v5.0.0 小程序/团购容器保护（L0，纵深防御第四层）═══
   682→         * 容器上下文里决策恒 NONE——即使 ScanFind 被外层填了候选
   683→         * （规则引擎/原生探针/订阅命中），也绝不在这里兑换成动作。
   684→         * 返回键同样禁止：容器内按返回 = 把用户踢出团购券页。 */
   685→        if (MiniProgramGuard.isMiniProgramWindow(ctx.pkg, ctx.activityId)) {
   686→            return Decision.none();
   687→        }
   688→        /* ── 开屏/插屏窗内 ── */
   689→        if (ctx.splashLive) {
   690→            /* v3.7.0 学习位置【前置】到 L1 之前（issue#10 根因）：
   691→             * 用户手动点通过的位置 = 真实手指触点真正生效的坐标——比盲点
   692→             * 文本候选可靠得多（文本候选可能是广告图内假「跳过」文本）。
   693→             * 教过一次，下次同窗优先直点这个位置。 */
   694→            // L1.4 学习位置点击（v3.7.0 前置）：手动点过的位置指纹 → 次启同窗 500ms 直点
   695→            //     安全门：仅在广告形态窗（简单树/SDK 窗/学习窗）内出手，
   696→            //     信息流等正常界面（可滚动/有输入框/节点多）绝不点
   697→            if (ctx.tapHintX >= 0 && ctx.tapHintY >= 0 && ctx.clickAttempts == 0
   698→                    && ctx.elapsedMs >= LEARNED_POS_MS
   699→                    && (isAdLike(f) || ctx.sdkAdWin || ctx.learnedCls)) {
   700→                Decision d = new Decision();
   701→                d.action = A_TAP_POINT;
   702→                d.x = ctx.tapHintX;
   703→                d.y = ctx.tapHintY;
   704→                d.reason = "learned_pos";
   705→                return d;
   706→            }
   707→            /* L1 文本/vid 跳过键多候选轮换（真候选 → 假跳过手势候选统一梯子）
   708→             * v3.8.0（issue#10 二轮·GKD 方法论对齐）：
   709→             *  · 新增假跳过手势候选——「跳过」文案+大容器可点祖先（微博实锤：isSmallClick
   710→             *    判假→从不点击只返回键，返回键又被广告拦截）→ 手势直点标签自身坐标
   711→             *    （clickCenter），performAction 大容器=点开广告绝不走；
   712→             *  · walk 内候选排序：自身可点/短文案/小面积优先，真键排前 */
   713→            int R = f.skipCands.size();
   714→            int F = f.fakeCands.size();
   715→            if (R + F > 0) {
   716→                int idx = Math.min(ctx.clickAttempts, R + F - 1);
   717→                if (idx < R) {
   718→                    SkipCand c = f.skipCands.get(idx);
   719→                    return click(c.clickNode, c.tapBounds, "skip_text");
   720→                }
   721→                SkipCand c = f.fakeCands.get(idx - R);
   722→                return click(null, c.tapBounds, "fake_skip_tap"); // 手势专用：大容器祖先不 performAction
   723→            }
   724→            // L1.6 开屏×关闭键（v3.5.0）：顶部小关闭键 + 广告形态树（非长文本内容页）
   725→            //     v3.7.0 树上限放宽（CORNER_TREE_MAX_NODES）：广告窗树常大，
   726→            //     40 节点上限把真关闭键挡在门外（微博根因之一）
   727→            if (f.foundClose && f.closeBounds != null && !f.longText
   728→                    && (isAdLikeLoose(f) || ctx.sdkAdWin || ctx.learnedCls)
   729→                    && inTopArea(f.closeBounds, ctx)
   730→                    && ctx.elapsedMs >= SPLASH_CLOSE_MS) {
   731→                return click(f.closeNode, f.closeBounds, "splash_close");
   732→            }
   733→            // L4 角落无字小可点控件（图片跳过键，多候选）→ 手势点击其中心
   734→            //     v3.5.0：复验失败换下一个候选（第 N 次点击用第 N 个候选点）
   735→            //     v3.6.0：改纯手势（node=null）——performAction 可能落到大热区
   736→            //     祖先（=点开广告本体）；手势点子控件坐标=真实手指点跳过键
   737→            //     v3.7.0：① 前置到 L3 之前（微博根因：陀螺仪文案 120ms 抢先
   738→            //     返回键，角落键 500ms 才轮到——有点击候选先点击，点不到再断）；
   739→            //     ② 树上限放宽 CORNER_TREE_MAX_NODES=150（广告窗树常大）
   740→            //     v3.8.0：前置到 L2 直断之前——角落键 500ms 时机先于 700ms 直断，
   741→            //     有可点目标绝不先按返回键
   742→            if (!f.corners.isEmpty() && ctx.clickAttempts < f.corners.size()
   743→                    && !f.scrollPresent && !f.editPresent
   744→                    && f.nodeCount <= CORNER_TREE_MAX_NODES
   745→                    && ctx.elapsedMs >= CORNER_MIN_MS && ctx.elapsedMs <= CORNER_MAX_MS
   746→                    // v3.9.0 广告形态确认门（不乱点红线）：角落无字控件必须
   747→                    // 同时具备广告信号（热区/角标/SDK 窗/学习窗）或树极简
   748→                    //（≤60 节点）才出手——普通主页角落图标绝不误点
   749→                    && (f.hotZone || f.adBadge || ctx.sdkAdWin || ctx.learnedCls
   750→                        || f.nodeCount <= CORNER_CONFIRM_NODES)) {
   751→                CornerCand c = f.corners.get(ctx.clickAttempts);
   752→                return click(null, c.tapBounds, "corner_image_skip");
   753→            }
   754→            // L2 SDK / 学习 className 窗口指纹 → 返回键直断
   755→            //    SDK 广告窗是独立 Activity，返回回到应用内页——根窗口也安全。
   756→            //    v3.6.0 学习指纹快断加双保险（真机事故：毒指纹把主页学成广告窗
   757→            //    → 入口窗快断=按返回键退出应用）：① 仅非根窗口或 SDK 窗可快断；
   758→            //    ② 须为广告形态树（信息流等可滚动页面绝不快断）
   759→            boolean sdkCut = ctx.sdkAdWin && ctx.elapsedMs >= CLS_BREAK_MS;
   760→            boolean learnedCut = ctx.learnedCls && ctx.elapsedMs >= CLS_LEARNED_BREAK_MS
   761→                    && (!ctx.rootWindow || ctx.sdkAdWin) && isAdLike(f);
   762→            if (sdkCut || learnedCut) {
   763→                return back(learnedCut && !sdkCut ? "sdk_window_learned" : "sdk_window");
   764→            }
   765→            /* v3.6.0 根窗口返回键安全门 ─────────────────────────────
   766→             * 应用入口窗（冷启动开屏/回前台首窗）上按返回键 = 退出应用。
   767→             * 真机事故（喜马拉雅）：v3.5.0 返回键兜底 → 喜马拉雅拦截返回
   768→             * 弹「再按一次退出到桌面」→ 引擎 600ms 后再按 → 真退到桌面，
   769→             * 用户「第二次进入就被模拟人操作退出来」。
   770→             * 此后 L3/L4.5/L5 的返回键决策仅在应用内窗（返回回到应用内）
   771→             * 或 SDK 广告窗上生效；根窗口只走点击梯子，用尽优雅收手。 */
   772→            boolean backSafe = !ctx.rootWindow || ctx.sdkAdWin;
   773→            // L3 陀螺仪/假跳过：无倒计时立即直断；带倒计时(假跳过) 800ms 直断
   774→            //     v3.6.0 加信息流安全门：可滚动/有输入框的页面出现「摇一摇/翻转」文案
   775→            //     多为互动玩法或 feed 内容（如「摇一摇抽签」），绝不返回键（防主页误退）
   776→            //     v3.7.0 移到 L4 之后（微博根因）：有可点候选先点击，返回键是
   777→            //     「无任何可点目标」时的兜底（广告 Activity 常拦截返回键）；
   778→            //     gyro_immediate 加「无可点候选」门——角落键 500ms 才到时机，
   779→            //     有候选时等它，不 120ms 抢跑返回键
   780→            boolean clickableCand = !f.corners.isEmpty() || !f.skipCands.isEmpty() || !f.fakeCands.isEmpty() || f.foundClose;
   781→            if (backSafe && f.gyroHint && !f.fakeSkip && !clickableCand
   782→                    && !f.scrollPresent && !f.editPresent
   783→                    && ctx.elapsedMs >= GYRO_IMMEDIATE_MS) {
   784→                return back("gyro_immediate");
   785→            }
   786→            if (backSafe && (f.gyroHint || f.fakeSkip) && !f.scrollPresent && !f.editPresent
   787→                    && ctx.elapsedMs >= GYRO_BREAK_MS) {
   788→                return back("gyro_break");
   789→            }
   790→            // L4.5 点击复验失败升级（v3.5.0）：同窗点击 ≥3 次仍未关闭 → 返回键兜底
   791→            //     v3.6.0：仅应用内窗（返回回到应用内）；根窗口绝不返回键（防退出）
   792→            if (ctx.clickAttempts >= CLICK_VERIFY_MAX && backSafe && isAdLike(f)
   793→                    && ctx.elapsedMs >= GYRO_BREAK_MS) {
   794→                return back("click_verify_fail");
   795→            }
   796→            // L5 全屏热区（自绘图片广告）→ 返回键直断（守护名单豁免；v3.6.0 根窗口豁免）
   797→            if (f.hotZone && backSafe && !ctx.guard && !f.editPresent && f.nodeCount <= HOT_ZONE_MAX_NODES
   798→                    && ctx.elapsedMs >= HOT_ZONE_MS) {
   799→                return back("hot_zone");
   800→            }
   801→            /* v3.9.0：L6 坐标级点击兜底已【整体移除】——固定百分比坐标
   802→             * 在不同屏幕上位置不同，盲点必然乱点（issue#12 闪退毒链：
   803→             * 桌面被误判→盲点桌面图标→拉起微信→回弹踢出用户）。零信号
   804→             * 场景由学习位置（用户教过）与 RuleEngine 规则接管，绝不盲点。 */
   805→        }
   806→
   807→        /* ── 弹窗窗期（独立于开屏窗） ── */
   808→        if (ctx.popupLive && f.foundClose && f.closeBounds != null) {
   809→            boolean strong = f.adBadge;                    // 角标强信号
   810→            boolean weak = f.dialogForm && f.ctaCount >= 2; // Dialog + ≥2 营销话术
   811→            if (strong || weak) {
   812→                Decision d = click(f.closeNode, f.closeBounds, strong ? "popup_badge" : "popup_cta");
   813→                d.action = A_CLOSE;
   814→                return d;
   815→            }
   816→        }
   817→
   818→        return Decision.none();
   819→    }

   983→        public boolean isSheet;     // 是「跳过后反馈面板」（强特征 ≥1 且总特征 ≥2）
   984→        public NodeView cancelNode; // 中性收起键（取消/暂不反馈/下次再说…）
   985→        public int[] cancelBounds;  // 收起键边界（手势点击目标）

   989→    /** 有界广度优先识别「跳过后反馈面板」：统计特征文案命中数（去重）+
   990→     *  找文案精确等于「取消」的可见控件。底部面板树浅，预算 80 零性能顾虑。
   991→     *  只识别不点击——点不点「取消」由服务侧门控（12s 内自动跳过过）决定。 */
   992→    public static FeedbackFind findFeedbackSheet(NodeView root, int budget) {
   993→        FeedbackFind out = new FeedbackFind();
   994→        if (root == null || budget <= 0) return out;
   995→        java.util.ArrayDeque<NodeView> q = new java.util.ArrayDeque<>();
   996→        q.add(root);
   997→        int seen = 0;
   998→        java.util.HashSet<String> hits = new java.util.HashSet<>();
   999→        boolean strong = false;
  1000→        while (!q.isEmpty() && seen < budget) {
  1001→            NodeView n = q.poll();
  1002→            if (n == null) continue;
  1003→            seen++;
  1004→            String text = nz(n.text());
  1005→            String desc = nz(n.desc());
  1006→            String label = text.length() > 0 ? text : desc;
  1007→            if (label.length() > 0) {
  1008→                for (String m : FEEDBACK_SHEET_STRONG) {
  1009→                    if (label.contains(m)) { hits.add(m); strong = true; break; }
  1010→                }
  1011→                for (String m : FEEDBACK_SHEET_WEAK) {
  1012→                    if (label.contains(m)) { hits.add(m); break; }
  1013→                }
  1014→                if (out.cancelNode == null && n.visible()) {
  1015→                    // v4.3.0：中性收起键候选（取消优先，其次暂不反馈/下次再说/
  1016→                    //  稍后再说/关闭——只收起，绝不提交任何反馈选项）
  1017→                    for (int ci = 0; ci < FEEDBACK_CANCEL_TEXTS.length; ci++) {
  1018→                        if (FEEDBACK_CANCEL_TEXTS[ci].equals(text)) {
  1019→                            int[] b = n.bounds();
  1020→                            if (b != null && b.length >= 4 && b[2] > b[0] && b[3] > b[1]) {
  1021→                                out.cancelNode = n;
  1022→                                out.cancelBounds = b;
  1023→                                out.cancelLabel = FEEDBACK_CANCEL_TEXTS[ci];
  1024→                            }
  1025→                            break;
  1026→                        }
  1027→                    }
  1028→                }
  1029→            }
  1030→            for (int i = 0; i < n.childCount(); i++) {
  1031→                NodeView c = n.child(i);
  1032→                if (c != null) q.add(c);
  1033→            }
  1034→        }
  1035→        out.isSheet = strong && hits.size() >= FEEDBACK_SHEET_MIN_HITS;
  1036→        return out;
  1037→    }
