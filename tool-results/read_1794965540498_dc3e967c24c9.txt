     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { AdCreative } from "@/components/ad-creative";
     5→import { SCENARIO_APPS, type ScenarioApp } from "@/lib/scenario-apps";
     6→import { planInstantBlock, AD_TYPE_LABEL, type BlockPlan } from "@/lib/instant-engine";
     7→import { Button } from "@/components/ui/button";
     8→import { cn } from "@/lib/utils";
     9→import {
    10→  ShoppingBag, ShoppingBasket, Rocket, Landmark, Building2, Wallet, Play, Phone, MessageCircle,
    11→  Globe, Camera, Signal, Wifi, BatteryFull, Zap, ShieldCheck, Headphones,
    12→  Link2Off, ChevronLeft, Search, ScanLine, ArrowLeftRight, Receipt, TrendingUp,
    13→  PiggyBank, UserRound, FileText, CalendarClock, ThumbsUp, Share2,
    14→  Flame, Tv, Music2, Activity, Gauge, Timer, Vibrate, Layers, X, Smartphone, Apple,
    15→} from "lucide-react";
    16→
    17→type Stage = "home" | "launch" | "ad" | "flash" | "main";
    18→
    19→const APP_ICONS: Record<string, React.ElementType> = {
    20→  "shopping-bag": ShoppingBag,
    21→  "shopping-basket": ShoppingBasket,
    22→  rocket: Rocket,
    23→  landmark: Landmark,
    24→  "building-2": Building2,
    25→  wallet: Wallet,
    26→  play: Play,
    27→  headphones: Headphones,
    28→};
    29→
    30→/** 喜马拉雅类广告的甄别阶段（演示节奏，已放慢） */
    31→type FakePhase = "scanning" | "ignored" | "chain";
    32→
    33→interface SessionLogItem {
    34→  id: number;
    35→  app: string;
    36→  kind: "block" | "preempt";
    37→  text: string;
    38→  ms: number;
    39→}
    40→
    41→export function Arena({ onFeedback }: { onFeedback: () => void }) {
    42→  const [app, setApp] = useState<ScenarioApp | null>(null);
    43→  const [stage, setStage] = useState<Stage>("home");
    44→  const [plan, setPlan] = useState<BlockPlan | null>(null);
    45→  const [fakePhase, setFakePhase] = useState<FakePhase | null>(null);
    46→  const [learnedApps, setLearnedApps] = useState<Set<string>>(new Set());
    47→  const [blocks, setBlocks] = useState(0);
    48→  const [preempts, setPreempts] = useState(0);
    49→  const [msSum, setMsSum] = useState(0);
    50→  const [log, setLog] = useState<SessionLogItem[]>([]);
    51→  const [toast, setToast] = useState<{ id: number; text: string; sub?: string } | null>(null);
    52→  const logIdRef = useRef(0);
    53→  const toastIdRef = useRef(0);
    54→
    55→  const pushToast = useCallback((text: string, sub?: string) => {
    56→    const id = ++toastIdRef.current;
    57→    setToast({ id, text, sub });
    58→    setTimeout(() => setToast((t) => (t?.id === id ? null : t)), 3600);
    59→  }, []);
    60→
    61→  const pushLog = useCallback((item: Omit<SessionLogItem, "id">) => {
    62→    setLog((cur) => [{ ...item, id: ++logIdRef.current }, ...cur].slice(0, 12));
    63→  }, []);
    64→
    65→  /* 学习指纹（命中即预拦截资格） */
    66→  const refreshLearning = useCallback(async () => {
    67→    try {
    68→      const r = await fetch("/api/learning");
    69→      if (!r.ok) return;
    70→      const d = await r.json();
    71→      setLearnedApps(new Set<string>((d.learnedRules ?? []).map((x: { appName: string }) => x.appName)));
    72→    } catch { /* 静默 */ }
    73→  }, []);
    74→  useEffect(() => {
    75→    let alive = true;
    76→    fetch("/api/learning")
    77→      .then((r) => (r.ok ? r.json() : null))
    78→      .then((d) => {
    79→        if (!alive || !d) return;
    80→        setLearnedApps(new Set<string>((d.learnedRules ?? []).map((x: { appName: string }) => x.appName)));
    81→      })
    82→      .catch(() => {});
    83→    return () => { alive = false; };
    84→  }, []);
    85→
    86→  /* ── 埋点 ── */
    87→  const postEvent = useCallback(
    88→    async (p: BlockPlan, note: string) => {
    89→      if (!app) return;
    90→      try {
    91→        const r = await fetch("/api/block-events", {
    92→          method: "POST",
    93→          headers: { "Content-Type": "application/json" },
    94→          body: JSON.stringify({
    95→            appKey: app.key,
    96→            kind: "block",
    97→            action: p.action,
    98→            latencyMs: p.totalMs,
    99→            confidence: p.confidence / 100,
   100→            scenario: p.scenario,
   101→            adType: app.adType,
   102→            note,
   103→          }),
   104→        });
   105→        if (r.ok) {
   106→          const d = await r.json();
   107→          if (d.learnedCreated) {
   108→            pushToast("广告指纹已学习", `${app.name} 下次启动将 P0 预拦截——广告未及呈现`);
   109→            void refreshLearning();
   110→          }
   111→        }
   112→      } catch { /* 静默 */ }
   113→    },
   114→    [app, pushToast, refreshLearning],
   115→  );
   116→
   117→  /* ── 启动应用 ── */
   118→  const launchApp = useCallback((a: ScenarioApp) => {
   119→    setApp(a);
   120→    setPlan(null);
   121→    setFakePhase(null);
   122→    setStage("launch");
   123→  }, []);
   124→
   125→  /* ── 阶段流转 ── */
   126→  useEffect(() => {
   127→    if (stage === "launch" && app) {
   128→      const isPreempt = learnedApps.has(app.name);
   129→      const t = setTimeout(() => {
   130→        const p = planInstantBlock(
   131→          {
   132→            name: app.name,
   133→            category: app.category,
   134→            adType: app.adType,
   135→            guard: app.guard,
   136→            ruleName: app.ruleName,
   137→            learnedRuleName: learnedApps.has(app.name) ? `${app.name} 学习指纹` : null,
   138→            traits: app.traits ?? null,
   139→          },
   140→          { preempt: isPreempt },
   141→        );
   142→        setPlan(p);
   143→        if (isPreempt) {
   144→          // P0 预拦截：广告未及呈现，直接进入拦截成功层
   145→          setPreempts((n) => n + 1);
   146→          setBlocks((n) => n + 1);
   147→          setMsSum((s) => s + p.totalMs);
   148→          pushLog({ app: app.name, kind: "preempt", text: `P0 预拦截 · ${AD_TYPE_LABEL[app.adType]} · 广告未及呈现`, ms: p.totalMs });
   149→          void postEvent(p, "P0 预拦截（学习指纹）");
   150→          setStage("flash");
   151→        } else {
   152→          setStage("ad");
   153→        }
   154→      }, 850);
   155→      return () => clearTimeout(t);
   156→    }
   157→    if (stage === "flash") {
   158→      const t = setTimeout(() => setStage("main"), 1250);
   159→      return () => clearTimeout(t);
   160→    }
   161→  }, [stage, app, learnedApps, postEvent, pushLog]);
   162→
   163→  /* ── 引擎拦截（广告展示中） ── */
   164→  useEffect(() => {
   165→    if (stage !== "ad" || !app || !plan) return;
   166→    const isAdversarial = !!app.traits?.fakeSkip && (!!app.traits?.chainJump || !!app.traits?.gyroFlip);
   167→    const isPopup = app.adType === "popup";
   168→    const timers: ReturnType<typeof setTimeout>[] = [];
   169→    if (isPopup) {
   170→      // 应用内弹窗广告 v2：Dialog 窗口事件 → 角标/营销话术双信号 → 代点关闭
   171→      timers.push(setTimeout(() => setFakePhase("scanning"), 400));
   172→      timers.push(setTimeout(() => setFakePhase("ignored"), 1000));
   173→      timers.push(
   174→        setTimeout(() => {
   175→          setBlocks((n) => n + 1);
   176→          setMsSum((s) => s + plan.totalMs);
   177→          pushLog({ app: app.name, kind: "block", text: `${plan.ruleName} · 弹窗广告已关闭`, ms: plan.totalMs });
   178→          void postEvent(plan, "弹窗 v2 双信号关闭");
   179→          setStage("flash");
   180→        }, 1700),
   181→      );
   182→    } else if (isAdversarial) {
   183→      // 喜马拉雅类：SDK 广告窗 className 识别 → 极速链复核无真跳过键 → 500ms 直断（v3.12.0：不再回弹）
   184→      timers.push(setTimeout(() => setFakePhase("scanning"), 400));
   185→      timers.push(setTimeout(() => setFakePhase("ignored"), 1100));
   186→      timers.push(setTimeout(() => setFakePhase("chain"), 1900));
   187→      timers.push(
   188→        setTimeout(() => {
   189→          setBlocks((n) => n + 1);
   190→          setMsSum((s) => s + plan.totalMs);
   191→          pushLog({ app: app.name, kind: "block", text: `${plan.ruleName} · ${app.traits?.sdkWindow ? "SDK 广告窗指纹直断" : "陀螺仪诱导直断"}`, ms: plan.totalMs });
   192→          void postEvent(plan, app.traits?.sdkWindow ? "SDK 广告窗指纹直断（抢先关闭）" : "假跳过甄别 + 链式前置直断");
   193→          setStage("flash");
   194→        }, 2700),
   195→      );
   196→    } else {
   197→      timers.push(
   198→        setTimeout(() => {
   199→          setBlocks((n) => n + 1);
   200→          setMsSum((s) => s + plan.totalMs);
   201→          pushLog({ app: app.name, kind: "block", text: `${plan.ruleName} · ${AD_TYPE_LABEL[app.adType]}`, ms: plan.totalMs });
   202→          void postEvent(plan, plan.ruleName);
   203→          setStage("flash");
   204→        }, 700),
   205→      );
   206→    }
   207→    return () => timers.forEach(clearTimeout);
   208→  }, [stage, app, plan, postEvent, pushLog]);
   209→
   210→  /* ── 假跳过陷阱教学点按 ── */
   211→  const tapFakeSkip = useCallback(() => {
   212→    pushToast(
   213→      "假跳过陷阱 · 引擎已忽略",
   214→      "这是广告图内嵌的热区：点击即跳落地页并二跳商店。多信号甄别（面积/容器归属/时序）判定为陷阱",
   215→    );
   216→  }, [pushToast]);
   217→
   218→  const backHome = useCallback(() => {
   219→    setApp(null);
   220→    setStage("home");
   221→    setPlan(null);
   222→    setFakePhase(null);
   223→  }, []);
   224→
   225→  const avgMs = blocks > 0 ? Math.round(msSum / blocks) : 0;
   226→
   227→  return (
   228→    <div className="flex flex-col gap-5 lg:flex-row lg:items-start">
   229→      {/* ═══ 左：手机模拟器 ═══ */}
   230→      <section className="flex w-full shrink-0 flex-col items-center gap-3 lg:w-[420px]" aria-label="演练场手机模拟器">
   231→        <div className="relative w-full max-w-[360px] rounded-[2.6rem] border-[10px] border-zinc-800 bg-zinc-800 shadow-2xl shadow-zinc-900/25 ring-1 ring-zinc-700">
   232→          <div className="overflow-hidden rounded-[2rem] bg-white">
   233→            <div className="relative aspect-[9/19] w-full select-none">
   234→              <PhoneScreen
   235→                stage={stage}
   236→                app={app}
   237→                plan={plan}
   238→                fakePhase={fakePhase}
   239→                onLaunch={launchApp}
   240→                onBack={backHome}
   241→                onTapFakeSkip={tapFakeSkip}
   242→              />
   243→
   244→              {/* 拦截 Toast */}
   245→              {toast && (
   246→                <div className="pointer-events-none absolute inset-x-3 top-11 z-[70] animate-[toast-in_.25s_ease-out] rounded-xl bg-zinc-900/95 p-2.5 text-[11px] leading-snug text-white shadow-lg ring-1 ring-white/15">
   247→                  <div className="flex items-center gap-1.5 font-bold">
   248→                    <Zap className="h-3.5 w-3.5 shrink-0 text-amber-300" />
   249→                    <span>{toast.text}</span>
   250→                  </div>
   251→                  {toast.sub && <div className="mt-0.5 opacity-80">{toast.sub}</div>}
   252→                </div>
   253→              )}
   254→            </div>
   255→          </div>
   256→        </div>
   257→        <p className="max-w-[360px] text-center text-[10px] leading-relaxed text-zinc-400">
   258→          同一应用第二次启动将触发 <span className="font-bold text-violet-600">P0 预拦截</span>：窗口附加阶段直接阻断，广告未及呈现
   259→        </p>
   260→      </section>
   261→
   262→      {/* ═══ 右：引擎面板 ═══ */}
   263→      <section className="flex min-w-0 flex-1 flex-col gap-4" aria-label="秒跳引擎面板">
   264→        {/* 引擎状态 */}
   265→        <div className="rounded-2xl border border-emerald-200 bg-emerald-50/60 p-4">
   266→          <div className="flex items-center gap-2.5">
   267→            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white">
   268→              <ShieldCheck className="h-4.5 w-4.5" />
   269→            </span>
   270→            <div className="min-w-0">
   271→              <div className="flex items-center gap-2 text-sm font-bold text-zinc-900">
   272→                秒跳引擎 · 运行中
   273→                <span className="rounded-full bg-emerald-600 px-2 py-0.5 text-[10px] font-bold text-white">十一层纵深</span>
   274→              </div>
   275→              <p className="mt-0.5 text-[11px] text-zinc-500">
   276→                P0 预拦截 → 学习位置点击 → 角落键多候选+坐标轮换 → 点击复验换候选重试（原标签坐标） → SDK 窗指纹 → 热区/立即直断 → 链式二跳前置直断（不回弹，v3.12.0） → 弹窗 v2 → 返回键兜底（仅应用内窗） → 根窗口安全门（入口窗绝不返回键，防退出应用）；守护应用只拦广告、不碰服务
   277→              </p>
   278→            </div>
   279→            <div className="ml-auto hidden shrink-0 items-center gap-1 text-[10px] font-bold text-emerald-700 sm:flex">
   280→              <Layers className="h-3.5 w-3.5" /> 1060 款语料 100% · 二十五轮
   281→            </div>
   282→          </div>
   283→        </div>
   284→
   285→        {/* 统计 chips */}
   286→        <div className="grid grid-cols-3 gap-2.5">
   287→          {[
   288→            { icon: Zap, label: "本次会话拦截", value: `${blocks} 次`, tone: "text-emerald-600" },
   289→            { icon: Gauge, label: "平均拦截耗时", value: blocks > 0 ? `${avgMs} ms` : "—", tone: "text-violet-600" },
   290→            { icon: Timer, label: "预拦截（0ms 可见）", value: `${preempts} 次`, tone: "text-fuchsia-600" },
   291→          ].map(({ icon: Icon, label, value, tone }) => (
   292→            <div key={label} className="rounded-xl border border-zinc-200 bg-white p-3">
   293→              <div className="flex items-center gap-1 text-[10px] font-semibold text-zinc-400">
   294→                <Icon className="h-3 w-3" /> {label}
   295→              </div>
   296→              <div className={cn("mt-1 text-lg font-black tabular-nums", tone)}>{value}</div>
   297→            </div>
   298→          ))}
   299→        </div>
   300→
   301→        {/* 判定流水线 */}
   302→        <div className="rounded-2xl border border-zinc-200 bg-white p-4">
   303→          <div className="flex items-center justify-between">
   304→            <h3 className="flex items-center gap-1.5 text-sm font-bold text-zinc-900">
   305→              <Activity className="h-4 w-4 text-violet-600" /> 毫秒级判定流水线
   306→            </h3>
   307→            {plan && (
   308→              <span className={cn("rounded-full px-2 py-0.5 text-[10px] font-bold ring-1",
   309→                plan.preempt ? "bg-fuchsia-50 text-fuchsia-700 ring-fuchsia-200" : "bg-violet-50 text-violet-700 ring-violet-200")}>
   310→                {plan.preempt ? "P0 预拦截" : plan.ruleName}
   311→              </span>
   312→            )}
   313→          </div>
   314→          {plan ? (
   315→            <>
   316→              <div className="mt-3 flex items-baseline gap-2">
   317→                <span className="text-2xl font-black tabular-nums text-zinc-900">{plan.totalMs}<span className="text-sm">ms</span></span>
   318→                <span className="text-[11px] font-semibold text-zinc-500">端到端耗时</span>
   319→                <span className="ml-auto text-[11px] font-bold text-emerald-600">置信度 {plan.confidence}%</span>
   320→              </div>
   321→              {plan.waitMs > 0 && (
   322→                <p className="mt-1 text-[10px] leading-relaxed text-amber-600">
   323→                  含等待广告方 { (plan.waitMs / 1000).toFixed(1) }s 显示真跳过（引擎出手仅 {plan.engineMs}ms）——假跳过陷阱甄别中，真跳过一挂载即出手
   324→                </p>
   325→              )}
   326→              <ol className="mt-3 space-y-1.5">
   327→                {plan.steps.map((s, i) => (
   328→                  <li key={i} className="flex items-start gap-2 rounded-lg bg-zinc-50 px-2.5 py-1.5">
   329→                    <span className="mt-px w-11 shrink-0 text-right text-[10px] font-bold tabular-nums text-violet-500">{s.atMs}ms</span>
   330→                    <span className="min-w-0">
   331→                      <span className="text-[11px] font-bold text-zinc-800">{s.label}</span>
   332→                      <span className="ml-1.5 text-[10px] leading-relaxed text-zinc-500">{s.detail}</span>
   333→                    </span>
   334→                  </li>
   335→                ))}
   336→              </ol>
   337→              <p className="mt-2 text-[10px] leading-relaxed text-zinc-400">※ 演示动画已放慢节奏以便观察，引擎真实耗时以上方流水线为准（真机由无障碍服务毫秒级执行）。</p>
   338→            </>
   339→          ) : (
   340→            <p className="mt-3 rounded-lg bg-zinc-50 px-3 py-6 text-center text-xs text-zinc-400">
   341→              点按左侧任意应用图标启动，观察秒跳引擎的拦截过程；再次启动同一应用可体验预拦截
   342→            </p>
   343→          )}
   344→        </div>
   345→
   346→        {/* 场景速览 + 会话日志 */}
   347→        <div className="grid gap-4 md:grid-cols-2">
   348→          <div className="rounded-2xl border border-zinc-200 bg-white p-4">
   349→            <h3 className="text-sm font-bold text-zinc-900">八大场景</h3>
   350→            <ul className="mt-2.5 space-y-2">
   351→              {SCENARIO_APPS.map((a) => {
   352→                const Icon = APP_ICONS[a.icon];
   353→                return (
   354→                  <li key={a.key} className="flex items-center gap-2.5">
   355→                    <span className={cn("flex h-7 w-7 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br text-white", a.gradient)}>
   356→                      <Icon className="h-3.5 w-3.5" />
   357→                    </span>
   358→                    <div className="min-w-0 flex-1">
   359→                      <div className="flex items-center gap-1.5 text-xs font-bold text-zinc-800">
   360→                        {a.name}
   361→                        <span className="rounded bg-zinc-100 px-1 text-[9px] font-semibold text-zinc-500">{a.category}</span>
   362→                        {a.guard && <span className="rounded bg-emerald-100 px-1 text-[9px] font-bold text-emerald-700">守护</span>}
   363→                        {a.traits?.fakeSkip && <span className="rounded bg-rose-100 px-1 text-[9px] font-bold text-rose-600">专案</span>}
   364→                        {learnedApps.has(a.name) && <span className="rounded bg-fuchsia-100 px-1 text-[9px] font-bold text-fuchsia-700">指纹</span>}
   365→                      </div>
   366→                      <div className="truncate text-[10px] text-zinc-400">{a.note} · {AD_TYPE_LABEL[a.adType]}</div>
   367→                    </div>
   368→                  </li>
   369→                );
   370→              })}
   371→            </ul>
   372→            <Button size="sm" variant="outline" data-testid="btn-arena-feedback" className="mt-3 h-7 w-full rounded-lg text-[11px]" onClick={onFeedback}>
   373→              遇到漏拦截 / 误拦截？反馈给开发者
   374→            </Button>
   375→          </div>
   376→          <div className="rounded-2xl border border-zinc-200 bg-white p-4">
   377→            <h3 className="text-sm font-bold text-zinc-900">会话拦截日志</h3>
   378→            {log.length === 0 ? (
   379→              <p className="mt-3 rounded-lg bg-zinc-50 px-3 py-6 text-center text-xs text-zinc-400">暂无记录</p>
   380→            ) : (
   381→              <ul className="mt-2.5 max-h-72 space-y-1.5 overflow-y-auto pr-1">
   382→                {log.map((l) => (
   383→                  <li key={l.id} className="flex items-center gap-2 rounded-lg bg-zinc-50 px-2.5 py-1.5 text-[11px]">
   384→                    <span className={cn("h-1.5 w-1.5 shrink-0 rounded-full", l.kind === "preempt" ? "bg-fuchsia-500" : "bg-emerald-500")} />
   385→                    <span className="font-bold text-zinc-800">{l.app}</span>
   386→                    <span className="min-w-0 flex-1 truncate text-zinc-500">{l.text}</span>
   387→                    <span className="shrink-0 font-bold tabular-nums text-emerald-600">{l.ms}ms</span>
   388→                  </li>
   389→                ))}
   390→              </ul>
   391→            )}
   392→          </div>
   393→        </div>
   394→      </section>
   395→    </div>
   396→  );
   397→}
