   457→function cmpVer(a, b) {
   458→  const pa = String(a).replace(/^v/, "").split("."), pb = String(b).replace(/^v/, "").split(".");
   459→  for (let i = 0; i < 3; i++) {
   460→    const d = (parseInt(pa[i], 10) || 0) - (parseInt(pb[i], 10) || 0);
   461→    if (d) return d > 0 ? 1 : -1;
   462→  }
   463→  return 0;
   464→}

   466→async function fetchLatest() {
   467→  if (HAS_BRIDGE) {
   468→    // 原生侧已带令牌 + 两重重试 + /releases 列表回退（v3.3.0）
   469→    try { return JSON.parse(AndroidApp.netGetLatestRelease()); } catch (e) { return { ok: false, error: "解析失败" }; }
   470→  }
   471→  try {
   472→    const r = await fetch(GH_API + "/releases/latest", { headers: { Accept: "application/vnd.github+json" } });
   473→    if (r.status === 404) return { ok: true, hasRelease: false };
   474→    if (r.ok) {
   475→      const j = await r.json();
   476→      const apk = (j.assets || []).filter(a => a && a.name && a.name.endsWith(".apk"))[0];
   477→      return {
   478→        ok: true, hasRelease: true, tag: j.tag_name, notes: j.body, url: j.html_url,
   479→        apkUrl: apk ? apk.browser_download_url : "",
   480→        apkMirrors: apk ? ["https://ghfast.top/" + apk.browser_download_url, "https://gh-proxy.com/" + apk.browser_download_url] : [],
   481→        apkFileName: apk ? apk.name : "adshield.apk",
   482→        apkMB: apk ? Math.round(apk.size / 1048576 * 100) / 100 : 0,
   483→      };
   484→    }
   485→    return { ok: false, error: "GitHub 返回 " + r.status };
   486→  } catch (e) { return { ok: false, error: "网络异常：" + (e.message || e) }; }
   487→}

   491→/* ── 统一页内下载流程（横幅与关于页共用；多通道自动切换） ── */
   492→function startPageDownload(d, prefix) {
   493→  const fileName = d.apkFileName || "adshield.apk";
   494→  if (!d.apkUrl) { toast("该版本没有 APK 附件"); return; }
   495→  if (!HAS_BRIDGE) { openLink(d.apkUrl); return; } // 浏览器预览：退化为直链
   496→  dl = {
   497→    channels: [d.apkUrl].concat(d.apkMirrors || []),
   498→    idx: 0, fileName, prefix,
   499→    lastBytes: -1, lastMoveAt: Date.now(),
   500→  };
   501→  beginChannel();
   502→}

   504→function beginChannel() {
   505→  const url = dl.channels[dl.idx];
   506→  try {
   507→    const r = JSON.parse(AndroidApp.startUpdateDownload(url, dl.fileName));
   508→    if (r.ok) {
   509→      const btn = $("#" + dl.prefix + "-btn");
   510→      if (btn) { btn.textContent = "下载中…"; btn.disabled = true; }
   511→      const wrap = $("#" + dl.prefix + "-progress-wrap");
   512→      if (wrap) wrap.style.display = "";
   513→      dl.lastBytes = -1;
   514→      dl.lastMoveAt = Date.now();
   515→      startProgressPoll(dl.prefix);
   516→    } else toast(r.error || "启动下载失败");
   517→  } catch (e) { toast("启动下载失败"); }
   518→}

   520→/* 切换下一个下载通道；无通道可用时返回 false */
   521→function switchChannel(reason) {
   522→  if (dl.idx + 1 < dl.channels.length) {
   523→    dl.idx++;
   524→    toast(reason + "，已切换备用通道（" + dl.idx + "/" + (dl.channels.length - 1) + "）");
   525→    beginChannel();
   526→    return true;
   527→  }
   528→  return false;
   529→}

   531→function startProgressPoll(prefix) {
   532→  if (ubTimer) clearInterval(ubTimer);
   533→  ubTimer = setInterval(() => {
   534→    try {
   535→      const d = JSON.parse(AndroidApp.updateProgress());
   536→      const bar = $("#" + prefix + "-bar"), txt = $("#" + prefix + "-text");
   537→      if (d.state === "running" || d.state === "paused") {
   538→        bar.style.width = d.percent + "%";
   539→        txt.textContent = d.percent + "% · " + (d.bytes / 1048576).toFixed(1) + " / " + (d.total / 1048576).toFixed(1) + " MB";
   540→        // 停滞检测：30s 无新字节 → 判定该通道不可用，自动切换
   541→        if (d.bytes !== dl.lastBytes) { dl.lastBytes = d.bytes; dl.lastMoveAt = Date.now(); }
   542→        else if (Date.now() - dl.lastMoveAt > 30000) {
   543→          if (switchChannel("下载长时间无进展")) return;
   544→          dl.lastMoveAt = Date.now(); // 无通道可切：再等一轮
   545→        }
   546→      } else if (d.state === "done") {
   547→        clearInterval(ubTimer); ubTimer = null;
   548→        bar.style.width = "100%";
   549→        txt.textContent = "下载完成 ✓ 点击安装即覆盖升级（学习数据保留）";
   550→        const btn = $("#" + prefix + "-btn");
   551→        btn.textContent = "立即安装";
   552→        btn.disabled = false;
   553→        btn.dataset.phase = "install";
   554→      } else if (d.state === "failed") {
   555→        clearInterval(ubTimer); ubTimer = null;
   556→        if (switchChannel("直连下载失败")) return;
   557→        txt.textContent = "下载失败（直连与备用通道均不可用），可稍后重试";
   558→        const btn = $("#" + prefix + "-btn");
   559→        btn.textContent = "重新下载";
   560→        btn.disabled = false;
   561→        btn.dataset.phase = "download";
   562→      }
   563→    } catch (e) { clearInterval(ubTimer); ubTimer = null; }
   564→  }, 800);
   565→}
