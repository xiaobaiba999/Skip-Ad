   363→        /** 自动拉活（v4.4.0 深度拉活版）：
   364→         *  · 服务已运行 → ok（若处于暂停自动解除——用户点「恢复」的意图）
   365→         *  · 设置仍登记且服务死 → 后台线程深挖最长 6s（v3.14.0 只等 1.5s，
   366→         *    MIUI/HyperOS 重绑慢的 ROM 不够→被误判 need_toggle→跳系统设置
   367→         *    看 10 秒强制警告，「恢复防护很麻烦」的元凶）；每 500ms 用
   368→         *    AccessibilityManager 查询踢一次系统（促使 AMS 重绑）；
   369→         *  · 未登记 → disabled 由页面引导开启。
   370→         *  （无障碍服务受 BIND_ACCESSIBILITY_SERVICE 保护，任何应用都不能
   371→         *   start/bind 它；设置页 10 秒强制等待仅剩「开关真被关」时才需要）
   372→         *  返回 JSON：{ok, running?, revived?, resumed?, enabledInSettings?, reason?} */
   373→        @JavascriptInterface
   374→        public String tryAutoRevive() {
   375→            org.json.JSONObject out = new org.json.JSONObject();
   376→            try {
   377→                if (AdSkipService.isAlive()) {
   378→                    boolean wasPaused = !AdSkipService.masterState(a);
   379→                    if (wasPaused) AdSkipService.setMaster(a, true); // 恢复即全开
   380→                    out.put("ok", true).put("running", true).put("resumed", wasPaused);
   381→                    return out.toString();
   382→                }
   383→                boolean enabled = a11yEnabledInSettings();
   384→                out.put("enabledInSettings", enabled);
   385→                if (!enabled) {
   386→                    out.put("ok", false).put("reason", "disabled");
   387→                    return out.toString();
   388→                }
   389→                // 设置仍登记：进程已拉起，AMS 自动重绑——深度观察 6s
   390→                //（每 500ms 踢一次 AccessibilityManager，重绑慢的 ROM 也能等到）
   391→                long deadline = System.currentTimeMillis() + 6000;
   392→                while (System.currentTimeMillis() < deadline && !AdSkipService.isAlive()) {
   393→                    kickA11yManager();
   394→                    Thread.sleep(500);
   395→                }
   396→                boolean revived = AdSkipService.isAlive();
   397→                if (revived) {
   398→                    boolean wasPaused = !AdSkipService.masterState(a);
   399→                    if (wasPaused) AdSkipService.setMaster(a, true); // 拉活成功 → 暂停自动解除
   400→                    out.put("ok", true).put("revived", true).put("resumed", wasPaused);
   401→                } else {
   402→                    out.put("ok", false).put("reason", "need_toggle"); // 页面提示重试/手动，绝不自动跳系统设置
   403→                }
   404→            } catch (Throwable t) {
   405→                try {
   406→                    out.put("ok", false).put("reason", "error");
   407→                } catch (Exception ignored) {
   408→                }
   409→            }
   410→            return out.toString();
   411→        }

   413→        /** v4.4.0：异步深度拉活（后台线程 6s 深挖，不阻塞 JS Bridge 线程；
   414→         *  结果经 window.__reviveCb(name, json) 回调） */
   415→        @JavascriptInterface
   416→        public void tryAutoReviveDeep(final String cb) {
   417→            new Thread(() -> {
   418→                final String r = tryAutoRevive();
   419→                a.runOnUiThread(() -> {
   420→                    try {
   421→                        if (a.webView != null) {
   422→                            a.webView.evaluateJavascript(
   423→                                    "window.__reviveCb && window.__reviveCb("
   424→                                            + JSON.stringifyCb(cb) + "," + r + ")", null);
   425→                        }
   426→                    } catch (Throwable ignored) {
   427→                    }
   428→                });
   429→            }, "revive-deep").start();
   430→        }

   432→        /** v4.4.0：查询已启用无障碍服务列表——促使 AMS 校对绑定状态
   433→         *  （进程存活但服务未绑的窗口期，这次查询常触发系统重绑） */
   434→        private void kickA11yManager() {
   435→            try {
   436→                android.view.accessibility.AccessibilityManager am =
   437→                        (android.view.accessibility.AccessibilityManager)
   438→                                a.getSystemService(Context.ACCESSIBILITY_SERVICE);
   439→                if (am != null) {
   440→                    am.getEnabledAccessibilityServiceList(
   441→                            android.accessibilityservice.AccessibilityServiceInfo.FEEDBACK_ALL_MASK);
   442→                }
   443→            } catch (Throwable ignored) {
   444→            }
   445→        }

   718→    /* ═══ v4.5.0 WebView 硬化：URL 劫持防护（攻击面收缩核心）═══
   719→     * 本地离线页装载了带 GitHub 令牌能力的 JS 桥——若页面被诱导跳转到
   720→     * 远程页面（更新横幅里的链接/订阅预览/恶意注入），远程页面将直接
   721→     * 拿到 AndroidApp 桥（netPostIssue 内置令牌、kvSet 任意写）。
   722→     * 对策：shouldOverrideUrlLoading 白名单制——只允许本应用 asset 目录
   723→     * 内导航；http(s) 链接一律转交系统浏览器（无桥）；其余 scheme 全部
   724→     * 拦截。返回 true = 本应用接管，WebView 绝不加载。 */
   725→    private static final class HardenedWebViewClient extends WebViewClient {
   726→        @Override
   727→        public boolean shouldOverrideUrlLoading(WebView view, String url) {
   728→            try {
   729→                if (url == null) return true;
   730→                String u = url.trim();
   731→                if (u.startsWith("file:///android_asset/")) return false; // 本地页面放行
   732→                if (u.startsWith("about:")) return false;                  // 页内占位
   733→                if (u.startsWith("https://") || u.startsWith("http://")) {
   734→                    try {
   735→                        // 外部链接 → 系统浏览器打开（无任何 JS 桥暴露）
   736→                        Intent it = new Intent(Intent.ACTION_VIEW, Uri.parse(u));
   737→                        it.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
   738→                        view.getContext().startActivity(it);
   739→                    } catch (Exception ignored) {
   740→                    }
   741→                }
   742→                // file:// 其他路径 / intent:// / javascript: / 自定义 scheme：全部拦截
   743→                return true;
   744→            } catch (Throwable t) {
   745→                return true;
   746→            }
   747→        }
   748→    }
