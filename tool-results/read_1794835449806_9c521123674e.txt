   574→/* ══════════ 应用内更新（多通道：直连 → 镜像1 → 镜像2；失败/停滞自动切换） ══════════ */
   575→const GH_API = "https://api.github.com/repos/xiaobaiba999/Skip-Ad";
   576→let ubTimer = null;

   579→/* 下载通道状态（直连 + 镜像；failed/停滞 30s 自动切换下一个） */
   580→let dl = { channels: [], idx: 0, fileName: "", prefix: "ub", lastBytes: -1, lastMoveAt: 0 };

   582→function cmpVer(a, b) {
   583→  const pa = String(a).replace(/^v/, "").split("."), pb = String(b).replace(/^v/, "").split(".");
   584→  for (let i = 0; i < 3; i++) {
   585→    const d = (parseInt(pa[i], 10) || 0) - (parseInt(pb[i], 10) || 0);
   586→    if (d) return d > 0 ? 1 : -1;
   587→  }
   588→  return 0;
   589→}

   591→async function fetchLatest() {
   592→  if (HAS_BRIDGE) {
   593→    // 原生侧已带令牌 + 两重重试 + /releases 列表回退（v3.3.0）
   594→    try { return JSON.parse(AndroidApp.netGetLatestRelease()); } catch (e) { return { ok: false, error: "解析失败" }; }
   595→  }
   596→  try {
   597→    const r = await fetch(GH_API + "/releases/latest", { headers: { Accept: "application/vnd.github+json" } });
   598→    if (r.status === 404) return { ok: true, hasRelease: false };
   599→    if (r.ok) {
   600→      const j = await r.json();
   601→      const apk = (j.assets || []).filter(a => a && a.name && a.name.endsWith(".apk"))[0];
   602→      return {
   603→        ok: true, hasRelease: true, tag: j.tag_name, notes: j.body, url: j.html_url,
   604→        apkUrl: apk ? apk.browser_download_url : "",
   605→        apkMirrors: apk ? ["https://ghfast.top/" + apk.browser_download_url, "https://gh-proxy.com/" + apk.browser_download_url] : [],
   606→        apkFileName: apk ? apk.name : "adshield.apk",
   607→        apkMB: apk ? Math.round(apk.size / 1048576 * 100) / 100 : 0,
   608→      };
   609→    }
   610→    return { ok: false, error: "GitHub 返回 " + r.status };
   611→  } catch (e) { return { ok: false, error: "网络异常：" + (e.message || e) }; }
   612→}

   614→function sizeText(mb) { return mb >= 1 ? mb + " MB" : Math.round(mb * 1024) + " KB"; }

   616→/* ── 统一页内下载流程（横幅与关于页共用；多通道自动切换） ── */
   617→function startPageDownload(d, prefix) {
   618→  const fileName = d.apkFileName || "adshield.apk";
   619→  if (!d.apkUrl) { toast("该版本没有 APK 附件"); return; }
   620→  if (!HAS_BRIDGE) { openLink(d.apkUrl); return; } // 浏览器预览：退化为直链
   621→  dl = {
   622→    channels: [d.apkUrl].concat(d.apkMirrors || []),
   623→    idx: 0, fileName, prefix,
   624→    lastBytes: -1, lastMoveAt: Date.now(),
   625→  };
   626→  beginChannel();
   627→}
   628→
   629→function beginChannel() {
   630→  const url = dl.channels[dl.idx];
   631→  try {
   632→    const r = JSON.parse(AndroidApp.startUpdateDownload(url, dl.fileName));
   633→    if (r.ok) {
   634→      const btn = $("#" + dl.prefix + "-btn");
   635→      if (btn) { btn.textContent = "下载中…"; btn.disabled = true; }
   636→      const wrap = $("#" + dl.prefix + "-progress-wrap");
   637→      if (wrap) wrap.style.display = "";
   638→      dl.lastBytes = -1;
   639→      dl.lastMoveAt = Date.now();
   640→      startProgressPoll(dl.prefix);
   641→    } else toast(r.error || "启动下载失败");
   642→  } catch (e) { toast("启动下载失败"); }
   643→}
   644→
   645→/* 切换下一个下载通道；无通道可用时返回 false */
   646→function switchChannel(reason) {
   647→  if (dl.idx + 1 < dl.channels.length) {
   648→    dl.idx++;
   649→    toast(reason + "，已切换备用通道（" + dl.idx + "/" + (dl.channels.length - 1) + "）");
   650→    beginChannel();
   651→    return true;
   652→  }
   653→  return false;
   654→}
   655→
   656→function startProgressPoll(prefix) {
   657→  if (ubTimer) clearInterval(ubTimer);
   658→  ubTimer = setInterval(() => {
   659→    try {
   660→      const d = JSON.parse(AndroidApp.updateProgress());
   661→      const bar = $("#" + prefix + "-bar"), txt = $("#" + prefix + "-text");
   662→      if (d.state === "running" || d.state === "paused") {
   663→        bar.style.width = d.percent + "%";
   664→        txt.textContent = d.percent + "% · " + (d.bytes / 1048576).toFixed(1) + " / " + (d.total / 1048576).toFixed(1) + " MB";
   665→        // 停滞检测：30s 无新字节 → 判定该通道不可用，自动切换
   666→        if (d.bytes !== dl.lastBytes) { dl.lastBytes = d.bytes; dl.lastMoveAt = Date.now(); }
   667→        else if (Date.now() - dl.lastMoveAt > 30000) {
   668→          if (switchChannel("下载长时间无进展")) return;
   669→          dl.lastMoveAt = Date.now(); // 无通道可切：再等一轮
   670→        }
   671→      } else if (d.state === "done") {
   672→        clearInterval(ubTimer); ubTimer = null;
   673→        bar.style.width = "100%";
   674→        txt.textContent = "下载完成 ✓ 点击安装即覆盖升级（学习数据保留）";
   675→        const btn = $("#" + prefix + "-btn");
   676→        btn.textContent = "立即安装";
   677→        btn.disabled = false;
   678→        btn.dataset.phase = "install";
   679→      } else if (d.state === "failed") {
   680→        clearInterval(ubTimer); ubTimer = null;
   681→        if (switchChannel("直连下载失败")) return;
   682→        txt.textContent = "下载失败（直连与备用通道均不可用），可稍后重试";
   683→        const btn = $("#" + prefix + "-btn");
   684→        btn.textContent = "重新下载";
   685→        btn.disabled = false;
   686→        btn.dataset.phase = "download";
   687→      }
   688→    } catch (e) { clearInterval(ubTimer); ubTimer = null; }
   689→  }, 800);
   690→}
