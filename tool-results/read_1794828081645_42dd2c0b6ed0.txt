   375→   375→                if (entering) rootWinCls.put(pkg, cls); // v3.6.0：记录入口窗
   376→   376→                backTries.put(pkg, 0);
   377→   377→                clickTries.put(pkg, 0); // v3.5.0：新窗重置复验计数

   394→   394→            case AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED: {

   409→   409→                observeMisclick(pkg, event, now);
   410→   410→                break;
   411→   411→            }
   412→   412→        }
   413→   413→    }

   594→   594→    private void tryScan(String pkg, long winStart, long atMs) {
   595→   595→        if (instance == null) return;
   596→   596→        long now = System.currentTimeMillis();
   597→   597→        long elapsed = now - winStart;
   598→   598→        boolean splashLive = learningEnabled() && elapsed <= WINDOW_MS;
   599→   599→        boolean popupLive = popupShieldEnabled() && now < popupUntilGet(pkg) && popupCooldownOK(pkg, now);
   600→   600→        if (!splashLive && !popupLive) return;
   601→   601→
   602→   602→        try {
   603→   603→            /* 多窗口扫描：活动窗口优先，其次 getWindows() 同包窗口（弹窗/覆盖层） */
   604→   604→            AccessibilityNodeInfo[] roots = collectRoots(pkg);
   605→   605→            if (roots.length == 0) return;
   606→   606→
   607→   607→            Long lastClick = lastClickAt.get(pkg);
   608→   608→            boolean clickCoolOK = lastClick == null || now - lastClick >= COOLDOWN_MS;
   609→   609→            // v3.5.0：学习位置指纹（手动教过的坐标）与复验次数
   610→   610→            int[] tapHint = learnedPos(pkg);
   611→   611→            Integer triesBox = clickTries.get(pkg);
   612→   612→            int clickAttempts = triesBox == null ? 0 : triesBox.intValue();
   613→   613→
   614→   614→            for (int ri = 0; ri < roots.length; ri++) {
   615→   615→                energyScans++;
   616→   616→                Ctx ctx = new Ctx();
   617→   617→                ctx.screenW = screenWidth;
   618→   618→                ctx.screenH = screenHeight;
   619→   619→                ctx.elapsedMs = Math.max(0, elapsed - ri * 30);
   620→   620→                ctx.splashLive = splashLive;
   621→   621→                ctx.popupLive = popupLive;
   622→   622→                ctx.guard = AdDecisionCore.isGuardPkg(pkg);
   623→   623→                // v3.6.0 毒链① 根治：sdkAdWin 只认【当前窗口类名】是 SDK 广告窗
   624→   624→                //（v3.5.0 的 12s TTL 让主页还拿着 SDK 资格按返回键 → 退出应用）
   625→   625→                ctx.sdkAdWin = AdDecisionCore.isAdActivityClass(winClassOf(pkg));
   626→   626→                Boolean learned = winClsLearned.get(pkg);
   627→   627→                ctx.learnedCls = learned != null && learned.booleanValue();
   628→   628→                // v3.6.0 毒链② 根治：根窗口（入口窗）标记——返回键=退出应用，
   629→   629→                // L3/L4.5/L5 决策核心禁用 + execute() 双保险拦截
   630→   630→                ctx.rootWindow = winClassOf(pkg).equals(rootWinClsOf(pkg));
   631→   631→                ctx.clickAttempts = clickAttempts;
   632→   632→                ctx.tapHintX = tapHint[0];
   633→   633→                ctx.tapHintY = tapHint[1];
   634→   634→                ctx.budgetNodes = splashLive ? 600 : 280; // 开屏窗预算扩容（重树）
   635→   635→
   636→   636→                ScanFind f = AdDecisionCore.walk(new NodeAdapter(roots[ri]), ctx);
   637→   637→                if (f.foundSkip || f.gyroHint || f.fakeSkip || f.hotZone || ctx.sdkAdWin || ctx.learnedCls) {
   638→   638→                    Long old = adFeatureUntil.get(pkg);
   639→   639→                    if (old == null || now > old) adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   640→   640→                }
   641→   641→                Decision d = AdDecisionCore.decide(f, ctx);
   642→   642→                if (d.action.equals(AdDecisionCore.A_NONE)) continue;
   643→   643→                if (execute(pkg, d, f, ctx, clickCoolOK, now)) return; // 命中即熔断
   644→   644→            }
   645→   645→        } catch (Exception ignored) {
   646→   646→        }
   647→   647→    }

   676→   676→    /** 执行决策；返回 true = 已出手（熔断扫描链） */
   677→   677→    private boolean execute(String pkg, Decision d, ScanFind f, Ctx ctx, boolean clickCoolOK, long now) {
   678→   678→        String how = d.action + "/" + d.reason;
   679→   679→        lastGesturePkg = pkg; // v3.7.0：手势取消诊断日志用
   680→   680→        // v3.6.0：点击梯子预算——同窗点击 ≥MAX_CLICKS_PER_WIN 即优雅收手
   681→   681→        //（绝不无限重试，更绝不升级返回键；等用户手动跳过，手动点击仍会被学习）
   682→   682→        if ((d.action.equals(AdDecisionCore.A_CLICK_BOUNDS) || d.action.equals(AdDecisionCore.A_CLOSE)
   683→   683→                || d.action.equals(AdDecisionCore.A_TAP_POINT))
   684→   684→                && triesOfClick(pkg) >= AdDecisionCore.MAX_CLICKS_PER_WIN) {
   685→   685→            Boolean done = gaveUp.get(pkg);
   686→   686→            if (done == null || !done.booleanValue()) {
   687→   687→                gaveUp.put(pkg, Boolean.TRUE);
   688→   688→                appendLog("🤲 点击梯子已用尽（" + shortPkg(pkg) + "）：" + AdDecisionCore.MAX_CLICKS_PER_WIN
   689→   689→                        + " 次尝试未关闭广告 → 优雅收手（本窗不再动作；你的手动跳过仍会被学习）");
   690→   690→            }
   691→   691→            return false;
   692→   692→        }
   693→   693→        if (d.action.equals(AdDecisionCore.A_BACK)) {
   694→   694→            // v3.6.0 根窗口安全门（双保险，决策核心已先拦一道）：
   695→   695→            // 入口窗上按返回键 = 退出应用（喜马拉雅「再按一次退出到桌面」事故）
   696→   696→            if (ctx.rootWindow && !ctx.sdkAdWin) {
   697→   697→                appendLog("⛔ 根窗口安全门（" + shortPkg(pkg) + "）：入口窗禁返回键（防退出应用），只走点击梯子");
   698→   698→                return false;
   699→   699→            }
   700→   700→            if (triesOf(pkg) >= BACK_RETRY_MAX || !backAllowed(now)) return false;
   701→   701→            performGlobalAction(GLOBAL_ACTION_BACK);
   702→   702→            energyBacks++;
   703→   703→            backTries.put(pkg, triesOf(pkg) + 1);
   704→   704→            adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   705→   705→            if (d.reason.startsWith("sdk_window")) learnClass(pkg, winClassOf(pkg));
   706→   706→            if ("click_verify_fail".equals(d.reason)) {
   707→   707→                appendLog("🛡 点击复验失败升级（" + shortPkg(pkg) + "）：同窗 " + ctx.clickAttempts
   708→   708→                        + " 次点击仍未关闭 → 返回键兜底温和关闭（不碰进程与服务）");
   709→   709→            } else {
   710→   710→                appendLog("🛡 广告窗直断（" + shortPkg(pkg) + "）：" + reasonZh(d.reason)
   711→   711→                        + " · " + ctx.elapsedMs + "ms → 返回键温和关闭（不碰进程与服务）");
   712→   712→            }
   713→   713→            bumpStat(pkg, "back");
   714→   714→            /* v3.7.0（issue#10 微博根因）：返回键直断后【不熔断】——广告 Activity
   715→   715→             * 常拦截返回键（按了没反应）。600ms 后复验：同包窗口仍在则重开扫描链
   716→   716→             * 继续点击梯子（角落候选/坐标轮换），不再「按完就睡」 */
   717→   717→            handler.postDelayed(() -> {
   718→   718→                if (instance == null) return;
   719→   719→                try {
   720→   720→                    AccessibilityNodeInfo[] roots = collectRoots(pkg);
   721→   721→                    if (roots.length > 0) {
   722→   722→                        appendLog("🔁 返回键复验（" + shortPkg(pkg) + "）：广告窗仍在（返回键被拦截）→ 继续点击梯子");
