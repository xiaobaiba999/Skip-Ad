   420→/* ══════════ 应用内更新（多通道：直连 → 镜像1 → 镜像2；失败/停滞自动切换） ══════════ */
   421→const GH_API = "https://api.github.com/repos/xiaobaiba999/Skip-Ad";
   422→let ubTimer = null;
   423→let latestRelease = null; // 最近一次检查结果（关于页与横幅共用）

   428→function cmpVer(a, b) {
   429→  const pa = String(a).replace(/^v/, "").split("."), pb = String(b).replace(/^v/, "").split(".");
   430→  for (let i = 0; i < 3; i++) {
   431→    const d = (parseInt(pa[i], 10) || 0) - (parseInt(pb[i], 10) || 0);
   432→    if (d) return d > 0 ? 1 : -1;
   433→  }
   434→  return 0;
   435→}

   437→async function fetchLatest() {
   438→  if (HAS_BRIDGE) {
   439→    // 原生侧已带令牌 + 两重重试 + /releases 列表回退（v3.3.0）
   440→    try { return JSON.parse(AndroidApp.netGetLatestRelease()); } catch (e) { return { ok: false, error: "解析失败" }; }
   441→  }
   442→  try {
   443→    const r = await fetch(GH_API + "/releases/latest", { headers: { Accept: "application/vnd.github+json" } });
   444→    if (r.status === 404) return { ok: true, hasRelease: false };
   445→    if (r.ok) {
   446→      const j = await r.json();
   447→      const apk = (j.assets || []).filter(a => a && a.name && a.name.endsWith(".apk"))[0];
   448→      return {
   449→        ok: true, hasRelease: true, tag: j.tag_name, notes: j.body, url: j.html_url,
   450→        apkUrl: apk ? apk.browser_download_url : "",
   451→        apkMirrors: apk ? ["https://ghfast.top/" + apk.browser_download_url, "https://gh-proxy.com/" + apk.browser_download_url] : [],
   452→        apkFileName: apk ? apk.name : "adshield.apk",
   453→        apkMB: apk ? Math.round(apk.size / 1048576 * 100) / 100 : 0,
   454→      };
   455→    }
   456→    return { ok: false, error: "GitHub 返回 " + r.status };
   457→  } catch (e) { return { ok: false, error: "网络异常：" + (e.message || e) }; }
   458→}

   460→function sizeText(mb) { return mb >= 1 ? mb + " MB" : Math.round(mb * 1024) + " KB"; }

   462→/* ── 统一页内下载流程（横幅与关于页共用；多通道自动切换） ── */
   463→function startPageDownload(d, prefix) {
   464→  const fileName = d.apkFileName || "adshield.apk";
   465→  if (!d.apkUrl) { toast("该版本没有 APK 附件"); return; }
   466→  if (!HAS_BRIDGE) { openLink(d.apkUrl); return; } // 浏览器预览：退化为直链
   467→  dl = {
   468→    channels: [d.apkUrl].concat(d.apkMirrors || []),
   469→    idx: 0, fileName, prefix,
   470→    lastBytes: -1, lastMoveAt: Date.now(),
   471→  };
   472→  beginChannel();
   473→}
   474→
   475→function beginChannel() {
   476→  const url = dl.channels[dl.idx];
   477→  try {
   478→    const r = JSON.parse(AndroidApp.startUpdateDownload(url, dl.fileName));
   479→    if (r.ok) {
   480→      const btn = $("#" + dl.prefix + "-btn");
   481→      if (btn) { btn.textContent = "下载中…"; btn.disabled = true; }
   482→      const wrap = $("#" + dl.prefix + "-progress-wrap");
   483→      if (wrap) wrap.style.display = "";
   484→      dl.lastBytes = -1;
   485→      dl.lastMoveAt = Date.now();
   486→      startProgressPoll(dl.prefix);
   487→    } else toast(r.error || "启动下载失败");
   488→  } catch (e) { toast("启动下载失败"); }
   489→}
   490→
   491→/* 切换下一个下载通道；无通道可用时返回 false */
   492→function switchChannel(reason) {
   493→  if (dl.idx + 1 < dl.channels.length) {
   494→    dl.idx++;
   495→    toast(reason + "，已切换备用通道（" + dl.idx + "/" + (dl.channels.length - 1) + "）");
   496→    beginChannel();
   497→    return true;
   498→  }
   499→  return false;
   500→}

   502→function startProgressPoll(prefix) {
   503→  if (ubTimer) clearInterval(ubTimer);
   504→  ubTimer = setInterval(() => {
   505→    try {
   506→      const d = JSON.parse(AndroidApp.updateProgress());
   507→      const bar = $("#" + prefix + "-bar"), txt = $("#" + prefix + "-text");
   508→      if (d.state === "running" || d.state === "paused") {
   509→        bar.style.width = d.percent + "%";
   510→        txt.textContent = d.percent + "% · " + (d.bytes / 1048576).toFixed(1) + " / " + (d.total / 1048576).toFixed(1) + " MB";
   511→        // 停滞检测：30s 无新字节 → 判定该通道不可用，自动切换
   512→        if (d.bytes !== dl.lastBytes) { dl.lastBytes = d.bytes; dl.lastMoveAt = Date.now(); }
   513→        else if (Date.now() - dl.lastMoveAt > 30000) {
   514→          if (switchChannel("下载长时间无进展")) return;
   515→          dl.lastMoveAt = Date.now(); // 无通道可切：再等一轮
   516→        }
   517→      } else if (d.state === "done") {
   518→        clearInterval(ubTimer); ubTimer = null;
   519→        bar.style.width = "100%";
   520→        txt.textContent = "下载完成 ✓ 点击安装即覆盖升级（学习数据保留）";
   521→        const btn = $("#" + prefix + "-btn");
   522→        btn.textContent = "立即安装";
   523→        btn.disabled = false;
   524→        btn.dataset.phase = "install";
   525→      } else if (d.state === "failed") {
   526→        clearInterval(ubTimer); ubTimer = null;
   527→        if (switchChannel("直连下载失败")) return;
   528→        txt.textContent = "下载失败（直连与备用通道均不可用），可稍后重试";
   529→        const btn = $("#" + prefix + "-btn");
   530→        btn.textContent = "重新下载";
   531→        btn.disabled = false;
   532→        btn.dataset.phase = "download";
   533→      }
   534→    } catch (e) { clearInterval(ubTimer); ubTimer = null; }
   535→  }, 800);
   536→}
