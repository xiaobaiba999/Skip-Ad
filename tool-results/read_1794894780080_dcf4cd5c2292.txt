   315→    public static ScanFind walk(NodeView root, Ctx ctx) {
   316→        ScanFind f = new ScanFind();
   317→        if (root == null) return f;
   318→        String rootCls = nz(root.cls()).toLowerCase();
   319→        f.dialogForm = rootCls.contains("dialog") || rootCls.contains("popup");
   320→        int[] budget = {ctx.budgetNodes};
   321→        visit(root, 0, budget, f, ctx);
   322→        // v3.8.0 候选排序：真键优先——① 自身可点（无祖先中转）② 文案更短（精确「跳过」
   323→        // 优先于「5s后可跳过」类长文案）③ 面积更小。防「第 1 候选是广告图内假文本」
   324→        try {
   325→            java.util.Collections.sort(f.skipCands, (a, b) -> {
   326→                if (a.selfClick != b.selfClick) return a.selfClick ? -1 : 1;
   327→                int la = a.label == null ? 0 : a.label.length();
   328→                int lb = b.label == null ? 0 : b.label.length();
   329→                if (la != lb) return la - lb;
   330→                return Double.compare(areaRatio(a.tapBounds, ctx), areaRatio(b.tapBounds, ctx));
   331→            });
   332→        } catch (Throwable ignored) {
   333→        }
   334→        return f;
   335→    }

   337→    private static void visit(NodeView node, int depth, int[] budget, ScanFind f, Ctx ctx) {
   338→        if (node == null || depth > 32 || budget[0] <= 0) return;
   339→        budget[0]--;
   340→        f.nodeCount++;
   341→        try {
   342→            String text = nz(node.text());
   343→            String desc = nz(node.desc());
   344→            String label = text.length() > 0 ? text : desc;
   345→            String cls = nz(node.cls()).toLowerCase();
   346→            boolean vis = node.visible();
   347→            if (label.length() > LONG_TEXT_LEN) f.longText = true; // 长文本=正常内容页（v3.5.0）
   348→            if (node.editable() || cls.contains("edittext")) f.editPresent = true;
   349→            if (node.scrollable() || cls.contains("scrollview") || cls.contains("recyclerview")
   350→                    || cls.contains("viewpager") || cls.contains("listview")) f.scrollPresent = true;
   351→
   352→            int[] b = node.bounds();
   353→            boolean clickable = node.clickable();
   354→            double areaRatio = areaRatio(b, ctx);
   355→
   356→            // 全屏热区：可点且 ≥55% 屏幕
   357→            if (!f.hotZone && clickable && areaRatio >= HOT_ZONE_AREA) f.hotZone = true;
   358→
   359→            if (label.length() > 0) {
   360→                if (label.length() <= 14 && matchesSkip(label)) {
   361→                    int cd = parseCountdown(label);
   362→                    if (cd > f.countdownN) f.countdownN = cd; // v3.8.0 倒计时解析（「跳过 3」→3，服务侧门控用）
   363→                    NodeView click = resolveClickable(node, 4);
   364→                    if (click != null && isSmallClick(click, ctx) && vis) {
   365→                        if (!f.foundSkip) {
   366→                            f.foundSkip = true;
   367→                            f.skipNode = click;
   368→                            // v3.6.0：手势点击用【原标签控件】坐标（点得准），
   369→                            // 不再用可点祖先中心（祖先常是全屏容器，点中心=点开广告，
   370→                            // 真机「已点击跳过但广告纹丝不动」根因之一）；performAction 仍走祖先
   371→                            f.skipBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   372→                            f.skipLabel = label;
   373→                        }
   374→                        // v3.7.0 多候选收集：广告图内常嵌假「跳过」文本（或同窗多个跳过键），
   375→                        // 复验第 N 次点第 N 个候选——不再同一坐标点四遍（issue#10 根因）
   376→                        if (f.skipCands.size() < SKIP_CANDS_MAX) {
   377→                            SkipCand c = new SkipCand();
   378→                            c.clickNode = click;
   379→                            c.tapBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   380→                            c.label = label;
   381→                            c.selfClick = (click == node);
   382→                            f.skipCands.add(c);
   383→                        }
   384→                    } else {
   385→                        f.fakeSkip = true; // 大面积/不可点 = 图内假跳过陷阱
   386→                        // v3.8.0：假跳过≠不可点——大容器可点祖先吞掉 performAction 时，
   387→                        // 标签自身坐标仍是有效触点（GKD clickCenter 方法论）：手势点标签
   388→                        // 位置，由应用按坐标分发到真跳过键。仅收角落形态小标签（防中央诱饵）
   389→                        if (vis && b != null && areaRatio(b, ctx) < FAKE_TAP_AREA_MAX
   390→                                && learnableSkipShape(b, ctx.screenW, ctx.screenH)
   391→                                && f.fakeCands.size() < SKIP_CANDS_MAX) {
   392→                            SkipCand c = new SkipCand();
   393→                            c.clickNode = click;
   394→                            c.tapBounds = b;
   395→                            c.label = label;
   396→                            c.gestureOnly = true;
   397→                            f.fakeCands.add(c);
   398→                        }
   399→                    }
   400→                } else if (label.length() <= 22 && matchesAny(label, GYRO_HINTS)) {
   401→                    f.gyroHint = true;
   402→                } else if (label.length() <= 6 && isCloseText(label)) {
   403→                    NodeView click = resolveClickable(node, 4);
   404→                    if (click != null && isSmallClick(click, ctx) && vis && !f.foundClose) {
   405→                        f.foundClose = true;
   406→                        f.closeNode = click;
   407→                        // v3.6.0：同 L1——手势点原标签坐标，performAction 走可点祖先
   408→                        f.closeBounds = (b != null && areaRatio(b, ctx) < SKIP_AREA_MAX) ? b : click.bounds();
   409→                        f.closeLabel = label;
   410→                    }
   411→                } else if (isAdBadge(label)) {
   412→                    f.adBadge = true;
   413→                    f.badgeLabel = label;
   414→                } else if (label.length() <= 14 && matchesAny(label, CTA_TEXTS)) {
   415→                    f.ctaCount++;
   416→                }
   417→            } else if (b != null && vis && node.childCount() == 0 && isVidSkip(node)) {
   418→                // v3.8.0：viewId 跳过键（GKD 全局规则同款）：vid 含 skip（穿山甲 tt_splash_skip_btn /
   419→                // 微博 iv_skip 类零文本图片键）或倒计时控件（count+down 非 download）——
   420→                // 零文本图片跳过键的头号信号，此前只能靠角落形态猜，猜不中就只能返回键
   421→                NodeView click = resolveClickable(node, 4);
   422→                if (click != null && isSmallClick(click, ctx)) {
   423→                    if (!f.foundSkip) {
   424→                        f.foundSkip = true;
   425→                        f.skipNode = click;
   426→                        f.skipBounds = b;
   427→                        f.skipLabel = "跳过";
   428→                    }
   429→                    if (f.skipCands.size() < SKIP_CANDS_MAX) {
   430→                        SkipCand c = new SkipCand();
   431→                        c.clickNode = click;
   432→                        c.tapBounds = b;
   433→                        c.label = "跳过";
   434→                        c.selfClick = (click == node);
   435→                        f.skipCands.add(c);
   436→                    }
   437→                } else if (areaRatio(b, ctx) < FAKE_TAP_AREA_MAX
   438→                        && learnableSkipShape(b, ctx.screenW, ctx.screenH)
   439→                        && f.fakeCands.size() < SKIP_CANDS_MAX) {
   440→                    SkipCand c = new SkipCand();
   441→                    c.clickNode = click;
   442→                    c.tapBounds = b;
   443→                    c.label = "跳过";
   444→                    c.gestureOnly = true;
   445→                    f.fakeCands.add(c);
   446→                }
   447→                int cd = parseCountdown(node.viewId());
   448→                if (cd > f.countdownN) f.countdownN = cd;
   449→            } else if (b != null && vis && f.corners.size() < MAX_CORNERS && isCornerControl(b, ctx)
   450→                    && !hasCornerChild(node, ctx)) {
   451→                // 【v3.5.0】角落无字小可点控件（多候选）：疑似图片跳过键
   452→                // （微博/喜马拉雅第一方开屏）；自身不可点时向上找可点祖先；
   453→                // 若还有更小的角落子控件则让位（子控件坐标点得更准）
   454→                NodeView click = clickable ? node : resolveClickable(node, 4);
   455→                if (click != null) {
   456→                    CornerCand c = new CornerCand();
   457→                    c.clickNode = click;
   458→                    c.tapBounds = b;
   459→                    f.corners.add(c);
   460→                }
   461→            }
   462→
   463→            int n = node.childCount();
   464→            for (int i = 0; i < n; i++) {
   465→                visit(node.child(i), depth + 1, budget, f, ctx);
   466→            }
   467→        } catch (Throwable ignored) {
   468→        }
   469→    }

   471→    /** 顶部区域判定（开屏×关闭键防误点：真广告×在顶部，隐私/业务弹窗按钮在底部） */
   472→    static boolean inTopArea(int[] b, Ctx ctx) {
   473→        if (b == null || ctx.screenH <= 0) return false;
   474→        int cy = (b[1] + b[3]) / 2;
   475→        return cy <= ctx.screenH * CLOSE_TOP_MAX;
   476→    }

   478→    /** 是否存在更小的角落子控件（有则父容器让位：子控件坐标点得更准） */
   479→    static boolean hasCornerChild(NodeView node, Ctx ctx) {
   480→        int n = node.childCount();
   481→        for (int i = 0; i < n; i++) {
   482→            NodeView c = node.child(i);
   483→            if (c == null) continue;
   484→            if (nz(c.text()).length() > 0 || nz(c.desc()).length() > 0) continue;
   485→            int[] cb = c.bounds();
   486→            if (cb != null && isCornerControl(cb, ctx)) return true;
   487→        }
   488→        return false;
   489→    }

   491→    /** 角落控件判定：顶部 14% 高内、贴右缘(≥72%)或左缘(≤28%)、边长 24px~35% 宽、无任何文本 */
   492→    static boolean isCornerControl(int[] b, Ctx ctx) {
   493→        if (b == null || ctx.screenW <= 0 || ctx.screenH <= 0) return false;
   494→        int w = b[2] - b[0];
   495→        int h = b[3] - b[1];
   496→        if (w < CORNER_MIN_PX || h < CORNER_MIN_PX) return false;
   497→        if (w > ctx.screenW * 0.35 || h > ctx.screenH * 0.20) return false;
   498→        if (b[1] > ctx.screenH * CORNER_TOP_MAX) return false;
   499→        boolean rightEdge = b[2] >= ctx.screenW * CORNER_RIGHT_MIN;
   500→        boolean leftEdge = b[0] <= ctx.screenW * CORNER_LEFT_MAX;
   501→        return rightEdge || leftEdge;
   502→    }

   504→    /** 向上找可点击祖先（最多 4 层） */
   505→    static NodeView resolveClickable(NodeView node, int maxUp) {
   506→        NodeView cur = node;
   507→        for (int i = 0; i < maxUp && cur != null; i++) {
   508→            if (cur.clickable()) return cur;
   509→            cur = cur.parent();
   510→        }
   511→        return null;
   512→    }
   513→
   514→    static boolean isSmallClick(NodeView n, Ctx ctx) {
   515→        if (n == null || !n.clickable()) return false;
   516→        int[] b = n.bounds();
   517→        if (ctx.screenW <= 0 || b == null) return true; // 无屏幕信息时放宽
   518→        double ratio = areaRatio(b, ctx);
   519→        int w = b[2] - b[0];
   520→        return ratio < SKIP_AREA_MAX && w < ctx.screenW * SKIP_WIDTH_MAX;
   521→    }
   522→
   523→    static double areaRatio(int[] b, Ctx ctx) {
   524→        if (b == null || ctx.screenW <= 0 || ctx.screenH <= 0) return 0;
   525→        double area = (double) (b[2] - b[0]) * (b[3] - b[1]);
   526→        return area / ((double) ctx.screenW * ctx.screenH);
   527→    }

   547→    /**
   548→     * v3.6.0 学习形状判定（比扫描用的 isCornerControl 更宽）：
   549→     * 手动点击教学时，跳过键可能是——
   550→     *  · 顶部 25% 区域内的小控件（最常见：右上「跳过」图片键）；
   551→     *  · 底部 20% 区域内的小控件（部分应用跳过键在右下角，真机反馈）；
   552→     *  · 标准角落形态（顶部 14% + 贴左/右缘）。
   553→     * 面积上限由调用方按 POS_LEARN_AREA_MAX 校验（大热区=CTA，不学）。
   554→     */
   555→    public static boolean learnableSkipShape(int[] b, int screenW, int screenH) {
   556→        if (b == null || screenW <= 0 || screenH <= 0) return false;
   557→        if (cornerLike(b, screenW, screenH)) return true;
   558→        int cy = (b[1] + b[3]) / 2;
   559→        boolean topArea = cy <= screenH * 0.25;
   560→        boolean bottomArea = cy >= screenH * 0.80;
   561→        boolean small = (b[2] - b[0]) <= screenW * 0.5 && (b[3] - b[1]) <= screenH * 0.15;
   562→        return (topArea || bottomArea) && small;
   563→    }

   575→    public static Decision decide(ScanFind f, Ctx ctx) {
   576→        /* ── 开屏/插屏窗内 ── */
   577→        if (ctx.splashLive) {
   578→            /* v3.7.0 学习位置【前置】到 L1 之前（issue#10 根因）：
   579→             * 用户手动点通过的位置 = 真实手指触点真正生效的坐标——比盲点
   580→             * 文本候选可靠得多（文本候选可能是广告图内假「跳过」文本）。
   581→             * 教过一次，下次同窗优先直点这个位置。 */
   582→            // L1.4 学习位置点击（v3.7.0 前置）：手动点过的位置指纹 → 次启同窗 500ms 直点
   583→            //     安全门：仅在广告形态窗（简单树/SDK 窗/学习窗）内出手，
   584→            //     信息流等正常界面（可滚动/有输入框/节点多）绝不点
   585→            if (ctx.tapHintX >= 0 && ctx.tapHintY >= 0 && ctx.clickAttempts == 0
   586→                    && ctx.elapsedMs >= LEARNED_POS_MS
   587→                    && (isAdLike(f) || ctx.sdkAdWin || ctx.learnedCls)) {
   588→                Decision d = new Decision();
   589→                d.action = A_TAP_POINT;
   590→                d.x = ctx.tapHintX;
   591→                d.y = ctx.tapHintY;
   592→                d.reason = "learned_pos";
   593→                return d;
   594→            }
   595→            /* L1 文本/vid 跳过键多候选轮换（真候选 → 假跳过手势候选统一梯子）
   596→             * v3.8.0（issue#10 二轮·GKD 方法论对齐）：
   597→             *  · 新增假跳过手势候选——「跳过」文案+大容器可点祖先（微博实锤：isSmallClick
   598→             *    判假→从不点击只返回键，返回键又被广告拦截）→ 手势直点标签自身坐标
   599→             *    （clickCenter），performAction 大容器=点开广告绝不走；
   600→             *  · walk 内候选排序：自身可点/短文案/小面积优先，真键排前 */
   601→            int R = f.skipCands.size();
   602→            int F = f.fakeCands.size();
   603→            if (R + F > 0) {
   604→                int idx = Math.min(ctx.clickAttempts, R + F - 1);
   605→                if (idx < R) {
   606→                    SkipCand c = f.skipCands.get(idx);
   607→                    return click(c.clickNode, c.tapBounds, "skip_text");
   608→                }
   609→                SkipCand c = f.fakeCands.get(idx - R);
   610→                return click(null, c.tapBounds, "fake_skip_tap"); // 手势专用：大容器祖先不 performAction
   611→            }
   612→            // L1.6 开屏×关闭键（v3.5.0）：顶部小关闭键 + 广告形态树（非长文本内容页）
   613→            //     v3.7.0 树上限放宽（CORNER_TREE_MAX_NODES）：广告窗树常大，
   614→            //     40 节点上限把真关闭键挡在门外（微博根因之一）
   615→            if (f.foundClose && f.closeBounds != null && !f.longText
   616→                    && (isAdLikeLoose(f) || ctx.sdkAdWin || ctx.learnedCls)
   617→                    && inTopArea(f.closeBounds, ctx)
   618→                    && ctx.elapsedMs >= SPLASH_CLOSE_MS) {
   619→                return click(f.closeNode, f.closeBounds, "splash_close");
   620→            }
   621→            // L4 角落无字小可点控件（图片跳过键，多候选）→ 手势点击其中心
   622→            //     v3.5.0：复验失败换下一个候选（第 N 次点击用第 N 个候选点）
   623→            //     v3.6.0：改纯手势（node=null）——performAction 可能落到大热区
   624→            //     祖先（=点开广告本体）；手势点子控件坐标=真实手指点跳过键
   625→            //     v3.7.0：① 前置到 L3 之前（微博根因：陀螺仪文案 120ms 抢先
   626→            //     返回键，角落键 500ms 才轮到——有点击候选先点击，点不到再断）；
   627→            //     ② 树上限放宽 CORNER_TREE_MAX_NODES=150（广告窗树常大）
   628→            //     v3.8.0：前置到 L2 直断之前——角落键 500ms 时机先于 700ms 直断，
   629→            //     有可点目标绝不先按返回键
   630→            if (!f.corners.isEmpty() && ctx.clickAttempts < f.corners.size()
   631→                    && !f.scrollPresent && !f.editPresent
   632→                    && f.nodeCount <= CORNER_TREE_MAX_NODES
   633→                    && ctx.elapsedMs >= CORNER_MIN_MS && ctx.elapsedMs <= CORNER_MAX_MS
   634→                    // v3.9.0 广告形态确认门（不乱点红线）：角落无字控件必须
   635→                    // 同时具备广告信号（热区/角标/SDK 窗/学习窗）或树极简
   636→                    //（≤60 节点）才出手——普通主页角落图标绝不误点
   637→                    && (f.hotZone || f.adBadge || ctx.sdkAdWin || ctx.learnedCls
   638→                        || f.nodeCount <= CORNER_CONFIRM_NODES)) {
   639→                CornerCand c = f.corners.get(ctx.clickAttempts);
   640→                return click(null, c.tapBounds, "corner_image_skip");
   641→            }
   642→            // L2 SDK / 学习 className 窗口指纹 → 返回键直断
   643→            //    SDK 广告窗是独立 Activity，返回回到应用内页——根窗口也安全。
   644→            //    v3.6.0 学习指纹快断加双保险（真机事故：毒指纹把主页学成广告窗
   645→            //    → 入口窗快断=按返回键退出应用）：① 仅非根窗口或 SDK 窗可快断；
   646→            //    ② 须为广告形态树（信息流等可滚动页面绝不快断）
   647→            boolean sdkCut = ctx.sdkAdWin && ctx.elapsedMs >= CLS_BREAK_MS;
   648→            boolean learnedCut = ctx.learnedCls && ctx.elapsedMs >= CLS_LEARNED_BREAK_MS
   649→                    && (!ctx.rootWindow || ctx.sdkAdWin) && isAdLike(f);
   650→            if (sdkCut || learnedCut) {
   651→                return back(learnedCut && !sdkCut ? "sdk_window_learned" : "sdk_window");
   652→            }
   653→            /* v3.6.0 根窗口返回键安全门 ─────────────────────────────
   654→             * 应用入口窗（冷启动开屏/回前台首窗）上按返回键 = 退出应用。
   655→             * 真机事故（喜马拉雅）：v3.5.0 返回键兜底 → 喜马拉雅拦截返回
   656→             * 弹「再按一次退出到桌面」→ 引擎 600ms 后再按 → 真退到桌面，
   657→             * 用户「第二次进入就被模拟人操作退出来」。
   658→             * 此后 L3/L4.5/L5 的返回键决策仅在应用内窗（返回回到应用内）
   659→             * 或 SDK 广告窗上生效；根窗口只走点击梯子，用尽优雅收手。 */
   660→            boolean backSafe = !ctx.rootWindow || ctx.sdkAdWin;
   661→            // L3 陀螺仪/假跳过：无倒计时立即直断；带倒计时(假跳过) 800ms 直断
   662→            //     v3.6.0 加信息流安全门：可滚动/有输入框的页面出现「摇一摇/翻转」文案
   663→            //     多为互动玩法或 feed 内容（如「摇一摇抽签」），绝不返回键（防主页误退）
   664→            //     v3.7.0 移到 L4 之后（微博根因）：有可点候选先点击，返回键是
   665→            //     「无任何可点目标」时的兜底（广告 Activity 常拦截返回键）；
   666→            //     gyro_immediate 加「无可点候选」门——角落键 500ms 才到时机，
   667→            //     有候选时等它，不 120ms 抢跑返回键
   668→            boolean clickableCand = !f.corners.isEmpty() || !f.skipCands.isEmpty() || !f.fakeCands.isEmpty() || f.foundClose;
   669→            if (backSafe && f.gyroHint && !f.fakeSkip && !clickableCand
   670→                    && !f.scrollPresent && !f.editPresent
   671→                    && ctx.elapsedMs >= GYRO_IMMEDIATE_MS) {
   672→                return back("gyro_immediate");
   673→            }
   674→            if (backSafe && (f.gyroHint || f.fakeSkip) && !f.scrollPresent && !f.editPresent
   675→                    && ctx.elapsedMs >= GYRO_BREAK_MS) {
   676→                return back("gyro_break");
   677→            }
   678→            // L4.5 点击复验失败升级（v3.5.0）：同窗点击 ≥3 次仍未关闭 → 返回键兜底
   679→            //     v3.6.0：仅应用内窗（返回回到应用内）；根窗口绝不返回键（防退出）
   680→            if (ctx.clickAttempts >= CLICK_VERIFY_MAX && backSafe && isAdLike(f)
   681→                    && ctx.elapsedMs >= GYRO_BREAK_MS) {
   682→                return back("click_verify_fail");
   683→            }
   684→            // L5 全屏热区（自绘图片广告）→ 返回键直断（守护名单豁免；v3.6.0 根窗口豁免）
   685→            if (f.hotZone && backSafe && !ctx.guard && !f.editPresent && f.nodeCount <= HOT_ZONE_MAX_NODES
   686→                    && ctx.elapsedMs >= HOT_ZONE_MS) {
   687→                return back("hot_zone");
   688→            }
   689→            /* v3.9.0：L6 坐标级点击兜底已【整体移除】——固定百分比坐标
   690→             * 在不同屏幕上位置不同，盲点必然乱点（issue#12 闪退毒链：
   691→             * 桌面被误判→盲点桌面图标→拉起微信→回弹踢出用户）。零信号
   692→             * 场景由学习位置（用户教过）与 RuleEngine 规则接管，绝不盲点。 */
   693→        }
   694→
   695→        /* ── 弹窗窗期（独立于开屏窗） ── */
   696→        if (ctx.popupLive && f.foundClose && f.closeBounds != null) {
   697→            boolean strong = f.adBadge;                    // 角标强信号
   698→            boolean weak = f.dialogForm && f.ctaCount >= 2; // Dialog + ≥2 营销话术
   699→            if (strong || weak) {
   700→                Decision d = click(f.closeNode, f.closeBounds, strong ? "popup_badge" : "popup_cta");
   701→                d.action = A_CLOSE;
   702→                return d;
   703→            }
   704→        }
   705→
   706→        return Decision.none();
   707→    }

   849→    public static final class FeedbackFind {
   850→        public boolean isSheet;     // 是「跳过后反馈面板」（强特征 ≥1 且总特征 ≥2）
   851→        public NodeView cancelNode; // 文案精确为「取消」的控件（可空）
   852→        public int[] cancelBounds;  // 取消键边界（手势点击目标）
   853→    }
   854→
   855→    /** 有界广度优先识别「跳过后反馈面板」：统计特征文案命中数（去重）+
   856→     *  找文案精确等于「取消」的可见控件。底部面板树浅，预算 80 零性能顾虑。
   857→     *  只识别不点击——点不点「取消」由服务侧门控（12s 内自动跳过过）决定。 */
   858→    public static FeedbackFind findFeedbackSheet(NodeView root, int budget) {
   859→        FeedbackFind out = new FeedbackFind();
   860→        if (root == null || budget <= 0) return out;
   861→        java.util.ArrayDeque<NodeView> q = new java.util.ArrayDeque<>();
   862→        q.add(root);
   863→        int seen = 0;
   864→        java.util.HashSet<String> hits = new java.util.HashSet<>();
   865→        boolean strong = false;
   866→        while (!q.isEmpty() && seen < budget) {
   867→            NodeView n = q.poll();
   868→            if (n == null) continue;
   869→            seen++;
   870→            String text = nz(n.text());
   871→            String desc = nz(n.desc());
   872→            String label = text.length() > 0 ? text : desc;
   873→            if (label.length() > 0) {
   874→                for (String m : FEEDBACK_SHEET_STRONG) {
   875→                    if (label.contains(m)) { hits.add(m); strong = true; break; }
   876→                }
   877→                for (String m : FEEDBACK_SHEET_WEAK) {
   878→                    if (label.contains(m)) { hits.add(m); break; }
   879→                }
   880→                if (out.cancelNode == null && "取消".equals(text) && n.visible()) {
   881→                    int[] b = n.bounds();
   882→                    if (b != null && b.length >= 4 && b[2] > b[0] && b[3] > b[1]) {
   883→                        out.cancelNode = n;
   884→                        out.cancelBounds = b;
   885→                    }
   886→                }
   887→            }
   888→            for (int i = 0; i < n.childCount(); i++) {
   889→                NodeView c = n.child(i);
   890→                if (c != null) q.add(c);
   891→            }
   892→        }
   893→        out.isSheet = strong && hits.size() >= FEEDBACK_SHEET_MIN_HITS;
   894→        return out;
   895→    }

   950→    /** v3.8.0：解析跳过文案中的倒计时秒数（「跳过 3」「3s后可跳过」「跳过广告 5」→ 3/3/5）。
   951→     *  穿山甲类 SDK 倒计时结束前跳过键不响应点击（点了白点还占冷却）——服务侧据此
   952→     *  门控到「倒计时将尽」再出手，出手即命中。无倒计时返回 0。 */
   953→    public static int parseCountdown(String label) {
   954→        if (label == null) return 0;
   955→        for (int i = 0; i < label.length(); i++) {
   956→            char ch = label.charAt(i);
   957→            if (ch >= '1' && ch <= '9') {
   958→                int v = ch - '0';
   959→                if (i + 1 < label.length()) {
   960→                    char nx = label.charAt(i + 1);
   961→                    if (nx >= '0' && nx <= '9') v = v * 10 + (nx - '0');
   962→                }
   963→                return (v >= 1 && v <= COUNTDOWN_MAX_S) ? v : 0;
   964→            }
   965→            if (ch == '0') return 0; // 「0元领」类营销数字非倒计时
   966→        }
   967→        return 0;
   968→    }
