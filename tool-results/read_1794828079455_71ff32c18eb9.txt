   233→    private void scheduleHeartbeat() {
   234→        try {
   235→            android.app.AlarmManager am = (android.app.AlarmManager) getSystemService(ALARM_SERVICE);
   236→            Intent it = new Intent(this, KeepAliveReceiver.class);
   237→            it.setAction(KeepAliveReceiver.ACTION_SELF_CHECK);
   238→            android.app.PendingIntent pi = android.app.PendingIntent.getBroadcast(
   239→                    this, 8, it,
   240→                    android.app.PendingIntent.FLAG_UPDATE_CURRENT | android.app.PendingIntent.FLAG_IMMUTABLE);
   241→            am.setInexactRepeating(android.app.AlarmManager.RTC,
   242→                    System.currentTimeMillis() + HEARTBEAT_MS, HEARTBEAT_MS, pi);
   243→        } catch (Throwable ignored) {
   244→        }
   245→    }

   260→    @Override
   261→    public void onTaskRemoved(Intent rootIntent) {
   262→        super.onTaskRemoved(rootIntent);
   263→        // v3.6.0：后台清理（滑掉最近任务）时，部分 ROM 会连无障碍服务一起杀。
   264→        // ① 若进程幸存：4s 后自检，服务真死了才发「待恢复」提醒（防误报）
   265→        // ② 自检闹钟（15s）：进程被杀但非 force-stop 的 ROM 上仍可触发提醒
   266→        // v3.7.0：追加 15 分钟心跳闹钟——进程被杀后持续自检，
   267→        //    服务一旦真死即提醒恢复（用户手动关闭 keepOn 时不排）
   268→        try {
   269→            if (!prefs().getBoolean(KEY_KEEPON, false)) return;
   270→            handler.postDelayed(() -> {
   271→                if (instance == null) KeepAliveReceiver.notifyAlert(AdSkipService.this);
   272→            }, 4000);
   273→            scheduleSelfCheckAlarm(15000);
   274→            scheduleHeartbeat();
   275→        } catch (Throwable ignored) {
   276→        }
   277→    }

   279→    /** v3.6.0：自检闹钟——清理后台后若服务已死（且非 force-stop 状态），发恢复提醒 */
   280→    private void scheduleSelfCheckAlarm(long delayMs) {
   281→        try {
   282→            android.app.AlarmManager am = (android.app.AlarmManager) getSystemService(ALARM_SERVICE);
   283→            Intent it = new Intent(this, KeepAliveReceiver.class);
   284→            it.setAction(KeepAliveReceiver.ACTION_SELF_CHECK);
   285→            android.app.PendingIntent pi = android.app.PendingIntent.getBroadcast(
   286→                    this, 7, it,
   287→                    android.app.PendingIntent.FLAG_UPDATE_CURRENT | android.app.PendingIntent.FLAG_IMMUTABLE);
   288→            am.set(android.app.AlarmManager.RTC, System.currentTimeMillis() + delayMs, pi);
   289→        } catch (Throwable ignored) {
   290→        }
   291→    }

   293→    /** v3.6.0：清理误学窗口指纹——只删 cls 规则中非广告窗类的（保留位置/文本规则） */
   294→    private int purgePoisonFingerprints() {
   295→        synchronized (lock) {
   296→            try {
   297→                JSONObject st = loadState();
   298→                JSONObject rules = st.optJSONObject("rules");
   299→                if (rules == null) return 0;
   300→                int purged = 0;
   301→                java.util.Iterator<String> it = rules.keys();
   302→                while (it.hasNext()) {
   303→                    String pkg = it.next();
   304→                    JSONArray list = rules.optJSONArray(pkg);
   305→                    if (list == null) continue;
   306→                    JSONArray kept = new JSONArray();
   307→                    for (int i = 0; i < list.length(); i++) {
   308→                        JSONObject r = list.optJSONObject(i);
   309→                        if (r == null) continue;
   310→                        String cls = r.optString("cls", "");
   311→                        boolean poison = cls.length() > 0
   312→                                && !r.optBoolean("pos", false)
   313→                                && !AdDecisionCore.isAdActivityClass(cls);
   314→                        if (poison) {
   315→                            purged++;
   316→                        } else {
   317→                            kept.put(r);
   318→                        }
   319→                    }

   342→    @Override
   343→    public void onAccessibilityEvent(AccessibilityEvent event) {
   344→        if (event == null) return;
   345→        CharSequence pkgCs = event.getPackageName();
   346→        if (pkgCs == null) return;
   347→        String pkg = pkgCs.toString();
   348→        if (pkg.equals(getPackageName())) return;
   349→        if (isSystemNoise(pkg)) return;
   350→
   351→        long now = System.currentTimeMillis();
   352→        energyEvents++;
   353→
   354→        switch (event.getEventType()) {
   355→            case AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED: {
   356→                String cls = cs(event.getClassName());
   357→                // v3.6.0 根窗口追踪：新进入前台的应用，其首个窗口 = 入口窗
   358→                //（冷启动开屏 / 从桌面·其他应用回前台的首窗）——入口窗上
   359→                // 按返回键 = 退出应用，安全门据此禁一切返回键兜底
   360→                boolean entering = !pkg.equals(lastFgPkg);
   361→                lastFgPkg = pkg;
   362→                if (reboundChainJump(pkg, now)) break;
   363→                if (reboundLanding(pkg, cls, now)) break;
   364→                // v3.5.0 复验通过：上一窗点过东西且窗口已切走 → 广告已关
   365→                Integer prevTries = clickTries.get(pkg);
   366→                if (prevTries != null && prevTries > 0 && !cls.equals(winClassOf(pkg))) {
   367→                    appendLog("✓ 复验通过（" + shortPkg(pkg) + "）：广告窗已关闭"
   368→                            + (prevTries > 1 ? "（共 " + prevTries + " 次尝试，重试梯子兜底）" : ""));
   369→                    // v3.6.0 毒链① 根治：广告已关，学习窗快断资格同步作废
   370→                    //（防止后续窗口还拿着旧资格按返回键）
   371→                    winClsLearned.remove(pkg);
   372→                }
   373→                windowAt.put(pkg, now);
   374→                winClass.put(pkg, cls);
   375→                if (entering) rootWinCls.put(pkg, cls); // v3.6.0：记录入口窗
   376→                backTries.put(pkg, 0);
   377→                clickTries.put(pkg, 0); // v3.5.0：新窗重置复验计数
   378→                gaveUp.put(pkg, Boolean.FALSE); // v3.6.0：新窗重置收手标记
   379→                winClsLearned.remove(pkg);
   380→                if (isDialogClass(cls) && !isOemUtilityPkg(pkg)) {
   381→                    popupUntil.put(pkg, now + POPUP_WINDOW_MS);
   382→                }
   383→                if (AdDecisionCore.isAdActivityClass(cls)) {
   384→                    adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   385→                    appendLog("🎯 SDK 广告窗识别（" + shortPkg(pkg) + "）：" + cls
   386→                            + " → 500ms 内无真跳过键将温和直断");
   387→                } else if (learnedClassHit(pkg, cls)) {
   388→                    adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   389→                    winClsLearned.put(pkg, Boolean.TRUE);
   390→                }
   391→                bumpScanChain(pkg);
   392→                break;
   393→            }
   394→            case AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED: {
   395→                if (now - winAt(pkg) < WINDOW_MS) {
   396→                    nudgeScan(pkg, now);
   397→                    break;
   398→                }
   399→                if (now < popupUntilGet(pkg)) {
   400→                    nudgePopupScan(pkg, now);
   401→                    break;
   402→                }
   403→                detectPopupBurst(pkg, now);
   404→                break;
   405→            }
   406→            case AccessibilityEvent.TYPE_VIEW_CLICKED: {
   407→                if (isOwnAutoClick(pkg, now)) return;
   408→                learnFromManualClick(pkg, event, now);
   409→                observeMisclick(pkg, event, now);
   410→                break;
   411→            }
   412→        }
   413→    }

   529→    private void bumpScanChain(final String pkg) {
   530→        final long winStart = winAt(pkg);
   531→        Integer g = scanGen.get(pkg);
   532→        final int gen = g == null ? 1 : g + 1;
   533→        scanGen.put(pkg, gen);
   534→        for (int i = 0; i < SCAN_AT.length; i++) {
   535→            final long at = SCAN_AT[i];
   536→            handler.postDelayed(() -> {
   537→                Integer cur = scanGen.get(pkg);
   538→                if (cur == null || cur.intValue() != gen) return;
   539→                tryScan(pkg, winStart, at);
   540→            }, at);
   541→        }
   542→    }

   544→    private void nudgeScan(final String pkg, long now) {
   545→        Integer gen = scanGen.get(pkg);
   546→        if (gen == null) {
   547→            bumpScanChain(pkg);
   548→            return;
   549→        }
   550→        final long winStart = winAt(pkg);
   551→        if (now - winStart < 150) return;
   552→        final int g = gen.intValue();
   553→        handler.post(() -> {
   554→            Integer cur = scanGen.get(pkg);
   555→            if (cur == null || cur.intValue() != g) return;
   556→            tryScan(pkg, winStart, -1);
   557→        });
   558→    }

   561→        Long last = lastPopupScanAt.get(pkg);

   563→        lastPopupScanAt.put(pkg, now);
   564→        final long winStart = winAt(pkg);
   565→        handler.post(() -> {
   566→            if (instance == null) return;
   567→            tryScan(pkg, winStart, -2);
   568→        });
   569→    }

   572→        if (!popupShieldEnabled() || isOemUtilityPkg(pkg)) return;

   584→        }
   585→    }

   592→    /* ───────── 扫描与决策执行（逻辑全部在 AdDecisionCore） ───────── */
   593→
   594→    private void tryScan(String pkg, long winStart, long atMs) {
   595→        if (instance == null) return;
   596→        long now = System.currentTimeMillis();
   597→        long elapsed = now - winStart;
   598→        boolean splashLive = learningEnabled() && elapsed <= WINDOW_MS;
   599→        boolean popupLive = popupShieldEnabled() && now < popupUntilGet(pkg) && popupCooldownOK(pkg, now);
   600→        if (!splashLive && !popupLive) return;
   601→
   602→        try {
   603→            /* 多窗口扫描：活动窗口优先，其次 getWindows() 同包窗口（弹窗/覆盖层） */
   604→            AccessibilityNodeInfo[] roots = collectRoots(pkg);
   605→            if (roots.length == 0) return;
   606→
   607→            Long lastClick = lastClickAt.get(pkg);
   608→            boolean clickCoolOK = lastClick == null || now - lastClick >= COOLDOWN_MS;
   609→            // v3.5.0：学习位置指纹（手动教过的坐标）与复验次数
   610→            int[] tapHint = learnedPos(pkg);
   611→            Integer triesBox = clickTries.get(pkg);
   612→            int clickAttempts = triesBox == null ? 0 : triesBox.intValue();
   613→
   614→            for (int ri = 0; ri < roots.length; ri++) {
   615→                energyScans++;
   616→                Ctx ctx = new Ctx();
   617→                ctx.screenW = screenWidth;
   618→                ctx.screenH = screenHeight;
   619→                ctx.elapsedMs = Math.max(0, elapsed - ri * 30);
   620→                ctx.splashLive = splashLive;
   621→                ctx.popupLive = popupLive;
   622→                ctx.guard = AdDecisionCore.isGuardPkg(pkg);
   623→                // v3.6.0 毒链① 根治：sdkAdWin 只认【当前窗口类名】是 SDK 广告窗
   624→                //（v3.5.0 的 12s TTL 让主页还拿着 SDK 资格按返回键 → 退出应用）
   625→                ctx.sdkAdWin = AdDecisionCore.isAdActivityClass(winClassOf(pkg));
   626→                Boolean learned = winClsLearned.get(pkg);
   627→                ctx.learnedCls = learned != null && learned.booleanValue();
   628→                // v3.6.0 毒链② 根治：根窗口（入口窗）标记——返回键=退出应用，
   629→                // L3/L4.5/L5 决策核心禁用 + execute() 双保险拦截
   630→                ctx.rootWindow = winClassOf(pkg).equals(rootWinClsOf(pkg));
   631→                ctx.clickAttempts = clickAttempts;
   632→                ctx.tapHintX = tapHint[0];
   633→                ctx.tapHintY = tapHint[1];
   634→                ctx.budgetNodes = splashLive ? 600 : 280; // 开屏窗预算扩容（重树）
   635→
   636→                ScanFind f = AdDecisionCore.walk(new NodeAdapter(roots[ri]), ctx);
   637→                if (f.foundSkip || f.gyroHint || f.fakeSkip || f.hotZone || ctx.sdkAdWin || ctx.learnedCls) {
   638→                    Long old = adFeatureUntil.get(pkg);
   639→                    if (old == null || now > old) adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   640→                }
   641→                Decision d = AdDecisionCore.decide(f, ctx);
   642→                if (d.action.equals(AdDecisionCore.A_NONE)) continue;
   643→                if (execute(pkg, d, f, ctx, clickCoolOK, now)) return; // 命中即熔断
   644→            }
   645→        } catch (Exception ignored) {
   646→        }
   647→    }

   649→    /** 收集同包可扫描窗口根节点（活动窗口优先，最多 3 个） */
   650→    private AccessibilityNodeInfo[] collectRoots(String pkg) {
   651→        java.util.ArrayList<AccessibilityNodeInfo> out = new java.util.ArrayList<>();
   652→        AccessibilityNodeInfo active = getRootInActiveWindow();
   653→        if (active != null) {
   654→            CharSequence rp = active.getPackageName();
   655→            if (rp != null && pkg.contentEquals(rp)) out.add(active);
   656→        }
   657→        if (out.isEmpty()) {
   658→            try {
   659→                List<AccessibilityWindowInfo> wins = getWindows();
   660→                if (wins != null) {
   661→                    for (AccessibilityWindowInfo w : wins) {
   662→                        if (out.size() >= MAX_WINDOWS_SCAN) break;
   663→                        CharSequence wp = w == null ? null : w.getRoot() == null ? null : w.getRoot().getPackageName();
   664→                        if (wp != null && pkg.contentEquals(wp)) {
   665→                            AccessibilityNodeInfo r = w.getRoot();
   666→                            if (r != null) out.add(r);
   667→                        }
   668→                    }
   669→                }
   670→            } catch (Exception ignored) {
   671→            }
   672→        }
   673→        return out.toArray(new AccessibilityNodeInfo[0]);
   674→    }

   676→    /** 执行决策；返回 true = 已出手（熔断扫描链） */
   677→    private boolean execute(String pkg, Decision d, ScanFind f, Ctx ctx, boolean clickCoolOK, long now) {
   678→        String how = d.action + "/" + d.reason;
   679→        lastGesturePkg = pkg; // v3.7.0：手势取消诊断日志用
   680→        // v3.6.0：点击梯子预算——同窗点击 ≥MAX_CLICKS_PER_WIN 即优雅收手
   681→        //（绝不无限重试，更绝不升级返回键；等用户手动跳过，手动点击仍会被学习）
   682→        if ((d.action.equals(AdDecisionCore.A_CLICK_BOUNDS) || d.action.equals(AdDecisionCore.A_CLOSE)
   683→                || d.action.equals(AdDecisionCore.A_TAP_POINT))
   684→                && triesOfClick(pkg) >= AdDecisionCore.MAX_CLICKS_PER_WIN) {
   685→            Boolean done = gaveUp.get(pkg);
   686→            if (done == null || !done.booleanValue()) {
   687→                gaveUp.put(pkg, Boolean.TRUE);
   688→                appendLog("🤲 点击梯子已用尽（" + shortPkg(pkg) + "）：" + AdDecisionCore.MAX_CLICKS_PER_WIN
   689→                        + " 次尝试未关闭广告 → 优雅收手（本窗不再动作；你的手动跳过仍会被学习）");
   690→            }
   691→            return false;
   692→        }
   693→        if (d.action.equals(AdDecisionCore.A_BACK)) {
   694→            // v3.6.0 根窗口安全门（双保险，决策核心已先拦一道）：
   695→            // 入口窗上按返回键 = 退出应用（喜马拉雅「再按一次退出到桌面」事故）
   696→            if (ctx.rootWindow && !ctx.sdkAdWin) {
   697→                appendLog("⛔ 根窗口安全门（" + shortPkg(pkg) + "）：入口窗禁返回键（防退出应用），只走点击梯子");
   698→                return false;
   699→            }
   700→            if (triesOf(pkg) >= BACK_RETRY_MAX || !backAllowed(now)) return false;
   701→            performGlobalAction(GLOBAL_ACTION_BACK);
   702→            energyBacks++;
   703→            backTries.put(pkg, triesOf(pkg) + 1);
   704→            adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   705→            if (d.reason.startsWith("sdk_window")) learnClass(pkg, winClassOf(pkg));
   706→            if ("click_verify_fail".equals(d.reason)) {
   707→                appendLog("🛡 点击复验失败升级（" + shortPkg(pkg) + "）：同窗 " + ctx.clickAttempts
   708→                        + " 次点击仍未关闭 → 返回键兜底温和关闭（不碰进程与服务）");
   709→            } else {
   710→                appendLog("🛡 广告窗直断（" + shortPkg(pkg) + "）：" + reasonZh(d.reason)
   711→                        + " · " + ctx.elapsedMs + "ms → 返回键温和关闭（不碰进程与服务）");
   712→            }
   713→            bumpStat(pkg, "back");
   714→            /* v3.7.0（issue#10 微博根因）：返回键直断后【不熔断】——广告 Activity
   715→             * 常拦截返回键（按了没反应）。600ms 后复验：同包窗口仍在则重开扫描链
   716→             * 继续点击梯子（角落候选/坐标轮换），不再「按完就睡」 */
   717→            handler.postDelayed(() -> {
   718→                if (instance == null) return;
   719→                try {
   720→                    AccessibilityNodeInfo[] roots = collectRoots(pkg);
   721→                    if (roots.length > 0) {
   722→                        appendLog("🔁 返回键复验（" + shortPkg(pkg) + "）：广告窗仍在（返回键被拦截）→ 继续点击梯子");
   723→                        bumpScanChain(pkg);
   724→                    }
   725→                } catch (Throwable ignored) {
   726→                }
   727→            }, 600);
   728→            return true;
   729→        }
   730→        if (d.action.equals(AdDecisionCore.A_CLICK_BOUNDS) || d.action.equals(AdDecisionCore.A_CLOSE)) {
   731→            if (!clickCoolOK) return false;
   732→            int attempts = triesOfClick(pkg);
   733→            // v3.5.0 复验重试：第 1 次 performAction→手势降级；第 ≥2 次一律手势真实触点
   734→            if (clickNode(d, attempts >= 1)) {
   735→                lastClickAt.put(pkg, now);
   736→                lastAutoClickPkg = pkg;
   737→                lastAutoClickAt = now;
   738→                clickTries.put(pkg, attempts + 1);
   739→                if (d.action.equals(AdDecisionCore.A_CLOSE)) {
   740→                    popupBlocks++;
   741→                    lastPopupAt.put(pkg, now);
   742→                    appendLog("🪟 弹窗广告已关闭（" + shortPkg(pkg) + "）："
   743→                            + (f.adBadge ? "「" + f.badgeLabel + "」角标" : "Dialog+营销话术")
   744→                            + " + 关闭键「" + f.closeLabel + "」");
   745→                    bumpStat(pkg, "popup");
   746→                } else {
   747→                    if (attempts == 0) {
   748→                        appendLog("⚡ 已点击跳过（" + shortPkg(pkg) + "）：" + reasonZh(d.reason)
   749→                                + "（" + (lastGesture ? "手势点击" : "无障碍代点") + "）→ 1.2s 后复验，未关将换候选重试");
   750→                    } else {
   751→                        appendLog("🔁 复验未关闭 → 第 " + (attempts + 1) + " 次出手（" + shortPkg(pkg) + "）："
   752→                                + reasonZh(d.reason) + "（手势真实触点·换第 " + (attempts + 1) + " 个候选，不再盲点同一坐标）");
   753→                    }
   754→                    bumpStat(pkg, "auto");
   755→                    adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   756→                }
   757→                // v3.5.0 关键：点击后【不熔断】扫描链——由后续扫描复验；
   758→                // 广告关掉会触发新窗口事件（计数清零）；仍在则换候选/升级兜底
   759→                return true;
   760→            }

   841→     *   ③ 坐标边界钳制——越界坐标（学习位置/旧指纹）直接被系统丢弃；
   842→     *   ④ 结果回调——被系统取消不再静默（诊断日志供下轮迭代）。

   847→        try {
   848→            if (x <= 0 || y <= 0) return false;
   849→            if (screenWidth > 0 && x >= screenWidth) x = screenWidth - 2; // 边界钳制
   850→            if (screenHeight > 0 && y >= screenHeight) y = screenHeight - 2;

   853→            p.lineTo(x + 1, y); // 非零长度路径：单点路径部分 ROM 不触发点击
   854→            GestureDescription gd = new GestureDescription.Builder()
   855→                    .addStroke(new GestureDescription.StrokeDescription(p, 40, 60))
   856→                    .build();
   857→            return dispatchGesture(gd, new TapGestureCallback(lastGesturePkg), null);
   858→        } catch (Throwable t) {
   859→            return false;
   860→        }
   861→    }

  1252→            JSONObject rules = loadState().optJSONObject("rules");
  1253→            if (rules == null) return new int[]{-1, -1};
  1254→            JSONArray list = rules.optJSONArray(pkg);
  1255→            if (list == null) return new int[]{-1, -1};
  1256→            JSONObject best = null;
  1257→            long bestTs = -1;
  1258→            for (int i = 0; i < list.length(); i++) {
  1259→                JSONObject r = list.optJSONObject(i);
  1260→                if (r == null || !r.optBoolean("pos", false)) continue;
  1261→                if (r.optInt("manual", 0) < 1) continue;
  1262→                if (!cls.equals(r.optString("cls", ""))) continue;
  1263→                long ts = r.optLong("ts", 0);
  1264→                if (ts > bestTs) {
  1265→                    bestTs = ts;
  1266→                    best = r;
  1267→                }
  1268→            }
  1269→            if (best == null) return new int[]{-1, -1};
  1270→            int x = (int) Math.round(best.optDouble("x", -1) * screenWidth);
  1271→            int y = (int) Math.round(best.optDouble("y", -1) * screenHeight);
  1272→            if (x <= 0 || y <= 0 || x >= screenWidth || y >= screenHeight) return new int[]{-1, -1};
  1273→            return new int[]{x, y};
  1274→        } catch (Exception ignored) {
  1275→            return new int[]{-1, -1};
  1276→        }
  1277→    }
