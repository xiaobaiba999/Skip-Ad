   424→        try {
   425→            android.app.AlarmManager am = (android.app.AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
   426→            Intent it = new Intent(ctx, KeepAliveReceiver.class);
   427→            it.setAction(KeepAliveReceiver.ACTION_SELF_CHECK);

   429→                    ctx, 8, it,

   435→                try {
   436→                    am.setExactAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);
   437→                } catch (SecurityException se) {
   438→                    am.setAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);

   440→            }

   442→        }

   496→    private int purgePoisonFingerprints() {
   497→        synchronized (lock) {
   498→            try {
   499→                JSONObject st = loadState();
   500→                JSONObject rules = st.optJSONObject("rules");
   501→                if (rules == null) return 0;
   502→                int purged = 0;
   503→                java.util.Iterator<String> it = rules.keys();
   504→                while (it.hasNext()) {
   505→                    String pkg = it.next();
   506→                    JSONArray list = rules.optJSONArray(pkg);
   507→                    if (list == null) continue;
   508→                    JSONArray kept = new JSONArray();
   509→                    for (int i = 0; i < list.length(); i++) {
   510→                        JSONObject r = list.optJSONObject(i);
   511→                        if (r == null) continue;
   512→                        String cls = r.optString("cls", "");
   513→                        boolean poison = cls.length() > 0
   514→                                && !r.optBoolean("pos", false)
   515→                                && !AdDecisionCore.isAdActivityClass(cls);
   516→                        if (poison) {
   517→                            purged++;
   518→                        } else {
   519→                            kept.put(r);
   520→                        }
   521→                    }
   522→                    if (kept.length() == 0) it.remove();
   523→                    else rules.put(pkg, kept);
   524→                }
   525→                if (purged > 0) saveState(st);
   526→                return purged;
   527→            } catch (Exception e) {
   528→                return 0;
   529→            }
   530→        }

   622→        CharSequence pkgCs = event.getPackageName();

   624→        String pkg = pkgCs.toString();

   626→        // 直接丢弃——绝不扫描（桌面简单树被误判广告窗）、绝不学习（桌面图标

   628→        // 微信」被当广告跳转踢出用户）。仅更新前台包名追踪。

   646→                    winClass.put(pkg, mc);

   654→        if (AdDecisionCore.isCameraPkg(pkg) || AdDecisionCore.isCameraClass(cs(event.getClassName()))

   660→        // 任何广告（没有可跳对象），此时唯一可能出现在前台的是来电/闹钟等敏感窗

   662→        // 前台包名追踪（亮屏后进入应用的入口窗判定仍正确），其余事件全部丢弃。
   663→        if (!isScreenOn(this)) {

   665→                lastFgPkg = pkg;
   666→            }
   667→            return;

   670→        // v3.13.0 验证码横幅防护①：短信/彩信/来电类覆盖层包（mms/smsextra/
   671→        // messaging/incallui）事件入口直接丢弃——绝不扫描（横幅上的小 × 绝不被
   672→        // 当「角落跳过键」点击）、绝不学习（手动点「复制验证码」绝不被学成跳过键）
   673→        if (AdDecisionCore.isSmsOverlayPkg(pkg)) return;

   705→                    //（防止后续窗口还拿着旧资格按返回键）
   706→                    winClsLearned.remove(pkg);
   707→                }
   708→                windowAt.put(pkg, now);
   709→                winClass.put(pkg, cls);
   710→                if (entering) rootWinCls.put(pkg, cls); // v3.6.0：记录入口窗
   711→                backTries.put(pkg, 0);
   712→                clickTries.put(pkg, 0); // v3.5.0：新窗重置复验计数
   713→                cdLastN.remove(pkg);   // v3.8.0：新窗重置倒计时锚点
   714→                cdAnchorAt.remove(pkg);
   715→                cdWaitUntil.remove(pkg);
   716→                gaveUp.put(pkg, Boolean.FALSE); // v3.6.0：新窗重置收手标记
   717→                winClsLearned.remove(pkg);
   718→                if (isDialogClass(cls) && !isOemUtilityPkg(pkg)) {
   719→                    popupUntil.put(pkg, now + POPUP_WINDOW_MS);
   720→                }
   721→                if (AdDecisionCore.isAdActivityClass(cls)) {
   722→                    appendLog("🎯 SDK 广告窗识别（" + shortPkg(pkg) + "）：" + cls
   723→                            + " → 700ms 内无真跳过键将温和直断");
   724→                } else if (learnedClassHit(pkg, cls)) {
   725→                    winClsLearned.put(pkg, Boolean.TRUE);
   726→                }
   727→                bumpScanChain(pkg);
   728→                break;
   729→            }
   730→            case AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED: {
   731→                // v4.1.0 前台门：非前台包的内容变化（灵岛动画/悬浮窗刷新）
   732→                // 绝不扫描、绝不突发检测——灵岛高频内容变化曾被当「内容
   733→                // 突发→覆盖层弹窗」开启弹窗识别窗 → 乱点灵岛控件
   734→                if (!pkg.equals(lastFgPkg)) {
   735→                    if (now < popupUntilGet(pkg)) nudgePopupScan(pkg, now);
   736→                    break;
   737→                }
   738→                if (now - winAt(pkg) < WINDOW_MS) {
   739→                    nudgeScan(pkg, now);
   740→                    break;
   741→                }
   742→                if (now < popupUntilGet(pkg)) {
   743→                    nudgePopupScan(pkg, now);
   744→                    break;
   745→                }
   746→                detectPopupBurst(pkg, now);
   747→                break;
   748→            }
   749→            case AccessibilityEvent.TYPE_VIEW_CLICKED: {
   750→                if (isOwnAutoClick(pkg, now)) return;
   751→                learnFromManualClick(pkg, event, now);
   752→                observeMisclick(pkg, event, now);
   753→                break;
   754→            }
   755→        }
   756→    }

   801→    /** v4.1.0：pkg 是否只存在「小窗」形态的窗口（灵动岛/悬浮窗：面积 <45%
   802→     *  屏或高度 <55% 屏）。灵岛小窗控件被当「角落跳过键」= 乱点元凶——

   810→            for (AccessibilityWindowInfo w : wins) {
   811→                if (w == null) continue;

   818→                double area = (b.width() * (double) b.height()) / (double) (screenWidth * screenHeight);

   820→                return false; // 存在全屏尺寸窗口 = 真实 Activity，非灵岛

   939→    /* ───────── 扫描调度：0ms 首扫 + 前密后疏重试链 ───────── */

   941→    private void bumpScanChain(final String pkg) {
   942→        final long winStart = winAt(pkg);
   943→        Integer g = scanGen.get(pkg);
   944→        final int gen = g == null ? 1 : g + 1;
   945→        scanGen.put(pkg, gen);
   946→        for (int i = 0; i < SCAN_AT.length; i++) {
   947→            final long at = SCAN_AT[i];
   948→            handler.postDelayed(() -> {
   949→                Integer cur = scanGen.get(pkg);
   950→                if (cur == null || cur.intValue() != gen) return;
   951→                tryScan(pkg, winStart, at);
   952→            }, at);
   953→        }
   954→    }
   955→
   956→    private void nudgeScan(final String pkg, long now) {
   957→        Integer gen = scanGen.get(pkg);
   958→        if (gen == null) {
   959→            bumpScanChain(pkg);
   960→            return;
   961→        }
   962→        final long winStart = winAt(pkg);
   963→        // v3.14.0：节流 150→50ms（前 4s 内）——开屏「跳过」键晚渲染时更快补扫，
   964→        // 露头就秒跳；4s 后放宽 300ms 控能耗
   965→        if (now - winStart < (now - winStart < 4000 ? 50 : 300)) return;
   966→        final int g = gen.intValue();
   967→        handler.post(() -> {
   968→            Integer cur = scanGen.get(pkg);
   969→            if (cur == null || cur.intValue() != g) return;
   970→            tryScan(pkg, winStart, -1);
   971→        });

  1174→    /* ───────── v3.9.0 倒计时门控（自 tryScan 抽取，规则/兜底两路共用）─────────

  1178→    private boolean countdownWait(String pkg, int cdN, int clickAttempts, long now) {
  1179→        if (cdN > 0) {

  1182→                cdLastN.put(pkg, cdN);

  1184→            }
  1185→            if (clickAttempts >= 1) {
  1186→                Long anchor = cdAnchorAt.get(pkg);
  1187→                long until = (anchor == null ? now : anchor.longValue())
  1188→                        + cdN * 1000L - AdDecisionCore.COUNTDOWN_EXTRA_MS;
  1189→                if (now < until) {
  1190→                    if (cdWaitUntil.put(pkg, until) == null) {
  1191→                        appendLog("⏳ 倒计时门控（" + shortPkg(pkg) + "）：跳过键 "
  1192→                                + cdN + "s 后才激活（提前点=白点）→ "
  1193→                                + (until - now) + "ms 后出手即命中");
  1194→                    }
  1195→                    return true;
  1196→                }
  1197→            }
  1198→            return false;
  1199→        }
  1200→        Long w = cdWaitUntil.get(pkg);
  1201→        if (w != null) {
  1202→            cdWaitUntil.remove(pkg);
  1203→            appendLog("⏳ 倒计时结束（" + shortPkg(pkg) + "）：跳过键已激活 → 出手");
  1204→        }
  1205→        return false;
  1206→    }

  1260→            // 入口窗上按返回键 = 退出应用（喜马拉雅「再按一次退出到桌面」事故）

  1269→            if (d.reason.startsWith("sdk_window")) learnClass(pkg, winClassOf(pkg));
  1270→            if ("click_verify_fail".equals(d.reason)) {

  1272→                        + " 次点击仍未关闭 → 返回键兜底温和关闭（不碰进程与服务）");

  1276→            }
  1277→            bumpStat(pkg, "back");
  1278→            /* v3.7.0（issue#10 微博根因）：返回键直断后【不熔断】——广告 Activity
  1279→             * 常拦截返回键（按了没反应）。600ms 后复验：同包窗口仍在则重开扫描链
  1280→             * 继续点击梯子（角落候选/坐标轮换），不再「按完就睡」 */
  1281→            handler.postDelayed(() -> {
  1282→                if (instance == null) return;
  1283→                try {
  1284→                    AccessibilityNodeInfo[] roots = collectRoots(pkg);
  1285→                    if (roots.length > 0) {
  1286→                        appendLog("🔁 返回键复验（" + shortPkg(pkg) + "）：广告窗仍在（返回键被拦截）→ 继续点击梯子");
  1287→                        bumpScanChain(pkg);
  1288→                    }
  1289→                } catch (Throwable ignored) {
  1290→                }
  1291→            }, 600);
  1292→            return true;
  1293→        }
  1294→        if (d.action.equals(AdDecisionCore.A_CLICK_BOUNDS) || d.action.equals(AdDecisionCore.A_CLOSE)) {
  1295→            if (!clickCoolOK) return false;

  1373→        }
  1374→        if (d.node instanceof NodeAdapter) {
  1375→            try {
  1376→                if (((NodeAdapter) d.node).n.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
  1377→                    return true;
  1378→                }
  1379→            } catch (Exception ignored) {
  1380→            }

  1412→    /** dispatchGesture 坐标点击（API 24+；真实触点，performAction 拦不住）。
  1413→     *  v3.7.0 根因修复（issue#10：「已点击跳过×4 无效果」）：
  1414→     *   ① 路径非零长度——单点路径（只有 moveTo）在部分 ROM 上不触发点击
  1415→     *    （业界验证：需 lineTo 形成真实触点轨迹）；
  1416→     *   ② 起始延迟 40ms——给前一帧渲染/事件处理留时间，立即派发易被丢弃；
  1417→     *   ③ 坐标边界钳制——越界坐标（学习位置/旧指纹）直接被系统丢弃；
  1418→     *   ④ 结果回调——被系统取消不再静默（诊断日志供下轮迭代）。
  1419→     *   ⑤ 前提：accessibility_service_config 已补 canPerformGestures=true
  1420→     *    （v3.6.0 及之前【缺失】——手势从未真正触达屏幕，喜马拉雅复验重试
  1421→     *    全部无效的头号根因）
  1422→     *   ⑥ 触点时长 60→110ms——更接近真人触点（80-150ms），防 SDK 触点时长校验 */
  1423→    private boolean gestureTap(int x, int y) {
  1424→        try {
  1425→            // v3.13.0 息屏熔断（纵深防御）：息屏时绝无可见广告，真实触点只会
  1426→            // 落在来电/闹钟等敏感窗上（误点=挂断电话/关闭闹钟）
  1427→            if (!isScreenOn(this)) return false;
  1428→            if (x <= 0 || y <= 0) return false;
  1429→            if (screenWidth > 0 && x >= screenWidth) x = screenWidth - 2; // 边界钳制
  1430→            if (screenHeight > 0 && y >= screenHeight) y = screenHeight - 2;
  1431→            // v3.13.0 验证码横幅防护③：触点被 SystemUI 顶部小窗（状态栏/
  1432→            // heads-up 横幅）覆盖时拒发手势——真实触点会被横幅截走（点掉
  1433→            // 验证码横幅的事故路径）。拒发后 clickNode 自动降级 performAction
  1434→            // 代点（走无障碍通道不经屏幕坐标，绝不碰横幅），学习位置点择机重试
  1435→            if (pointUnderHeadsUp(x, y)) {
  1436→                headsUpGuardLog(lastGesturePkg, System.currentTimeMillis());
  1437→                return false;
  1438→            }
  1439→            Path p = new Path();
  1440→            p.moveTo(x, y);
  1441→            p.lineTo(x + 1, y); // 非零长度路径：单点路径部分 ROM 不触发点击
  1442→            GestureDescription gd = new GestureDescription.Builder()
  1443→                    .addStroke(new GestureDescription.StrokeDescription(p, 40, 110))
  1444→                    .build();
  1445→            return dispatchGesture(gd, new TapGestureCallback(lastGesturePkg), null);
  1446→        } catch (Throwable t) {
  1447→            return false;
  1448→        }
  1449→    }

  1639→    private void observeMisclick(String pkg, AccessibilityEvent event, long now) {
  1640→        try {

  1646→            if (label.length() == 0) label = cs(src.getContentDescription());

  1648→            Rect r = new Rect();
  1649→            src.getBoundsInScreen(r);
  1650→            if (r.isEmpty()) return;

  1653→            boolean huge = (area / screen > 0.35) && (r.width() > screenWidth * 0.55);
  1654→            // v3.13.0：「快于人类反应」启发式【移除】——真机日志（issue#16/#17）
