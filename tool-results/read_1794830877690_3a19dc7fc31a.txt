   526→function cmpVer(a, b) {
   527→  const pa = String(a).replace(/^v/, "").split("."), pb = String(b).replace(/^v/, "").split(".");
   528→  for (let i = 0; i < 3; i++) {
   529→    const d = (parseInt(pa[i], 10) || 0) - (parseInt(pb[i], 10) || 0);
   530→    if (d) return d > 0 ? 1 : -1;
   531→  }
   532→  return 0;
   533→}

   535→async function fetchLatest() {
   536→  if (HAS_BRIDGE) {
   537→    // 原生侧已带令牌 + 两重重试 + /releases 列表回退（v3.3.0）
   538→    try { return JSON.parse(AndroidApp.netGetLatestRelease()); } catch (e) { return { ok: false, error: "解析失败" }; }
   539→  }
   540→  try {
   541→    const r = await fetch(GH_API + "/releases/latest", { headers: { Accept: "application/vnd.github+json" } });
   542→    if (r.status === 404) return { ok: true, hasRelease: false };
   543→    if (r.ok) {
   544→      const j = await r.json();
   545→      const apk = (j.assets || []).filter(a => a && a.name && a.name.endsWith(".apk"))[0];
   546→      return {
   547→        ok: true, hasRelease: true, tag: j.tag_name, notes: j.body, url: j.html_url,
   548→        apkUrl: apk ? apk.browser_download_url : "",
   549→        apkMirrors: apk ? ["https://ghfast.top/" + apk.browser_download_url, "https://gh-proxy.com/" + apk.browser_download_url] : [],
   550→        apkFileName: apk ? apk.name : "adshield.apk",
   551→        apkMB: apk ? Math.round(apk.size / 1048576 * 100) / 100 : 0,
   552→      };
   553→    }
   554→    return { ok: false, error: "GitHub 返回 " + r.status };
   555→  } catch (e) { return { ok: false, error: "网络异常：" + (e.message || e) }; }
   556→}

   560→/* ── 统一页内下载流程（横幅与关于页共用；多通道自动切换） ── */
   561→function startPageDownload(d, prefix) {
   562→  const fileName = d.apkFileName || "adshield.apk";
   563→  if (!d.apkUrl) { toast("该版本没有 APK 附件"); return; }
   564→  if (!HAS_BRIDGE) { openLink(d.apkUrl); return; } // 浏览器预览：退化为直链
   565→  dl = {
   566→    channels: [d.apkUrl].concat(d.apkMirrors || []),
   567→    idx: 0, fileName, prefix,
   568→    lastBytes: -1, lastMoveAt: Date.now(),
   569→  };
   570→  beginChannel();
   571→}

   573→function beginChannel() {
   574→  const url = dl.channels[dl.idx];
   575→  try {
   576→    const r = JSON.parse(AndroidApp.startUpdateDownload(url, dl.fileName));
   577→    if (r.ok) {
   578→      const btn = $("#" + dl.prefix + "-btn");
   579→      if (btn) { btn.textContent = "下载中…"; btn.disabled = true; }
   580→      const wrap = $("#" + dl.prefix + "-progress-wrap");
   581→      if (wrap) wrap.style.display = "";
   582→      dl.lastBytes = -1;
   583→      dl.lastMoveAt = Date.now();
   584→      startProgressPoll(dl.prefix);
   585→    } else toast(r.error || "启动下载失败");
   586→  } catch (e) { toast("启动下载失败"); }
   587→}

   589→/* 切换下一个下载通道；无通道可用时返回 false */
   590→function switchChannel(reason) {
   591→  if (dl.idx + 1 < dl.channels.length) {
   592→    dl.idx++;
   593→    toast(reason + "，已切换备用通道（" + dl.idx + "/" + (dl.channels.length - 1) + "）");
   594→    beginChannel();
   595→    return true;
   596→  }
   597→  return false;
   598→}

   600→function startProgressPoll(prefix) {
   601→  if (ubTimer) clearInterval(ubTimer);
   602→  ubTimer = setInterval(() => {
   603→    try {
   604→      const d = JSON.parse(AndroidApp.updateProgress());
   605→      const bar = $("#" + prefix + "-bar"), txt = $("#" + prefix + "-text");
   606→      if (d.state === "running" || d.state === "paused") {
   607→        bar.style.width = d.percent + "%";
   608→        txt.textContent = d.percent + "% · " + (d.bytes / 1048576).toFixed(1) + " / " + (d.total / 1048576).toFixed(1) + " MB";
   609→        // 停滞检测：30s 无新字节 → 判定该通道不可用，自动切换
   610→        if (d.bytes !== dl.lastBytes) { dl.lastBytes = d.bytes; dl.lastMoveAt = Date.now(); }
   611→        else if (Date.now() - dl.lastMoveAt > 30000) {
   612→          if (switchChannel("下载长时间无进展")) return;
   613→          dl.lastMoveAt = Date.now(); // 无通道可切：再等一轮
   614→        }
   615→      } else if (d.state === "done") {
   616→        clearInterval(ubTimer); ubTimer = null;
   617→        bar.style.width = "100%";
   618→        txt.textContent = "下载完成 ✓ 点击安装即覆盖升级（学习数据保留）";
   619→        const btn = $("#" + prefix + "-btn");
   620→        btn.textContent = "立即安装";
   621→        btn.disabled = false;
   622→        btn.dataset.phase = "install";
   623→      } else if (d.state === "failed") {
   624→        clearInterval(ubTimer); ubTimer = null;
   625→        if (switchChannel("直连下载失败")) return;
   626→        txt.textContent = "下载失败（直连与备用通道均不可用），可稍后重试";
   627→        const btn = $("#" + prefix + "-btn");
   628→        btn.textContent = "重新下载";
   629→        btn.disabled = false;
   630→        btn.dataset.phase = "download";
   631→      }
   632→    } catch (e) { clearInterval(ubTimer); ubTimer = null; }
   633→  }, 800);
   634→}
