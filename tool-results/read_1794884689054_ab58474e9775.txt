   355→    /** v3.8.0：排下一次心跳（静态：KeepAliveReceiver 收到自检后自续期共用——
   356→     *  心跳链永不中断）。亮屏：精确闹钟优先（Doze 中也能准点触发），未授予
   357→     *  精确闹钟权限（Android 12+）降级 setAndAllowWhileIdle；息屏且允许降频
   358→     *  （nightDeferrable，开机 2s 快查除外）：改窗口式非精确闹钟——系统批量
   359→     *  调度，不唤醒设备（v3.13.0 待机功耗治理） */
   360→    static void scheduleNextHeartbeat(Context ctx, long atMs) {
   361→        scheduleNextHeartbeat(ctx, atMs, false);
   362→    }

   364→    static void scheduleNextHeartbeat(Context ctx, long atMs, boolean nightDeferrable) {
   365→        try {
   366→            android.app.AlarmManager am = (android.app.AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
   367→            Intent it = new Intent(ctx, KeepAliveReceiver.class);
   368→            it.setAction(KeepAliveReceiver.ACTION_SELF_CHECK);
   369→            android.app.PendingIntent pi = android.app.PendingIntent.getBroadcast(
   370→                    ctx, 8, it,
   371→                    android.app.PendingIntent.FLAG_UPDATE_CURRENT | android.app.PendingIntent.FLAG_IMMUTABLE);
   372→            if (nightDeferrable && !isScreenOn(ctx)) {
   373→                am.setWindow(android.app.AlarmManager.RTC,
   374→                        System.currentTimeMillis() + NIGHT_HEARTBEAT_MS, NIGHT_HEARTBEAT_WINDOW_MS, pi);
   375→            } else {
   376→                try {
   377→                    am.setExactAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);
   378→                } catch (SecurityException se) {
   379→                    am.setAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);
   380→                }
   381→            }
   382→        } catch (Throwable ignored) {
   383→        }
   384→    }

   583→        // 前台包名追踪（亮屏后进入应用的入口窗判定仍正确），其余事件全部丢弃。
   584→        if (!isScreenOn(this)) {
   585→            if (event.getEventType() == AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) {
   586→                lastFgPkg = pkg;
   587→            }
   588→            return;
   589→        }
   590→
   591→        // v3.13.0 验证码横幅防护①：短信/彩信/来电类覆盖层包（mms/smsextra/
   592→        // messaging/incallui）事件入口直接丢弃——绝不扫描（横幅上的小 × 绝不被
   593→        // 当「角落跳过键」点击）、绝不学习（手动点「复制验证码」绝不被学成跳过键）
   594→        if (AdDecisionCore.isSmsOverlayPkg(pkg)) return;
   595→
   596→        energyEvents++;
   597→
   598→        switch (event.getEventType()) {
   599→            case AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED: {
   600→                String cls = cs(event.getClassName());
   601→                // v3.6.0 根窗口追踪：新进入前台的应用，其首个窗口 = 入口窗
   602→                //（冷启动开屏 / 从桌面·其他应用回前台的首窗）——入口窗上
   603→                // 按返回键 = 退出应用，安全门据此禁一切返回键兜底
   604→                boolean entering = !pkg.equals(lastFgPkg);
   605→                lastFgPkg = pkg;
   606→                // v3.5.0 复验通过：上一窗点过东西且窗口已切走 → 广告已关
   607→                Integer prevTries = clickTries.get(pkg);
   608→                if (prevTries != null && prevTries > 0 && !cls.equals(winClassOf(pkg))) {
   609→                    appendLog("✓ 复验通过（" + shortPkg(pkg) + "）：广告窗已关闭"
   610→                            + (prevTries > 1 ? "（共 " + prevTries + " 次尝试，重试梯子兜底）" : ""));
   611→                    // v3.6.0 毒链① 根治：广告已关，学习窗快断资格同步作废
   612→                    //（防止后续窗口还拿着旧资格按返回键）
   613→                    winClsLearned.remove(pkg);
   614→                }
   615→                windowAt.put(pkg, now);
   616→                winClass.put(pkg, cls);
   617→                if (entering) rootWinCls.put(pkg, cls); // v3.6.0：记录入口窗
   618→                backTries.put(pkg, 0);
   619→                clickTries.put(pkg, 0); // v3.5.0：新窗重置复验计数
   620→                cdLastN.remove(pkg);   // v3.8.0：新窗重置倒计时锚点
   621→                cdAnchorAt.remove(pkg);
   622→                cdWaitUntil.remove(pkg);
   623→                gaveUp.put(pkg, Boolean.FALSE); // v3.6.0：新窗重置收手标记
   624→                winClsLearned.remove(pkg);
   625→                if (isDialogClass(cls) && !isOemUtilityPkg(pkg)) {
   626→                    popupUntil.put(pkg, now + POPUP_WINDOW_MS);
   627→                }
   628→                if (AdDecisionCore.isAdActivityClass(cls)) {
   629→                    appendLog("🎯 SDK 广告窗识别（" + shortPkg(pkg) + "）：" + cls
   630→                            + " → 700ms 内无真跳过键将温和直断");
   631→                } else if (learnedClassHit(pkg, cls)) {
   632→                    winClsLearned.put(pkg, Boolean.TRUE);
   633→                }
   634→                bumpScanChain(pkg);
   635→                break;
   636→            }
   637→            case AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED: {
   638→                if (now - winAt(pkg) < WINDOW_MS) {
   639→                    nudgeScan(pkg, now);
   640→                    break;
   641→                }
   642→                if (now < popupUntilGet(pkg)) {
   643→                    nudgePopupScan(pkg, now);
   644→                    break;
   645→                }
   646→                detectPopupBurst(pkg, now);
   647→                break;
   648→            }
   649→            case AccessibilityEvent.TYPE_VIEW_CLICKED: {
   650→                if (isOwnAutoClick(pkg, now)) return;
   651→                learnFromManualClick(pkg, event, now);
   652→                observeMisclick(pkg, event, now);
   653→                break;
   654→            }
   655→        }
   656→    }

   800→    /* ───────── 扫描调度：0ms 首扫 + 前密后疏重试链 ───────── */
   801→
   802→    private void bumpScanChain(final String pkg) {
   803→        final long winStart = winAt(pkg);
   804→        Integer g = scanGen.get(pkg);
   805→        final int gen = g == null ? 1 : g + 1;
   806→        scanGen.put(pkg, gen);
   807→        for (int i = 0; i < SCAN_AT.length; i++) {
   808→            final long at = SCAN_AT[i];
   809→            handler.postDelayed(() -> {
   810→                Integer cur = scanGen.get(pkg);
   811→                if (cur == null || cur.intValue() != gen) return;
   812→                tryScan(pkg, winStart, at);
   813→            }, at);
   814→        }
   815→    }
   816→
   817→    private void nudgeScan(final String pkg, long now) {
   818→        Integer gen = scanGen.get(pkg);
   819→        if (gen == null) {
   820→            bumpScanChain(pkg);
   821→            return;
   822→        }
   823→        final long winStart = winAt(pkg);
   824→        // v3.9.0：节流 150→80ms（前 4s 内）——开屏「跳过」键晚渲染时更快补扫，
   825→        // 露头就秒跳；4s 后放宽 300ms 控能耗
   826→        if (now - winStart < (now - winStart < 4000 ? 80 : 300)) return;
   827→        final int g = gen.intValue();
   828→        handler.post(() -> {
   829→            Integer cur = scanGen.get(pkg);
   830→            if (cur == null || cur.intValue() != g) return;
   831→            tryScan(pkg, winStart, -1);
   832→        });
   833→    }
   834→
   835→    private void nudgePopupScan(final String pkg, long now) {
   836→        Long last = lastPopupScanAt.get(pkg);
   837→        if (last != null && now - last < POPUP_SCAN_THROTTLE_MS) return;
   838→        lastPopupScanAt.put(pkg, now);
   839→        final long winStart = winAt(pkg);
   840→        handler.post(() -> {
   841→            if (instance == null) return;
   842→            tryScan(pkg, winStart, -2);
   843→        });
   844→    }
   845→
   846→    private void detectPopupBurst(String pkg, long now) {
   847→        if (!popupShieldEnabled() || isOemUtilityPkg(pkg)) return;
   848→        long[] w = burstWin.get(pkg);
   849→        if (w == null || now - w[0] > BURST_WIN_MS) {
   850→            burstWin.put(pkg, new long[]{now, 1});
   851→            return;
   852→        }
   853→        w[1]++;
   854→        if (w[1] >= BURST_COUNT) {
   855→            burstWin.remove(pkg);
   856→            popupUntil.put(pkg, now + POPUP_WINDOW_MS);
   857→            appendLog("🪟 检测到内容突发（" + shortPkg(pkg) + "）：疑似覆盖层弹窗 → 开启弹窗识别窗");
   858→            nudgePopupScan(pkg, now);
   859→        }
   860→    }

   863→        Integer g = scanGen.get(pkg);
   864→        scanGen.put(pkg, g == null ? 1 : g + 1);
   865→    }

   867→    /* ───────── 扫描与决策执行（逻辑全部在 AdDecisionCore） ───────── */
   868→
   869→    private void tryScan(String pkg, long winStart, long atMs) {
   870→        if (instance == null) return;
   871→        long now = System.currentTimeMillis();
   872→        long elapsed = now - winStart;
   873→        boolean splashLive = learningEnabled() && elapsed <= WINDOW_MS;
   874→        boolean popupLive = popupShieldEnabled() && now < popupUntilGet(pkg) && popupCooldownOK(pkg, now);
   875→        if (!splashLive && !popupLive) return;
   876→
   877→        try {
   878→            /* 多窗口扫描：活动窗口优先，其次 getWindows() 同包窗口（弹窗/覆盖层） */
   879→            AccessibilityNodeInfo[] roots = collectRoots(pkg);
   880→            if (roots.length == 0) return;
   881→
   882→            Long lastClick = lastClickAt.get(pkg);
   883→            boolean clickCoolOK = lastClick == null || now - lastClick >= COOLDOWN_MS;
   884→            // v3.5.0：学习位置指纹（手动教过的坐标）与复验次数
   885→            int[] tapHint = learnedPos(pkg);
   886→            Integer triesBox = clickTries.get(pkg);
   887→            int clickAttempts = triesBox == null ? 0 : triesBox.intValue();
   888→
   889→            for (int ri = 0; ri < roots.length; ri++) {
   890→                // v3.13.0 验证码横幅防护②：树内任一节点带「验证码/校验码/
   891→                // 动态码/动态密码」文案（SystemUI 横幅窗/应用内自绘提示条/
   892→                // 银行登录页）——本窗口绝不扫描、绝不点击、绝不学习。
   893→                // 预算 60：横幅窗树极浅，零性能顾虑
   894→                if (AdDecisionCore.treeHasCodeBannerMarker(new NodeAdapter(roots[ri]), 60)) {
   895→                    codeBannerGuardLog(pkg, now);
   896→                    continue;
   897→                }
   898→                energyScans++;
   899→                Ctx ctx = new Ctx();
   900→                ctx.screenW = screenWidth;
   901→                ctx.screenH = screenHeight;
   902→                ctx.pkg = pkg; // v3.10.0：分应用规则表
   903→                ctx.elapsedMs = Math.max(0, elapsed - ri * 30);
   904→                ctx.splashLive = splashLive;
   905→                ctx.popupLive = popupLive;
   906→                ctx.guard = AdDecisionCore.isGuardPkg(pkg);
   907→                // v3.6.0 毒链① 根治：sdkAdWin 只认【当前窗口类名】是 SDK 广告窗
   908→                //（v3.5.0 的 12s TTL 让主页还拿着 SDK 资格按返回键 → 退出应用）
   909→                ctx.sdkAdWin = AdDecisionCore.isAdActivityClass(winClassOf(pkg));
   910→                Boolean learned = winClsLearned.get(pkg);
   911→                ctx.learnedCls = learned != null && learned.booleanValue();
   912→                // v3.6.0 毒链② 根治：根窗口（入口窗）标记——返回键=退出应用，
   913→                // L3/L4.5/L5 决策核心禁用 + execute() 双保险拦截
   914→                ctx.rootWindow = winClassOf(pkg).equals(rootWinClsOf(pkg));
   915→                ctx.clickAttempts = clickAttempts;
   916→                ctx.tapHintX = tapHint[0];
   917→                ctx.tapHintY = tapHint[1];
   918→                ctx.budgetNodes = splashLive ? 600 : 280; // 开屏窗预算扩容（重树）
   919→
   920→                /* ═══ v3.9.0 主路径：内嵌规则引擎 0ms 出手 ═══
   921→                 * 规则只认控件属性（text/desc/vid），点击的是控件自身 bounds 中心
   922→                 * ——应用按实际屏幕渲染，适配任意分辨率，绝无固定坐标盲点。
   923→                 * Keep 开屏「跳过」文本键：窗口事件 0ms 首扫即中即点=露头秒跳。
   924→                 * v3.10.0：五层规则（分应用表/全局精确/倒计时形态/短包含+诱导否决/
   925→                 * vid skip 族+SDK 窗 close 族） */
   926→                if (splashLive) {
   927→                    Hit hit = RuleEngine.match(new NodeAdapter(roots[ri]), ctx);
   928→                    if (hit != null) {
   929→                        int cdHit = AdDecisionCore.parseCountdown(hit.label);
   930→                        if (!countdownWait(pkg, cdHit, clickAttempts, now)) {
   931→                            Decision rd = new Decision();
   932→                            rd.action = AdDecisionCore.A_CLICK_BOUNDS;
   933→                            rd.node = hit.gestureOnly ? null : hit.clickNode;
   934→                            rd.bounds = hit.tapBounds;
   935→                            rd.reason = "rule_" + hit.appId;
   936→                            if (execute(pkg, rd, null, ctx, clickCoolOK, now)) return; // 命中即熔断
   937→                        }
   938→                        // 规则已命中（或倒计时等待中）——本窗就是标准跳过场景，
   939→                        // 不再走决策核心兜底（避免角落键/热区在正常窗乱点）
   940→                        continue;
   941→                    }
   942→                }
   943→
   944→                /* ═══ 兜底路径：决策核心（复杂信号）═══ */
   945→                ScanFind f = AdDecisionCore.walk(new NodeAdapter(roots[ri]), ctx);
   946→                /* v3.8.0 倒计时门控（issue#10 二轮·喜马拉雅）：穿山甲类 SDK 倒计时
   947→                 * 结束前跳过键不响应——「已点击×4 不跳转」真凶之一（提前点全白点）。
   948→                 * v3.9.0 抽取为 countdownWait()（规则路径/兜底路径共用） */
   949→                if (countdownWait(pkg, f.countdownN, clickAttempts, now)) continue;
   950→                Decision d = AdDecisionCore.decide(f, ctx);
   951→                if (d.action.equals(AdDecisionCore.A_NONE)) continue;
   952→                if (execute(pkg, d, f, ctx, clickCoolOK, now)) return; // 命中即熔断
   953→            }
   954→        } catch (Exception ignored) {
   955→        }
   956→    }
   957→
   958→    /* ───────── v3.9.0 倒计时门控（自 tryScan 抽取，规则/兜底两路共用）─────────
   959→     * 穿山甲类 SDK 倒计时结束前跳过键不响应（提前点=白点还占冷却）。
   960→     * 文案数字变化→重新锚定；首击未关→等到「锚点+剩余秒数-0.5s」再出手；
   961→     * 倒计时走完（文案无数字）→立即恢复出手。返回 true=继续等待。 */
   962→    private boolean countdownWait(String pkg, int cdN, int clickAttempts, long now) {
   963→        if (cdN > 0) {
   964→            Integer lastN = cdLastN.get(pkg);
   965→            if (lastN == null || lastN.intValue() != cdN) {
   966→                cdLastN.put(pkg, cdN);
   967→                cdAnchorAt.put(pkg, now);
   968→            }
   969→            if (clickAttempts >= 1) {
   970→                Long anchor = cdAnchorAt.get(pkg);
   971→                long until = (anchor == null ? now : anchor.longValue())
   972→                        + cdN * 1000L - AdDecisionCore.COUNTDOWN_EXTRA_MS;
   973→                if (now < until) {
   974→                    if (cdWaitUntil.put(pkg, until) == null) {
   975→                        appendLog("⏳ 倒计时门控（" + shortPkg(pkg) + "）：跳过键 "
   976→                                + cdN + "s 后才激活（提前点=白点）→ "
   977→                                + (until - now) + "ms 后出手即命中");
   978→                    }
   979→                    return true;
   980→                }
   981→            }
   982→            return false;
   983→        }
   984→        Long w = cdWaitUntil.get(pkg);
   985→        if (w != null) {
   986→            cdWaitUntil.remove(pkg);
   987→            appendLog("⏳ 倒计时结束（" + shortPkg(pkg) + "）：跳过键已激活 → 出手");
   988→        }
   989→        return false;
   990→    }

   992→    /** 收集同包可扫描窗口根节点（活动窗口优先，最多 3 个） */
   993→    private AccessibilityNodeInfo[] collectRoots(String pkg) {
   994→        java.util.ArrayList<AccessibilityNodeInfo> out = new java.util.ArrayList<>();
   995→        AccessibilityNodeInfo active = getRootInActiveWindow();
   996→        if (active != null) {
   997→            CharSequence rp = active.getPackageName();
   998→            if (rp != null && pkg.contentEquals(rp)) out.add(active);
   999→        }
  1000→        if (out.isEmpty()) {
  1001→            try {
  1002→                List<AccessibilityWindowInfo> wins = getWindows();
  1003→                if (wins != null) {

  1201→            // v3.13.0 息屏熔断（纵深防御）：息屏时绝无可见广告，真实触点只会
  1202→            // 落在来电/闹钟等敏感窗上（误点=挂断电话/关闭闹钟）
  1203→            if (!isScreenOn(this)) return false;

  1205→            if (screenWidth > 0 && x >= screenWidth) x = screenWidth - 2; // 边界钳制
  1206→            if (screenHeight > 0 && y >= screenHeight) y = screenHeight - 2;
  1207→            // v3.13.0 验证码横幅防护③：触点被 SystemUI 顶部小窗（状态栏/

  1209→            // 验证码横幅的事故路径）。拒发后 clickNode 自动降级 performAction
  1210→            // 代点（走无障碍通道不经屏幕坐标，绝不碰横幅），学习位置点择机重试
  1211→            if (pointUnderHeadsUp(x, y)) {
  1212→                headsUpGuardLog(lastGesturePkg, System.currentTimeMillis());
  1213→                return false;
  1214→            }
  1215→            Path p = new Path();
  1216→            p.moveTo(x, y);
  1217→            p.lineTo(x + 1, y); // 非零长度路径：单点路径部分 ROM 不触发点击

  1219→                    .addStroke(new GestureDescription.StrokeDescription(p, 40, 110))
  1220→                    .build();

  1222→        } catch (Throwable t) {
  1223→            return false;
  1224→        }
  1225→    }

  1849→    /* ───────── v3.8.0 WRITE_SECURE_SETTINGS 自愈重绑（可选高级能力）─────────
  1850→     * Android 安全设计：无障碍服务受 BIND_ACCESSIBILITY_SERVICE 签名级权限保护，
  1851→     * 任何应用（含自己）都不能 startService/bindService 它——v3.6.0/v3.7.0 的
  1852→     * 「startService 拉活重绑」实为 SecurityException，自动恢复从未真正生效。
  1853→     * 系统提供的唯一正道（GKD 进阶玩法同款）：用户经 adb 一次性授权
  1854→     *   adb shell pm grant com.zai.adshield android.permission.WRITE_SECURE_SETTINGS
  1855→     * 之后本应用可直接写 Settings.Secure：
  1856→     *  · 服务被杀未解绑 → 从 ENABLED_ACCESSIBILITY_SERVICES 摘除再写回 →
  1857→     *    系统立刻重新 bind（进程随之拉起）——「清理后台重进要手动开」终结
  1858→     *  · 覆盖安装被系统关闭（MY_PACKAGE_REPLACED 必关）→ 直接写回开启状态 */
  1859→    public static boolean hasSecureWrite(Context ctx) {
  1860→        try {
  1861→            return ctx.checkPermission("android.permission.WRITE_SECURE_SETTINGS",
  1862→                    android.os.Process.myPid(), android.os.Process.myUid())
  1863→                    == android.content.pm.PackageManager.PERMISSION_GRANTED;
  1864→        } catch (Throwable t) {
  1865→            return false;
  1866→        }
  1867→    }
  1868→
  1869→    /** 自愈重绑：true=已发起（重绑异步生效）；未授权返回 false */
  1870→    public static boolean selfHealBind(Context ctx) {
  1871→        if (!hasSecureWrite(ctx)) return false;
  1872→        try {
  1873→            final android.content.ContentResolver cr = ctx.getContentResolver();
  1874→            final String key = Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES;
  1875→            final ComponentName cn = new ComponentName(ctx, AdSkipService.class);
  1876→            final String flat = cn.flattenToString();
  1877→            final String flatShort = cn.flattenToShortString(); // v3.8.1：兼容短格式条目（adb 手动写入场景）
  1878→            final String cur = Settings.Secure.getString(cr, key);
  1879→            // v3.8.1：登记判断双格式匹配——全格式或短格式条目任一命中即视为已登记，
  1880→            // 否则短格式场景会被误判「未登记」再追加产生冗余双条目
  1881→            if (cur == null || !(cur.contains(flat) || cur.contains(flatShort))) {
  1882→                // 未登记（被系统关闭/覆盖安装后清除）→ 直接写回开启
  1883→                String next = (cur == null || cur.trim().isEmpty()) ? flat : cur + ":" + flat;
  1884→                Settings.Secure.putString(cr, key, next);
  1885→                Settings.Secure.putString(cr, Settings.Secure.ACCESSIBILITY_ENABLED, "1");
  1886→                return true;
  1887→            }
  1888→            // 已登记但进程已死 → 摘除再写回，强制系统重绑（bind 会拉起进程）
  1889→            StringBuilder keep = new StringBuilder();
  1890→            for (String s : cur.split(":")) {
  1891→                String t = s.trim();
  1892→                if (t.length() == 0 || t.equals(flat) || t.equals(flatShort)) continue; // v3.8.1：双格式都摘除
  1893→                if (keep.length() > 0) keep.append(':');
  1894→                keep.append(t);
  1895→            }
  1896→            String without = keep.toString();
  1897→            Settings.Secure.putString(cr, key, without); // 摘除 → 系统解绑
  1898→            Thread.sleep(400);                            // 等解绑落地
  1899→            Settings.Secure.putString(cr, key,            // 写回 → 系统 bind（拉起进程）
  1900→                    without.isEmpty() ? flat : without + ":" + flat);
  1901→            return true;
  1902→        } catch (Throwable t) {
  1903→            return false;
  1904→        }
  1905→    }

  1908→        if (pkg == null) return "";
  1909→        int i = pkg.lastIndexOf('.');
