   238→    /** v3.7.0：心跳自检闹钟（15 分钟重复，服务连上即撤） */
   239→    public static final long HEARTBEAT_MS = 15 * 60 * 1000;
   240→
   241→    /** v3.8.0：排下一次心跳（静态：KeepAliveReceiver 收到自检后自续期共用——
   242→     *  心跳链永不中断）。精确闹钟优先（Doze 中也能准点触发），未授予精确闹钟
   243→     *  权限（Android 12+）降级 setAndAllowWhileIdle——仍远优于 setInexactRepeating
   244→     *  （可被系统无限推迟，v3.7.0 心跳常不触发的根因） */
   245→    static void scheduleNextHeartbeat(Context ctx, long atMs) {
   246→        try {
   247→            android.app.AlarmManager am = (android.app.AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
   248→            Intent it = new Intent(ctx, KeepAliveReceiver.class);
   249→            it.setAction(KeepAliveReceiver.ACTION_SELF_CHECK);
   250→            android.app.PendingIntent pi = android.app.PendingIntent.getBroadcast(
   251→                    ctx, 8, it,
   252→                    android.app.PendingIntent.FLAG_UPDATE_CURRENT | android.app.PendingIntent.FLAG_IMMUTABLE);
   253→            try {
   254→                am.setExactAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);
   255→            } catch (SecurityException se) {
   256→                am.setAndAllowWhileIdle(android.app.AlarmManager.RTC, atMs, pi);
   257→            }
   258→        } catch (Throwable ignored) {
   259→        }
   260→    }
   261→
   262→    private void scheduleHeartbeat() {
   263→        scheduleNextHeartbeat(this, System.currentTimeMillis() + HEARTBEAT_MS);
   264→    }

   279→    @Override
   280→    public void onTaskRemoved(Intent rootIntent) {
   281→        super.onTaskRemoved(rootIntent);
   282→        // v3.6.0：后台清理（滑掉最近任务）时，部分 ROM 会连无障碍服务一起杀。
   283→        // ① 若进程幸存：4s 后自检，服务真死了才发「待恢复」提醒（防误报）
   284→        // ② 自检闹钟（15s）：进程被杀但非 force-stop 的 ROM 上仍可触发提醒
   285→        // v3.7.0：追加 15 分钟心跳闹钟——进程被杀后持续自检，
   286→        //    服务一旦真死即提醒恢复（用户手动关闭 keepOn 时不排）
   287→        try {
   288→            if (!prefs().getBoolean(KEY_KEEPON, false)) return;
   289→            handler.postDelayed(() -> {
   290→                if (instance == null) KeepAliveReceiver.notifyAlert(AdSkipService.this);
   291→            }, 4000);
   292→            scheduleSelfCheckAlarm(15000);
   293→            scheduleHeartbeat();
   294→        } catch (Throwable ignored) {
   295→        }
   296→    }

   298→    /** v3.6.0：自检闹钟——清理后台后若服务已死（且非 force-stop 状态），发恢复提醒 */
   299→    private void scheduleSelfCheckAlarm(long delayMs) {

   361→    @Override
   362→    public void onAccessibilityEvent(AccessibilityEvent event) {
   363→        if (event == null) return;
   364→        CharSequence pkgCs = event.getPackageName();
   365→        if (pkgCs == null) return;
   366→        String pkg = pkgCs.toString();
   367→        if (pkg.equals(getPackageName())) return;
   368→        if (isSystemNoise(pkg)) return;
   369→
   370→        long now = System.currentTimeMillis();
   371→        energyEvents++;
   372→
   373→        switch (event.getEventType()) {
   374→            case AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED: {
   375→                String cls = cs(event.getClassName());
   376→                // v3.6.0 根窗口追踪：新进入前台的应用，其首个窗口 = 入口窗
   377→                //（冷启动开屏 / 从桌面·其他应用回前台的首窗）——入口窗上
   378→                // 按返回键 = 退出应用，安全门据此禁一切返回键兜底
   379→                boolean entering = !pkg.equals(lastFgPkg);
   380→                lastFgPkg = pkg;
   381→                if (reboundChainJump(pkg, now)) break;
   382→                if (reboundLanding(pkg, cls, now)) break;
   383→                // v3.5.0 复验通过：上一窗点过东西且窗口已切走 → 广告已关
   384→                Integer prevTries = clickTries.get(pkg);
   385→                if (prevTries != null && prevTries > 0 && !cls.equals(winClassOf(pkg))) {
   386→                    appendLog("✓ 复验通过（" + shortPkg(pkg) + "）：广告窗已关闭"
   387→                            + (prevTries > 1 ? "（共 " + prevTries + " 次尝试，重试梯子兜底）" : ""));
   388→                    // v3.6.0 毒链① 根治：广告已关，学习窗快断资格同步作废
   389→                    //（防止后续窗口还拿着旧资格按返回键）
   390→                    winClsLearned.remove(pkg);
   391→                }
   392→                windowAt.put(pkg, now);
   393→                winClass.put(pkg, cls);
   394→                if (entering) rootWinCls.put(pkg, cls); // v3.6.0：记录入口窗
   395→                backTries.put(pkg, 0);
   396→                clickTries.put(pkg, 0); // v3.5.0：新窗重置复验计数
   397→                cdLastN.remove(pkg);   // v3.8.0：新窗重置倒计时锚点
   398→                cdAnchorAt.remove(pkg);
   399→                cdWaitUntil.remove(pkg);
   400→                gaveUp.put(pkg, Boolean.FALSE); // v3.6.0：新窗重置收手标记
   401→                winClsLearned.remove(pkg);
   402→                if (isDialogClass(cls) && !isOemUtilityPkg(pkg)) {
   403→                    popupUntil.put(pkg, now + POPUP_WINDOW_MS);
   404→                }
   405→                if (AdDecisionCore.isAdActivityClass(cls)) {
   406→                    adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   407→                    appendLog("🎯 SDK 广告窗识别（" + shortPkg(pkg) + "）：" + cls
   408→                            + " → 500ms 内无真跳过键将温和直断");
   409→                } else if (learnedClassHit(pkg, cls)) {
   410→                    adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   411→                    winClsLearned.put(pkg, Boolean.TRUE);
   412→                }
   413→                bumpScanChain(pkg);
   414→                break;
   415→            }
   416→            case AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED: {
   417→                if (now - winAt(pkg) < WINDOW_MS) {
   418→                    nudgeScan(pkg, now);
   419→                    break;
   420→                }
   421→                if (now < popupUntilGet(pkg)) {
   422→                    nudgePopupScan(pkg, now);
   423→                    break;
   424→                }
   425→                detectPopupBurst(pkg, now);
   426→                break;
   427→            }
   428→            case AccessibilityEvent.TYPE_VIEW_CLICKED: {
   429→                if (isOwnAutoClick(pkg, now)) return;
   430→                learnFromManualClick(pkg, event, now);
   431→                observeMisclick(pkg, event, now);
   432→                break;
   433→            }
   434→        }
   435→    }

   631→    /* ───────── 扫描与决策执行（逻辑全部在 AdDecisionCore） ───────── */
   632→
   633→    private void tryScan(String pkg, long winStart, long atMs) {
   634→        if (instance == null) return;
   635→        long now = System.currentTimeMillis();
   636→        long elapsed = now - winStart;
   637→        boolean splashLive = learningEnabled() && elapsed <= WINDOW_MS;
   638→        boolean popupLive = popupShieldEnabled() && now < popupUntilGet(pkg) && popupCooldownOK(pkg, now);
   639→        if (!splashLive && !popupLive) return;
   640→
   641→        try {
   642→            /* 多窗口扫描：活动窗口优先，其次 getWindows() 同包窗口（弹窗/覆盖层） */
   643→            AccessibilityNodeInfo[] roots = collectRoots(pkg);
   644→            if (roots.length == 0) return;
   645→
   646→            Long lastClick = lastClickAt.get(pkg);
   647→            boolean clickCoolOK = lastClick == null || now - lastClick >= COOLDOWN_MS;
   648→            // v3.5.0：学习位置指纹（手动教过的坐标）与复验次数
   649→            int[] tapHint = learnedPos(pkg);
   650→            Integer triesBox = clickTries.get(pkg);
   651→            int clickAttempts = triesBox == null ? 0 : triesBox.intValue();
   652→
   653→            for (int ri = 0; ri < roots.length; ri++) {
   654→                energyScans++;
   655→                Ctx ctx = new Ctx();
   656→                ctx.screenW = screenWidth;
   657→                ctx.screenH = screenHeight;
   658→                ctx.elapsedMs = Math.max(0, elapsed - ri * 30);
   659→                ctx.splashLive = splashLive;
   660→                ctx.popupLive = popupLive;
   661→                ctx.guard = AdDecisionCore.isGuardPkg(pkg);
   662→                // v3.6.0 毒链① 根治：sdkAdWin 只认【当前窗口类名】是 SDK 广告窗
   663→                //（v3.5.0 的 12s TTL 让主页还拿着 SDK 资格按返回键 → 退出应用）
   664→                ctx.sdkAdWin = AdDecisionCore.isAdActivityClass(winClassOf(pkg));
   665→                Boolean learned = winClsLearned.get(pkg);
   666→                ctx.learnedCls = learned != null && learned.booleanValue();
   667→                // v3.6.0 毒链② 根治：根窗口（入口窗）标记——返回键=退出应用，
   668→                // L3/L4.5/L5 决策核心禁用 + execute() 双保险拦截
   669→                ctx.rootWindow = winClassOf(pkg).equals(rootWinClsOf(pkg));
   670→                ctx.clickAttempts = clickAttempts;
   671→                ctx.tapHintX = tapHint[0];
   672→                ctx.tapHintY = tapHint[1];
   673→                ctx.budgetNodes = splashLive ? 600 : 280; // 开屏窗预算扩容（重树）
   674→
   675→                ScanFind f = AdDecisionCore.walk(new NodeAdapter(roots[ri]), ctx);
   676→                if (f.foundSkip || f.gyroHint || f.fakeSkip || f.hotZone || ctx.sdkAdWin || ctx.learnedCls) {
   677→                    Long old = adFeatureUntil.get(pkg);
   678→                    if (old == null || now > old) adFeatureUntil.put(pkg, now + CHAIN_GUARD_MS);
   679→                }
   680→                /* v3.8.0 倒计时门控（issue#10 二轮·喜马拉雅）：穿山甲类 SDK 倒计时
   681→                 * 结束前跳过键不响应——「已点击×4 不跳转」真凶之一（提前点全白点，
   682→                 * 还各占 1.2s 冷却）。文案数字变化→重新锚定；首击未关→等到
   683→                 * 锚点+剩余秒数-0.5s 再出手；倒计时走完（文案无数字）→立即恢复出手 */
   684→                boolean cdWait = false;
   685→                int cdN = f.countdownN;
   686→                if (cdN > 0) {
   687→                    Integer lastN = cdLastN.get(pkg);
   688→                    if (lastN == null || lastN.intValue() != cdN) {
   689→                        cdLastN.put(pkg, cdN);
   690→                        cdAnchorAt.put(pkg, now);
   691→                    }
   692→                    if (clickAttempts >= 1) {
   693→                        Long anchor = cdAnchorAt.get(pkg);
   694→                        long until = (anchor == null ? now : anchor.longValue())
   695→                                + cdN * 1000L - AdDecisionCore.COUNTDOWN_EXTRA_MS;
   696→                        if (now < until) {
   697→                            cdWait = true;
   698→                            if (cdWaitUntil.put(pkg, until) == null) {
   699→                                appendLog("⏳ 倒计时门控（" + shortPkg(pkg) + "）：跳过键 "
   700→                                        + cdN + "s 后才激活（提前点=白点）→ "
   701→                                        + (until - now) + "ms 后出手即命中");
   702→                            }
   703→                        }
   704→                    }
   705→                } else {
   706→                    Long w = cdWaitUntil.get(pkg);
   707→                    if (w != null) {
   708→                        cdWaitUntil.remove(pkg);
   709→                        appendLog("⏳ 倒计时结束（" + shortPkg(pkg) + "）：跳过键已激活 → 出手");
   710→                    }
   711→                }
   712→                if (cdWait) continue;
   713→                Decision d = AdDecisionCore.decide(f, ctx);
   714→                if (d.action.equals(AdDecisionCore.A_NONE)) continue;
   715→                if (execute(pkg, d, f, ctx, clickCoolOK, now)) return; // 命中即熔断
   716→            }
   717→        } catch (Exception ignored) {
   718→        }
   719→    }

   721→    /** 收集同包可扫描窗口根节点（活动窗口优先，最多 3 个） */
   722→    private AccessibilityNodeInfo[] collectRoots(String pkg) {
   723→        java.util.ArrayList<AccessibilityNodeInfo> out = new java.util.ArrayList<>();
   724→        AccessibilityNodeInfo active = getRootInActiveWindow();
   725→        if (active != null) {
   726→            CharSequence rp = active.getPackageName();
   727→            if (rp != null && pkg.contentEquals(rp)) out.add(active);
   728→        }
   729→        if (out.isEmpty()) {
   730→            try {
   731→                List<AccessibilityWindowInfo> wins = getWindows();
   732→                if (wins != null) {
   733→                    for (AccessibilityWindowInfo w : wins) {
   734→                        if (out.size() >= MAX_WINDOWS_SCAN) break;
   735→                        CharSequence wp = w == null ? null : w.getRoot() == null ? null : w.getRoot().getPackageName();
   736→                        if (wp != null && pkg.contentEquals(wp)) {
   737→                            AccessibilityNodeInfo r = w.getRoot();
   738→                            if (r != null) out.add(r);
   739→                        }
   740→                    }
   741→                }
   742→            } catch (Exception ignored) {
   743→            }
   744→        }
   745→        return out.toArray(new AccessibilityNodeInfo[0]);
   746→    }

   748→    /** 执行决策；返回 true = 已出手（熔断扫描链） */
   749→    private boolean execute(String pkg, Decision d, ScanFind f, Ctx ctx, boolean clickCoolOK, long now) {
   750→        String how = d.action + "/" + d.reason;
   751→        lastGesturePkg = pkg; // v3.7.0：手势取消诊断日志用
   752→        // v3.6.0：点击梯子预算——同窗点击 ≥MAX_CLICKS_PER_WIN 即优雅收手
   753→        //（绝不无限重试，更绝不升级返回键；等用户手动跳过，手动点击仍会被学习）
   754→        if ((d.action.equals(AdDecisionCore.A_CLICK_BOUNDS) || d.action.equals(AdDecisionCore.A_CLOSE)
   755→                || d.action.equals(AdDecisionCore.A_TAP_POINT))
   756→                && triesOfClick(pkg) >= AdDecisionCore.MAX_CLICKS_PER_WIN) {
   757→            Boolean done = gaveUp.get(pkg);
   758→            if (done == null || !done.booleanValue()) {
   759→                gaveUp.put(pkg, Boolean.TRUE);
   760→                appendLog("🤲 点击梯子已用尽（" + shortPkg(pkg) + "）：" + AdDecisionCore.MAX_CLICKS_PER_WIN
   761→                        + " 次尝试未关闭广告 → 优雅收手（本窗不再动作；你的手动跳过仍会被学习）");
   762→            }
   763→            return false;
   764→        }
   765→        if (d.action.equals(AdDecisionCore.A_BACK)) {
   766→            // v3.6.0 根窗口安全门（双保险，决策核心已先拦一道）：
   767→            // 入口窗上按返回键 = 退出应用（喜马拉雅「再按一次退出到桌面」事故）
   768→            if (ctx.rootWindow && !ctx.sdkAdWin) {

   863→    /** v3.8.0：手势真实触点【首选】（GKD clickCenter 方法论）。
   864→     *  v3.4.0~v3.7.0 以 performAction 优先：无障碍代点返回 true 也可能被应用
   865→     *  无声吞掉——issue#10 二轮实锤「4 次点击成功，广告纹丝不动，手动点同一
   866→     *  按钮立刻生效」。performAction 走无障碍点击通道，广告 SDK/自绘控件普遍
   867→     *  只认真实触点。改为先 dispatchGesture 点【原标签控件】中心（=手动点击的
   868→     *  完全等价物），派发失败（系统手势限流）才降级 performAction 兜底。
   869→     *  假跳过候选（d.node=null）只走手势——大容器祖先 performAction=点开广告 */
   870→    private boolean clickNode(Decision d) {
   871→        lastGesture = false;
   872→        if (d.bounds != null) {
   873→            int cx = (d.bounds[0] + d.bounds[2]) / 2;
   874→            int cy = (d.bounds[1] + d.bounds[3]) / 2;
   875→            if (gestureTap(cx, cy)) {
   876→                lastGesture = true;
   877→                energyGestures++;
   878→                return true;
   879→            }
   880→        }
   881→        if (d.node instanceof NodeAdapter) {
   882→            try {
   883→                if (((NodeAdapter) d.node).n.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
   884→                    return true;
   885→                }
   886→            } catch (Exception ignored) {
   887→            }
   888→        }
   889→        return false;
   890→    }

   913→    /** dispatchGesture 坐标点击（API 24+；真实触点，performAction 拦不住）。
   914→     *  v3.7.0 根因修复（issue#10：「已点击跳过×4 无效果」）：
   915→     *   ① 路径非零长度——单点路径（只有 moveTo）在部分 ROM 上不触发点击
   916→     *    （业界验证：需 lineTo 形成真实触点轨迹）；
   917→     *   ② 起始延迟 40ms——给前一帧渲染/事件处理留时间，立即派发易被丢弃；
   918→     *   ③ 坐标边界钳制——越界坐标（学习位置/旧指纹）直接被系统丢弃；
   919→     *   ④ 结果回调——被系统取消不再静默（诊断日志供下轮迭代）。
   920→     *   ⑤ 前提：accessibility_service_config 已补 canPerformGestures=true
   921→     *    （v3.6.0 及之前【缺失】——手势从未真正触达屏幕，喜马拉雅复验重试
   922→     *    全部无效的头号根因）
   923→     *   ⑥ 触点时长 60→110ms——更接近真人触点（80-150ms），防 SDK 触点时长校验 */
   924→    private boolean gestureTap(int x, int y) {
   925→        try {
   926→            if (x <= 0 || y <= 0) return false;
   927→            if (screenWidth > 0 && x >= screenWidth) x = screenWidth - 2; // 边界钳制
   928→            if (screenHeight > 0 && y >= screenHeight) y = screenHeight - 2;
   929→            Path p = new Path();
   930→            p.moveTo(x, y);
   931→            p.lineTo(x + 1, y); // 非零长度路径：单点路径部分 ROM 不触发点击
   932→            GestureDescription gd = new GestureDescription.Builder()
   933→                    .addStroke(new GestureDescription.StrokeDescription(p, 40, 110))
   934→                    .build();
   935→            return dispatchGesture(gd, new TapGestureCallback(lastGesturePkg), null);
   936→        } catch (Throwable t) {
   937→            return false;
   938→        }
   939→    }

   998→                    + " → 返回键送回原应用（" + (now - winAt(host)) + "ms 内）");

  1043→        if (now - winAt(pkg) > LEARN_WINDOW_MS) return;
  1044→

  1046→        String viewId = "";
  1047→        AccessibilityNodeInfo srcNode = null;
  1048→        try {
  1049→            StringBuilder sb = new StringBuilder();

  1057→            if (src != null) {
  1058→                srcNode = src;
  1059→                if (text.length() == 0) text = cs(src.getContentDescription());
  1060→                viewId = cs(src.getViewIdResourceName());
  1061→            }
  1062→        } catch (Exception ignored) {
  1063→        }
  1064→        if (!AdDecisionCore.matchesSkip(text) && !AdDecisionCore.isCloseText(text)) {
  1065→            // v3.5.0 位置学习：图片跳过键零文本也要学（真机痛点：「手动记录两次
  1066→            // 还是不能跳过」——v3.4.0 只认文本，图片键永远学不到）
  1067→            learnPositionFromManualClick(pkg, srcNode, now);
  1068→            return;
  1069→        }
  1070→
  1071→        synchronized (lock) {
  1072→            try {
  1073→                JSONObject st = loadState();
  1074→                JSONObject rules = st.optJSONObject("rules");
  1075→                if (rules == null) {
  1076→                    rules = new JSONObject();
  1077→                    st.put("rules", rules);
  1078→                }
  1079→                JSONArray list = rules.optJSONArray(pkg);
  1080→                if (list == null) {
  1081→                    list = new JSONArray();
  1082→                    rules.put(pkg, list);
  1083→                }
  1084→                boolean merged = false;
  1085→                for (int i = 0; i < list.length(); i++) {
  1086→                    JSONObject r = list.optJSONObject(i);
  1087→                    if (r == null) continue;

  1089→                    boolean sameText = text.length() > 0 && text.equals(r.optString("text", ""));
  1090→                    if (sameId || sameText) {
  1091→                        r.put("manual", r.optInt("manual", 0) + 1);
  1092→                        r.put("ts", now);
  1093→                        if (r.optString("viewId", "").length() == 0 && viewId.length() > 0) {
  1094→                            r.put("viewId", viewId);
  1095→                        }
  1096→                        merged = true;
  1097→                        break;
  1098→                    }
  1099→                }
  1100→                if (!merged) {
  1101→                    if (list.length() >= MAX_RULES_PER_APP) return;
  1102→                    JSONObject r = new JSONObject();
  1103→                    r.put("text", text);
  1104→                    r.put("viewId", viewId);
  1105→                    r.put("manual", 1);
  1106→                    r.put("auto", 0);
  1107→                    r.put("ts", now);
  1108→                    list.put(r);
  1109→                }
  1110→                bumpManualStat(st, pkg);
  1111→                saveState(st);
  1112→            } catch (Exception ignored) {
  1113→            }
  1114→        }
  1115→        /* v3.7.0（issue#10 喜马拉雅根因）：文本键手动点击【同时学位置】——
  1116→         * 用户手指点通过的坐标 = 真实触点真正生效的位置（文本候选可能点不中，
  1117→         * 教过的位置下次同窗 500ms 直点，优先于文本候选盲点） */

  1120→    }

  1140→    private void learnClass(String pkg, String cls) {

  1142→        // v3.6.0 毒链① 断源：只沉淀真正的广告窗类（SDK 广告 Activity）。

  1576→    /** 自愈重绑：true=已发起（重绑异步生效）；未授权返回 false */
  1577→    public static boolean selfHealBind(Context ctx) {
  1578→        if (!hasSecureWrite(ctx)) return false;
  1579→        try {
  1580→            final android.content.ContentResolver cr = ctx.getContentResolver();
  1581→            final String key = Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES;
  1582→            final ComponentName cn = new ComponentName(ctx, AdSkipService.class);
  1583→            final String flat = cn.flattenToString();
  1584→            final String flatShort = cn.flattenToShortString(); // v3.8.1：兼容短格式条目（adb 手动写入场景）
  1585→            final String cur = Settings.Secure.getString(cr, key);
  1586→            // v3.8.1：登记判断双格式匹配——全格式或短格式条目任一命中即视为已登记，
  1587→            // 否则短格式场景会被误判「未登记」再追加产生冗余双条目
  1588→            if (cur == null || !(cur.contains(flat) || cur.contains(flatShort))) {
  1589→                // 未登记（被系统关闭/覆盖安装后清除）→ 直接写回开启
  1590→                String next = (cur == null || cur.trim().isEmpty()) ? flat : cur + ":" + flat;
  1591→                Settings.Secure.putString(cr, key, next);
  1592→                Settings.Secure.putString(cr, Settings.Secure.ACCESSIBILITY_ENABLED, "1");
  1593→                return true;
  1594→            }
  1595→            // 已登记但进程已死 → 摘除再写回，强制系统重绑（bind 会拉起进程）
  1596→            StringBuilder keep = new StringBuilder();
  1597→            for (String s : cur.split(":")) {
  1598→                String t = s.trim();
  1599→                if (t.length() == 0 || t.equals(flat) || t.equals(flatShort)) continue; // v3.8.1：双格式都摘除
  1600→                if (keep.length() > 0) keep.append(':');
  1601→                keep.append(t);
  1602→            }
  1603→            String without = keep.toString();
  1604→            Settings.Secure.putString(cr, key, without); // 摘除 → 系统解绑
  1605→            Thread.sleep(400);                            // 等解绑落地
  1606→            Settings.Secure.putString(cr, key,            // 写回 → 系统 bind（拉起进程）
  1607→                    without.isEmpty() ? flat : without + ":" + flat);
  1608→            return true;
  1609→        } catch (Throwable t) {
  1610→            return false;
  1611→        }
  1612→    }
